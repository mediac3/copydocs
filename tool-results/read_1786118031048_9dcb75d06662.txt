   578→    setLoading(true)

   580→      const params = new URLSearchParams()
   581→
   582→      if (category !== 'all') params.set('category', category)
   583→      if (legalArea !== 'all') params.set('legalArea', legalArea)
   584→      if (audience !== 'all') params.set('audience', audience)
   585→      if (priceFilter === 'free') params.set('price', 'free')
   586→      if (priceFilter === 'paid') params.set('price', 'paid')
   587→      if (debouncedSearch.trim()) params.set('search', debouncedSearch.trim())
   588→
   589→      const query = params.toString() ? `?${params.toString()}` : ''
   590→      const headers: HeadersInit = {}
   591→      if (user) headers['x-user-id'] = user.id
   592→      const res = await fetch(`/api/templates${query}`, { headers })
   593→
   594→      if (res.ok) {
   595→        const data = await res.json()
   596→        setTemplates(Array.isArray(data) ? data : data.templates || [])
   597→      }
   598→    } catch {
   599→      // keep empty on error
   600→    } finally {
   601→      setLoading(false)
   602→    }
   603→  }, [user, category, legalArea, audience, priceFilter, debouncedSearch])
   604→
   605→  useEffect(() => {
   606→    fetchTemplates()
   607→  }, [fetchTemplates])
   608→
   609→  /* ---- Handlers ---- */
   610→  const handleSelectTemplate = useCallback((template: Template) => {
   611→    setSelectedTemplate(template)
   612→    setDialogOpen(true)
   613→  }, [])
   614→

   617→    if (!open) {
   618→      // Keep template reference for animation but can clear if desired
   619→    }
   620→  }, [])
   621→
   622→  const hasActiveFilters =
   623→    searchInput.trim() !== '' ||
   624→    category !== 'all' ||
   625→    legalArea !== 'all' ||
   626→    audience !== 'all' ||
   627→    priceFilter !== 'all'
   628→
   629→  /* ---- Render ---- */

   634→        <div className="border-b border-white/5 bg-[#0F1D32]/80 backdrop-blur-sm">
   635→          <div className="mx-auto flex max-w-7xl items-center justify-between px-4 py-3 sm:px-6 lg:px-8">
   636→            <div className="flex items-center gap-2">
   637→              <Scale className="h-5 w-5 text-[#C9A94E]" />
   638→              <span className="text-sm font-bold text-white">LexDoc</span>
   639→            </div>
   640→            <button
   641→              onClick={() => window.location.reload()}
   642→              className="text-xs font-medium text-[#C9A94E] transition-colors hover:text-[#D4B965]"
   643→            >
   644→              Iniciar sesion
   645→            </button>
   646→          </div>
   647→        </div>

   650→        {/* ---------- Header ---------- */}
   651→        <div>
   652→          <h1 className="text-2xl font-bold tracking-tight text-white sm:text-3xl">
   653→            {isVisitor ? 'Genera tu Documento Legal' : 'Catálogo de Plantillas'}
   654→          </h1>
   655→          <p className="mt-1 text-sm text-white/50">
   656→            {isVisitor
   657→              ? 'Selecciona una plantilla, completa los campos y envia tu solicitud por WhatsApp.'
   658→              : 'Explora y genera documentos legales colombianos con plantillas profesionales.'}
   659→          </p>
   660→        </div>
   661→
   662→        {/* ---------- Search bar ---------- */}
   663→        <div className="relative">
   664→          <Search className="absolute top-1/2 left-3 h-4 w-4 -translate-y-1/2 text-white/30" />
   665→          <Input
   666→            placeholder="Buscar plantillas por nombre, tipo o palabra clave..."
   667→            value={searchInput}
   668→            onChange={(e) => setSearchInput(e.target.value)}
   669→            className="h-11 w-full border-white/10 bg-[#0F1D32] pl-10 pr-4 text-sm text-white placeholder:text-white/30 focus-visible:border-[#C9A94E]/40 focus-visible:ring-[#C9A94E]/20"
   670→          />
   671→        </div>
   672→
   673→        {/* ---------- Filters row ---------- */}
   674→        <div className="flex flex-col gap-3 sm:flex-row sm:flex-wrap sm:items-center sm:gap-3">
   675→          {/* Category */}
   676→          <Select value={category} onValueChange={setCategory}>
   677→            <SelectTrigger
   678→              className="w-full border-white/10 bg-[#0F1D32] text-white/70 text-xs sm:w-[155px]"
   679→              size="sm"
   680→            >
   681→              <SelectValue placeholder="Categoría" />
   682→            </SelectTrigger>
   683→            <SelectContent className="border-white/10 bg-[#0F1D32]">
   684→              <SelectItem value="all" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   685→                Todas
   686→              </SelectItem>
   687→              <SelectItem value="Contratos" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   688→                Contratos
   689→              </SelectItem>
   690→              <SelectItem value="Actas" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   691→                Actas

   700→          <Select value={legalArea} onValueChange={setLegalArea}>
   701→            <SelectTrigger
   702→              className="w-full border-white/10 bg-[#0F1D32] text-white/70 text-xs sm:w-[165px]"

   704→            >

   706→            </SelectTrigger>
   707→            <SelectContent className="border-white/10 bg-[#0F1D32]">

   719→              </SelectItem>

   746→
   747→          {/* Price */}
   748→          <Select value={priceFilter} onValueChange={setPriceFilter}>
   749→            <SelectTrigger
   750→              className="w-full border-white/10 bg-[#0F1D32] text-white/70 text-xs sm:w-[135px]"
   751→              size="sm"
   752→            >
   753→              <SelectValue placeholder="Precio" />

   767→
   768→          {/* Spacer */}

   770→

   772→          <ToggleGroup

   819→
   820→        {/* ---------- Content ---------- */}
   821→        {loading ? (

   823→            <div className="grid grid-cols-1 gap-4 md:grid-cols-2 xl:grid-cols-3">
   824→              {Array.from({ length: 6 }).map((_, i) => (
   825→                <SkeletonCard key={i} />
   826→              ))}
   827→            </div>
   828→          ) : (
   829→            <div className="flex flex-col gap-3">
   830→              {Array.from({ length: 6 }).map((_, i) => (
   831→                <SkeletonRow key={i} />
   832→              ))}
   833→            </div>
   834→          )
   835→        ) : templates.length === 0 ? (
   836→          <EmptyState hasFilters={hasActiveFilters} />
   837→        ) : viewMode === 'grid' ? (
   838→          <div className="grid grid-cols-1 gap-4 md:grid-cols-2 xl:grid-cols-3">
   839→            {templates.map((t) => (
   840→              <TemplateCard key={t.id} template={t} onSelect={handleSelectTemplate} />
   841→            ))}
   842→          </div>
   843→        ) : (
   844→          <div className="flex flex-col gap-3">
   845→            {templates.map((t) => (
   846→              <TemplateRow key={t.id} template={t} onSelect={handleSelectTemplate} />
   847→            ))}
   848→          </div>
   849→        )}
   850→
   851→        {/* ---------- Bottom CTA (logged-in only) ---------- */}
   852→        {!isVisitor && (
   853→        <div className="flex items-center justify-center pt-4">
   854→          <button
   855→            className="group flex items-center gap-3 rounded-xl border border-[#C9A94E]/15 bg-[#0F1D32]/60 px-6 py-4 text-left backdrop-blur-sm transition-all hover:border-[#C9A94E]/30 hover:bg-[#C9A94E]/5 hover:shadow-lg hover:shadow-[#C9A94E]/5"
   856→          >
   857→            <div className="flex h-11 w-11 shrink-0 items-center justify-center rounded-lg bg-[#C9A94E]/15 text-[#C9A94E] transition-colors group-hover:bg-[#C9A94E]/25">
   858→              <Plus className="h-5 w-5" />
   859→            </div>
   860→            <div className="min-w-0 flex-1">
   861→              <p className="text-sm font-semibold text-white/80 group-hover:text-white">
   862→                No encuentras lo que buscas?
   863→              </p>
   864→              <p className="text-xs text-white/40">
   865→                Solicita un documento personalizado y nuestro equipo lo preparara para ti.
   866→              </p>
   867→            </div>
   868→            <ArrowRight className="h-4 w-4 shrink-0 text-white/20 transition-transform group-hover:translate-x-0.5 group-hover:text-[#C9A94E]" />
   869→          </button>
   870→        </div>

   872→      </div>
   873→
   874→      {/* ---------- Detail Dialog ---------- */}
   875→      <TemplateDetailDialog

   877→        open={dialogOpen}
   878→        onOpenChange={handleDialogClose}
   879→      />
   880→    </main>
