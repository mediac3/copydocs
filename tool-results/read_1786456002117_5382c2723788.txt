   163→/* ---- Media helper: stores type + dimensions alongside base64 data ---- */
   164→interface MediaContent {
   165→  type: 'text' | 'image'
   166→  text?: string
   167→  dataUrl?: string
   168→  width?: number
   169→  height?: number
   170→}
   171→
   172→function parseMedia(raw: string | null | undefined): MediaContent {
   173→  if (!raw) return { type: 'text', text: '' }
   174→  if (raw.startsWith('data:image')) {
   175→    try {
   176→      const parsed = JSON.parse(raw)
   177→      if (parsed && parsed.type === 'image' && parsed.dataUrl) return parsed
   178→    } catch { /* not JSON, treat as raw base64 */ }
   179→    return { type: 'image', dataUrl: raw, width: 468, height: 60 }
   180→  }
   181→  return { type: 'text', text: raw }
   182→}
   183→
   184→function encodeMedia(m: MediaContent): string {
   185→  if (m.type === 'text') return m.text || ''
   186→  return JSON.stringify({ type: 'image', dataUrl: m.dataUrl, width: m.width, height: m.height })
   187→}
   188→
   189→function isMediaImage(m: MediaContent): boolean {
   190→  return m.type === 'image' && !!m.dataUrl
   191→}

   368→function MediaFieldEditor({
   369→  label, rawValue, onChange, defaultW, defaultH, maxH, hint
   370→}: {
   371→  label: string
   372→  rawValue: string
   373→  onChange: (encoded: string) => void
   374→  defaultW: number
   375→  defaultH: number
   376→  maxH: number
   377→  hint: string
   378→}) {
   379→  const [mode, setMode] = useState<'text' | 'image'>(() => {
   380→    if (!rawValue) return 'text'
   381→    if (rawValue.startsWith('data:image')) return 'image'
   382→    try { const p = JSON.parse(rawValue); if (p?.type === 'image') return 'image' } catch {}
   383→    return 'text'
   384→  })
   385→  const [imgData, setImgData] = useState<string>('')
   386→  const [imgW, setImgW] = useState(defaultW)
   387→  const [imgH, setImgH] = useState(defaultH)
   388→  const [textVal, setTextVal] = useState('')
   389→  const initialized = useRef(false)
   390→
   391→  // Init from rawValue on mount
   392→  useEffect(() => {
   393→    if (initialized.current) return
   394→    initialized.current = true
   395→    if (!rawValue) return
   396→    if (rawValue.startsWith('data:image')) {
   397→      setImgData(rawValue)
   398→    } else {
   399→      try {
   400→        const p = JSON.parse(rawValue)
   401→        if (p?.type === 'image') {
   402→          setImgData(p.dataUrl || '')
   403→          if (p.width) setImgW(p.width)
   404→          if (p.height) setImgH(p.height)
   405→        } else {
   406→          setTextVal(typeof p?.text === 'string' ? p.text : rawValue)
   407→        }
   408→      } catch {
   409→        setTextVal(rawValue)
   410→      }
   411→    }
   412→  }, [])
   413→
   414→  const switchMode = (t: 'text' | 'image') => {
   415→    setMode(t)
   416→    if (t === 'text') {
   417→      setTextVal('')
   418→      onChange('')
   419→    } else {
   420→      setImgData('')
   421→      setImgW(defaultW)
   422→      setImgH(defaultH)
   423→      onChange(JSON.stringify({ type: 'image', dataUrl: '', width: defaultW, height: defaultH }))
   424→    }
   425→  }
   426→
   427→  const handleFile = (e: React.ChangeEvent<HTMLInputElement>) => {
   428→    const file = e.target.files?.[0]
   429→    if (!file) return
   430→    if (file.size > 2 * 1024 * 1024) { toast.error('La imagen no puede superar 2 MB'); return }
   431→    const reader = new FileReader()
   432→    reader.onload = () => {
   433→      const dataUrl = reader.result as string
   434→      const img = new window.Image()
   435→      img.onload = () => {
   436→        const ratio = img.width / img.height
   437→        let h = maxH
   438→        let w = Math.round(h * ratio)
   439→        if (w > 468) { w = 468; h = Math.round(w / ratio) }
   440→        setImgData(dataUrl)
   441→        setImgW(w)
   442→        setImgH(h)
   443→        onChange(JSON.stringify({ type: 'image', dataUrl, width: w, height: h }))
   444→      }
   445→      img.onerror = () => {
   446→        setImgData(dataUrl)
   447→        setImgW(defaultW)
   448→        setImgH(defaultH)
   449→        onChange(JSON.stringify({ type: 'image', dataUrl, width: defaultW, height: defaultH }))
   450→      }
   451→      img.src = dataUrl
   452→    }
   453→    reader.readAsDataURL(file)
   454→    e.target.value = ''
   455→  }
   456→
   457→  const removeImage = () => {
   458→    setImgData('')
   459→    setImgW(defaultW)
   460→    setImgH(defaultH)
   461→    onChange(JSON.stringify({ type: 'image', dataUrl: '', width: defaultW, height: defaultH }))
   462→  }
   463→
   464→  return (
   465→    <div className="grid gap-2">
   466→      <Label className="text-white/70">{label}</Label>
   467→      <div className="flex gap-1 mb-1">
   468→        {(['text', 'image'] as const).map(t => (
   469→          <button
   470→            key={t}
   471→            type="button"
   472→            onClick={() => switchMode(t)}
   473→            className={`flex-1 rounded-md px-3 py-1.5 text-[11px] font-medium transition-colors ${
   474→              mode === t
   475→                ? 'bg-[#28A745]/20 text-[#28A745] border border-[#28A745]/30'
   476→                : 'bg-white/5 text-white/40 border border-white/10 hover:bg-white/10'
   477→            }`}
   478→          >
   479→            {t === 'text' ? 'Texto' : 'Imagen'}
   480→          </button>
   481→        ))}
   482→      </div>
   483→      {mode === 'text' ? (
   484→        <Textarea
   485→          value={textVal}
   486→          onChange={(e) => { setTextVal(e.target.value); onChange(e.target.value) }}
   487→          placeholder={`Texto del ${label.toLowerCase()}...`}
   488→          rows={3}
   489→          className="border-white/10 bg-white/5 text-white placeholder:text-white/30"
   490→        />
   491→      ) : (
   492→        <div className="space-y-2">
   493→          {imgData && (
   494→            <div className="flex items-center justify-center rounded-lg border border-dashed border-white/15 bg-white/[0.02] p-3">
   495→              <img src={imgData} alt="Preview" className="max-h-24 max-w-full object-contain rounded" />
   496→            </div>
   497→          )}
   498→          <label className="flex cursor-pointer items-center justify-center gap-2 rounded-md border border-white/10 bg-white/5 px-3 py-2.5 text-xs text-white/50 hover:bg-white/10 hover:text-white/70 transition-colors">
   499→            <Upload className="h-3.5 w-3.5" />
   500→            {imgData ? 'Cambiar imagen' : 'Subir imagen (PNG, JPG, WebP)'}
   501→            <input type="file" accept="image/png,image/jpeg,image/webp" className="hidden" onChange={handleFile} />
   502→          </label>
   503→          <div className="grid grid-cols-2 gap-2">
   504→            <div className="grid gap-1">
   505→              <span className="text-[9px] uppercase tracking-wider text-white/25">Ancho (px)</span>
   506→              <Input
   507→                type="number"
   508→                value={imgW}
   509→                onChange={(e) => {
   510→                  const w = parseInt(e.target.value) || defaultW
   511→                  setImgW(w)
   512→                  onChange(JSON.stringify({ type: 'image', dataUrl: imgData, width: w, height: imgH }))
   513→                }}
   514→                className="h-7 border-white/10 bg-white/5 text-white text-xs"
   515→              />
   516→            </div>
   517→            <div className="grid gap-1">
   518→              <span className="text-[9px] uppercase tracking-wider text-white/25">Alto (px)</span>
   519→              <Input
   520→                type="number"
   521→                value={imgH}
   522→                onChange={(e) => {
   523→                  const h = parseInt(e.target.value) || defaultH
   524→                  setImgH(h)
   525→                  onChange(JSON.stringify({ type: 'image', dataUrl: imgData, width: imgW, height: h }))
   526→                }}
   527→                className="h-7 border-white/10 bg-white/5 text-white text-xs"
   528→              />
   529→            </div>
   530→          </div>
   531→          <p className="text-[10px] text-white/25 text-center">{hint}. Max 2 MB.</p>
   532→          {imgData && (
   533→            <button
   534→              type="button"
   535→              onClick={removeImage}
   536→              className="w-full text-center text-[11px] text-red-400/60 hover:text-red-400 transition-colors"
   537→            >
   538→              Eliminar imagen
   539→            </button>
   540→          )}
   541→        </div>
   542→      )}
   543→    </div>
   544→  )
   545→}

   551→const FIELD_TYPE_OPTIONS: Array<{ value: WizardFieldDef['type']; label: string; icon: React.ElementType }> = [
   552→  { value: 'text', label: 'Texto', icon: Type },
   553→  { value: 'number', label: 'Número', icon: Hash },

   559→
   560→function WizardStepEditor({
   561→  step, stepIndex, allSteps, onChange, onRemove
   562→}: {
   563→  step: WizardStepDef
   564→  stepIndex: number
   565→  allSteps: WizardStepDef[]
   566→  onChange: (updated: WizardStepDef) => void
   567→  onRemove: () => void
   568→}) {
   569→  const [expanded, setExpanded] = useState(true)
   570→
   571→  const updateField = (fi: number, patch: Partial<WizardFieldDef>) => {
   572→    const fields = step.fields.map((f, i) => i === fi ? { ...f, ...patch } : f)
   573→    onChange({ ...step, fields })
   574→  }
   575→
   576→  const removeField = (fi: number) => {
   577→    onChange({ ...step, fields: step.fields.filter((_, i) => i !== fi) })
   578→  }
   579→
   580→  const addField = () => {
   581→    onChange({
   582→      ...step,
   583→      fields: [...step.fields, { key: '', label: '', type: 'text' }]
   584→    })
   585→  }
   586→
