   161→function useDebounce<T>(value: T, delayMs: number): T {
   162→  const [debounced, setDebounced] = useState(value)
   163→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   164→
   165→  useEffect(() => {
   166→    if (timerRef.current) clearTimeout(timerRef.current)
   167→    timerRef.current = setTimeout(() => setDebounced(value), delayMs)
   168→    return () => {
   169→      if (timerRef.current) clearTimeout(timerRef.current)
   170→    }
   171→  }, [value, delayMs])
   172→
   173→  return debounced
   174→}

   584→            disabled={loading}
   585→            className="border-white/10 bg-white/5 text-white/70 hover:bg-white/10 hover:text-white"
   586→          >
   587→            Cancelar
   588→          </AlertDialogCancel>
   589→          <AlertDialogAction
   590→            onClick={(e) => {
   591→              e.preventDefault()
   592→              onConfirm()
   593→            }}
   594→            disabled={loading}
   595→            className="bg-red-600 text-white hover:bg-red-700 focus:ring-red-500"
   596→          >
   597→            {loading ? 'Eliminando…' : 'Eliminar'}
   598→          </AlertDialogAction>
   599→        </AlertDialogFooter>
   600→      </AlertDialogContent>
   601→    </AlertDialog>
   602→  )
   603→}

   606→/*  ContactsPage                                                             */
   607→/* -------------------------------------------------------------------------- */
   608→
   609→export default function ContactsPage() {
   610→  const { user } = useAppStore()
   611→
   612→  /* ---- State ---- */
   613→  const [activeTab, setActiveTab] = useState<TabKey>('all')
   614→  const [searchQuery, setSearchQuery] = useState('')
   615→  const debouncedSearch = useDebounce(searchQuery, 300)
   616→  const [contacts, setContacts] = useState<Contact[]>([])
   617→  const [loading, setLoading] = useState(true)
   618→
   619→  /* Form dialog */
   620→  const [formDialogOpen, setFormDialogOpen] = useState(false)
   621→  const [editingContact, setEditingContact] = useState<Contact | null>(null)
   622→  const [saving, setSaving] = useState(false)
   623→
   624→  /* Delete dialog */
   625→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false)
   626→  const [contactToDelete, setContactToDelete] = useState<Contact | null>(null)
   627→  const [deleting, setDeleting] = useState(false)
   628→
   629→  /* ---- Fetch contacts ---- */
   630→  const fetchContacts = useCallback(async () => {
   631→    if (!user) return
   632→    setLoading(true)
   633→    try {
   634→      const params = new URLSearchParams()
   635→      const tabConfig = TAB_CONFIG[activeTab]
   636→      if (tabConfig.apiValue) {
   637→        params.set('type', tabConfig.apiValue)
   638→      }
   639→      if (debouncedSearch.trim()) {
   640→        params.set('search', debouncedSearch.trim())
   641→      }
   642→      const res = await fetch(`/api/contacts?${params.toString()}`, {
   643→        headers: { 'x-user-id': user.id },
   644→      })
   645→      if (!res.ok) throw new Error('Error fetching contacts')
   646→      const data = await res.json()
   647→      setContacts(Array.isArray(data) ? data : data.contacts || [])
   648→    } catch {
   649→      toast.error('Error al cargar contactos')
   650→    } finally {
   651→      setLoading(false)
   652→    }
   653→  }, [user, activeTab, debouncedSearch])
   654→
   655→  useEffect(() => {
   656→    fetchContacts()
   657→  }, [fetchContacts])
   658→
   659→  /* ---- Client-side search fallback ---- */
   660→  const filteredContacts = contacts.filter((c) => {
   661→    if (!debouncedSearch.trim()) return true
   662→    const q = debouncedSearch.toLowerCase()
   663→    return (
   664→      c.name.toLowerCase().includes(q) ||
   665→      c.contactType.toLowerCase().includes(q) ||
   666→      c.documentNumber.toLowerCase().includes(q) ||
   667→      c.companyName.toLowerCase().includes(q) ||
   668→      c.city.toLowerCase().includes(q) ||
   669→      c.email.toLowerCase().includes(q)
   670→    )
   671→  })
   672→
   673→  /* ---- Create / Update ---- */
   674→  const handleSave = async (formData: ContactFormData) => {
   675→    if (!user) return
   676→    setSaving(true)
   677→    try {
   678→      const isEdit = !!formData.id
   679→      const method = isEdit ? 'PUT' : 'POST'
   680→      const res = await fetch('/api/contacts', {
   681→        method,
   682→        headers: {
   683→          'Content-Type': 'application/json',
   684→          'x-user-id': user.id,
   685→        },
   686→        body: JSON.stringify(formData),
   687→      })
   688→      if (!res.ok) throw new Error('Error saving contact')
   689→      toast.success(
   690→        isEdit
   691→          ? 'Contacto actualizado correctamente'
   692→          : 'Contacto creado correctamente'
   693→      )
   694→      setFormDialogOpen(false)
   695→      setEditingContact(null)
   696→      fetchContacts()
   697→    } catch {
   698→      toast.error('Error al guardar el contacto')
   699→    } finally {
   700→      setSaving(false)
   701→    }
   702→  }
   703→
   704→  /* ---- Delete ---- */
   705→  const handleDelete = async () => {
   706→    if (!user || !contactToDelete) return
   707→    setDeleting(true)
   708→    try {
   709→      const res = await fetch(
   710→        `/api/contacts?id=${contactToDelete.id}`,
   711→        {
   712→          method: 'DELETE',
   713→          headers: { 'x-user-id': user.id },
   714→        }
   715→      )
   716→      if (!res.ok) throw new Error('Error deleting contact')
   717→      toast.success('Contacto eliminado correctamente')
   718→      setDeleteDialogOpen(false)
   719→      setContactToDelete(null)
   720→      fetchContacts()
   721→    } catch {
   722→      toast.error('Error al eliminar el contacto')
   723→    } finally {
   724→      setDeleting(false)
   725→    }
   726→  }
   727→
   728→  /* ---- Handlers ---- */
   729→  const openNewContact = () => {
   730→    setEditingContact(null)
   731→    setFormDialogOpen(true)
   732→  }
   733→
   734→  const openEditContact = (contact: Contact) => {
   735→    setEditingContact(contact)
   736→    setFormDialogOpen(true)
   737→  }
   738→
   739→  const openDeleteContact = (contact: Contact) => {
   740→    setContactToDelete(contact)
   741→    setDeleteDialogOpen(true)
   742→  }
   743→
   744→  /* ---- Render ---- */
   745→  return (
   746→    <div className="flex h-full flex-col overflow-hidden bg-[#0A1628]">
   747→      {/* Header */}
   748→      <div className="flex shrink-0 items-center justify-between border-b border-white/[0.06] px-6 py-4">
   749→        <div>
   750→          <h1 className="text-lg font-semibold text-white">
   751→            Mis Datos
   752→          </h1>
   753→          <p className="mt-0.5 text-xs text-white/40">
   754→            Gestiona tus contactos y partes para documentos legales
   755→          </p>
   756→        </div>
   757→        <Button
   758→          onClick={openNewContact}
   759→          className="bg-[#C9A94E] text-[#0A1628] font-semibold hover:bg-[#B8983E] focus:ring-[#C9A94E]/50"
   760→        >
   761→          <Plus className="mr-1.5 h-4 w-4" />
   762→          Nuevo contacto
   763→        </Button>
   764→      </div>
   765→
   766→      {/* Toolbar: search + tabs */}
   767→      <div className="shrink-0 space-y-3 border-b border-white/[0.06] px-6 py-4">
   768→        {/* Search bar */}
   769→        <div className="relative">
   770→          <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-white/30" />
   771→          <Input
   772→            value={searchQuery}
   773→            onChange={(e) => setSearchQuery(e.target.value)}
   774→            placeholder="Buscar por nombre, documento, ciudad, empresa..."
   775→            className="border-white/10 bg-white/[0.03] pl-9 text-sm text-white placeholder:text-white/25 focus-visible:border-[#C9A94E]/40 focus-visible:ring-[#C9A94E]/20"
   776→          />
   777→        </div>
   778→
   779→        {/* Type filter tabs */}
   780→        <Tabs
   781→          value={activeTab}
   782→          onValueChange={(v) => setActiveTab(v as TabKey)}
   783→        >
   784→          <TabsList className="h-9 bg-white/[0.04] p-0.5">
   785→            {(Object.keys(TAB_CONFIG) as TabKey[]).map((key) => (
   786→              <TabsTrigger
   787→                key={key}
   788→                value={key}
   789→                className="h-7 rounded-md px-3 text-xs data-[state=active]:bg-[#C9A94E]/15 data-[state=active]:text-[#C9A94E] data-[state=active]:shadow-none text-white/40 hover:text-white/60"
   790→              >
   791→                {TAB_CONFIG[key].label}
   792→              </TabsTrigger>
   793→            ))}
   794→          </TabsList>
   795→        </Tabs>
   796→      </div>
   797→
   798→      {/* Contact grid */}
   799→      <div className="flex-1 overflow-y-auto px-6 py-5">
   800→        {loading ? (
   801→          <div className="flex items-center justify-center py-20">
   802→            <div className="h-8 w-8 animate-spin rounded-full border-2 border-[#C9A94E]/30 border-t-[#C9A94E]" />
   803→          </div>
   804→        ) : filteredContacts.length === 0 ? (
   805→          <EmptyState />
   806→        ) : (
   807→          <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-3">
   808→            {filteredContacts.map((contact) => (
   809→              <ContactCard
   810→                key={contact.id}
   811→                contact={contact}
   812→                onEdit={openEditContact}
   813→                onDelete={openDeleteContact}
   814→              />
   815→            ))}
   816→          </div>
   817→        )}
   818→      </div>
   819→
   820→      {/* Form Dialog */}
   821→      <ContactFormDialog
   822→        key={editingContact?.id ?? 'new'}
   823→        open={formDialogOpen}
   824→        onOpenChange={setFormDialogOpen}
   825→        contact={editingContact}
   826→        onSave={handleSave}
   827→        saving={saving}
   828→      />
   829→
   830→      {/* Delete Dialog */}
   831→      <DeleteDialog
   832→        open={deleteDialogOpen}
   833→        onOpenChange={setDeleteDialogOpen}
   834→        contactName={contactToDelete?.name ?? ''}
   835→        onConfirm={handleDelete}
   836→        loading={deleting}
   837→      />
   838→    </div>
   839→  )
   840→}
