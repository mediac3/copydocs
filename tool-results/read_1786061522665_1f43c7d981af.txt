   534→  // Filters
   535→  const [searchInput, setSearchInput] = useState('')
   536→  const [debouncedSearch, setDebouncedSearch] = useState('')
   537→  const [category, setCategory] = useState('all')
   538→  const [legalArea, setLegalArea] = useState('all')
   539→  const [audience, setAudience] = useState('all')
   540→  const [priceFilter, setPriceFilter] = useState('all')
   541→
   542→  // Dialog
   543→  const [selectedTemplate, setSelectedTemplate] = useState<Template | null>(null)
   544→  const [dialogOpen, setDialogOpen] = useState(false)
   545→
   546→  // Debounce timer ref
   547→  const debounceTimerRef = useRef<NodeJS.Timeout | null>(null)
   548→
   549→  /* ---- Debounced search ---- */
   550→  const updateDebouncedSearch = useCallback((value: string) => {
   551→    if (debounceTimerRef.current) {
   552→      clearTimeout(debounceTimerRef.current)
   553→    }
   554→    debounceTimerRef.current = setTimeout(() => {
   555→      setDebouncedSearch(value)
   556→    }, 300)
   557→  }, [])
   558→
   559→  // Sync search input to debounced value
   560→  useEffect(() => {
   561→    updateDebouncedSearch(searchInput)
   562→    return () => {
   563→      if (debounceTimerRef.current) {
   564→        clearTimeout(debounceTimerRef.current)
   565→      }
   566→    }
   567→  }, [searchInput, updateDebouncedSearch])
   568→
   569→  /* ---- Fetch templates ---- */
   570→  const fetchTemplates = useCallback(async () => {
   571→    if (!user) return
   572→
   573→    setLoading(true)
   574→    try {
   575→      const params = new URLSearchParams()
   576→
   577→      if (category !== 'all') params.set('category', category)
   578→      if (legalArea !== 'all') params.set('legalArea', legalArea)
   579→      if (audience !== 'all') params.set('audience', audience)
   580→      if (priceFilter === 'free') params.set('price', 'free')
   581→      if (priceFilter === 'paid') params.set('price', 'paid')
   582→      if (debouncedSearch.trim()) params.set('search', debouncedSearch.trim())
   583→
   584→      const query = params.toString() ? `?${params.toString()}` : ''
   585→      const res = await fetch(`/api/templates${query}`, {
   586→        headers: { 'x-user-id': user.id },
   587→      })
   588→
   589→      if (res.ok) {
   590→        const data = await res.json()
   591→        setTemplates(Array.isArray(data) ? data : data.templates || [])
   592→      }
   593→    } catch {
   594→      // keep empty on error
   595→    } finally {
   596→      setLoading(false)
   597→    }
   598→  }, [user, category, legalArea, audience, priceFilter, debouncedSearch])
   599→
   600→  useEffect(() => {
   601→    fetchTemplates()
   602→  }, [fetchTemplates])
   603→
   604→  /* ---- Handlers ---- */
   605→  const handleSelectTemplate = useCallback((template: Template) => {
   606→    setSelectedTemplate(template)
   607→    setDialogOpen(true)
   608→  }, [])
   609→
   610→  const handleDialogClose = useCallback((open: boolean) => {
   611→    setDialogOpen(open)
   612→    if (!open) {
   613→      // Keep template reference for animation but can clear if desired
   614→    }
   615→  }, [])
   616→
   617→  const hasActiveFilters =
   618→    searchInput.trim() !== '' ||
   619→    category !== 'all' ||
   620→    legalArea !== 'all' ||
   621→    audience !== 'all' ||
   622→    priceFilter !== 'all'
   623→
   624→  /* ---- Render ---- */
   625→  return (
   626→    <main className="min-h-screen bg-[#0A1628]">
   627→      <div className="mx-auto max-w-7xl space-y-6 px-4 py-8 sm:px-6 lg:px-8">
   628→        {/* ---------- Header ---------- */}
   629→        <div>
   630→          <h1 className="text-2xl font-bold tracking-tight text-white sm:text-3xl">
   631→            Catálogo de Plantillas
   632→          </h1>
   633→          <p className="mt-1 text-sm text-white/50">
   634→            Explora y genera documentos legales colombianos con plantillas profesionales.
   635→          </p>
   636→        </div>
   637→
   638→        {/* ---------- Search bar ---------- */}
   639→        <div className="relative">
   640→          <Search className="absolute top-1/2 left-3 h-4 w-4 -translate-y-1/2 text-white/30" />
   641→          <Input
   642→            placeholder="Buscar plantillas por nombre, tipo o palabra clave..."
   643→            value={searchInput}
   644→            onChange={(e) => setSearchInput(e.target.value)}
   645→            className="h-11 w-full border-white/10 bg-[#0F1D32] pl-10 pr-4 text-sm text-white placeholder:text-white/30 focus-visible:border-[#C9A94E]/40 focus-visible:ring-[#C9A94E]/20"
   646→          />
   647→        </div>
   648→
   649→        {/* ---------- Filters row ---------- */}
   650→        <div className="flex flex-col gap-3 sm:flex-row sm:flex-wrap sm:items-center sm:gap-3">
   651→          {/* Category */}
   652→          <Select value={category} onValueChange={setCategory}>
   653→            <SelectTrigger
   654→              className="w-full border-white/10 bg-[#0F1D32] text-white/70 text-xs sm:w-[155px]"
   655→              size="sm"
   656→            >
   657→              <SelectValue placeholder="Categoría" />
   658→            </SelectTrigger>
   659→            <SelectContent className="border-white/10 bg-[#0F1D32]">
   660→              <SelectItem value="all" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   661→                Todas
   662→              </SelectItem>
   663→              <SelectItem value="Contratos" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   664→                Contratos
   665→              </SelectItem>
   666→              <SelectItem value="Actas" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   667→                Actas
   668→              </SelectItem>
   669→              <SelectItem value="Derechos de Petición" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   670→                Derechos de Petición
   671→              </SelectItem>
   672→            </SelectContent>
   673→          </Select>
   674→
   675→          {/* Legal Area */}
   676→          <Select value={legalArea} onValueChange={setLegalArea}>
   677→            <SelectTrigger
   678→              className="w-full border-white/10 bg-[#0F1D32] text-white/70 text-xs sm:w-[165px]"
   679→              size="sm"
   680→            >
   681→              <SelectValue placeholder="Área del Derecho" />
   682→            </SelectTrigger>
   683→            <SelectContent className="border-white/10 bg-[#0F1D32]">
   684→              <SelectItem value="all" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   685→                Todos
   686→              </SelectItem>
   687→              <SelectItem value="Civil" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   688→                Civil
   689→              </SelectItem>
   690→              <SelectItem value="Mercantil" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   691→                Mercantil
   692→              </SelectItem>
   693→              <SelectItem value="Laboral" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   694→                Laboral
   695→              </SelectItem>
   696→              <SelectItem value="Constitucional" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   697→                Constitucional
   698→              </SelectItem>
   699→            </SelectContent>
   700→          </Select>
   701→
   702→          {/* Audience */}
   703→          <Select value={audience} onValueChange={setAudience}>
   704→            <SelectTrigger
   705→              className="w-full border-white/10 bg-[#0F1D32] text-white/70 text-xs sm:w-[155px]"
   706→              size="sm"
   707→            >
   708→              <SelectValue placeholder="Audiencia" />
   709→            </SelectTrigger>
   710→            <SelectContent className="border-white/10 bg-[#0F1D32]">
   711→              <SelectItem value="all" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   712→                Todos
   713→              </SelectItem>
   714→              <SelectItem value="Profesionales" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   715→                Profesionales
   716→              </SelectItem>
   717→              <SelectItem value="Particulares" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   718→                Particulares
   719→              </SelectItem>
   720→            </SelectContent>
   721→          </Select>
   722→
   723→          {/* Price */}
   724→          <Select value={priceFilter} onValueChange={setPriceFilter}>
   725→            <SelectTrigger
   726→              className="w-full border-white/10 bg-[#0F1D32] text-white/70 text-xs sm:w-[135px]"
   727→              size="sm"
   728→            >
   729→              <SelectValue placeholder="Precio" />
   730→            </SelectTrigger>
   731→            <SelectContent className="border-white/10 bg-[#0F1D32]">
   732→              <SelectItem value="all" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   733→                Todos
   734→              </SelectItem>
   735→              <SelectItem value="free" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   736→                Gratuito
   737→              </SelectItem>
   738→              <SelectItem value="paid" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   739→                De Pago
   740→              </SelectItem>
   741→            </SelectContent>
   742→          </Select>
   743→
   744→          {/* Spacer */}
   745→          <div className="flex-1" />
   746→
   747→          {/* View toggle */}
   748→          <ToggleGroup
   749→            type="single"
   750→            value={viewMode}
   751→            onValueChange={(v) => {
   752→              if (v) setViewMode(v as 'grid' | 'list')
   753→            }}
   754→            className="border-white/10 bg-[#0F1D32]"
   755→          >
   756→            <ToggleGroupItem
   757→              value="grid"
   758→              aria-label="Vista de cuadrícula"
   759→              className="data-[state=on]:bg-[#C9A94E]/15 data-[state=on]:text-[#C9A94E] text-white/40 hover:bg-white/5 hover:text-white/60 h-8 w-8 rounded-l-md"
   760→            >
   761→              <LayoutGrid className="h-4 w-4" />
   762→            </ToggleGroupItem>
   763→            <ToggleGroupItem
   764→              value="list"
   765→              aria-label="Vista de lista"
   766→              className="data-[state=on]:bg-[#C9A94E]/15 data-[state=on]:text-[#C9A94E] text-white/40 hover:bg-white/5 hover:text-white/60 h-8 w-8 rounded-r-md"
   767→            >
   768→              <List className="h-4 w-4" />
   769→            </ToggleGroupItem>
   770→          </ToggleGroup>
   771→        </div>
   772→
   773→        {/* ---------- Results count ---------- */}
   774→        <div className="flex items-center justify-between">
   775→          <p className="text-xs text-white/40">
   776→            {loading
   777→              ? 'Cargando...'
   778→              : `${templates.length} plantilla${templates.length !== 1 ? 's' : ''} encontrada${templates.length !== 1 ? 's' : ''}`}
   779→          </p>
   780→          {hasActiveFilters && (
   781→            <button
   782→              onClick={() => {
   783→                setSearchInput('')
   784→                setCategory('all')
   785→                setLegalArea('all')
   786→                setAudience('all')
   787→                setPriceFilter('all')
   788→              }}
   789→              className="text-[11px] font-medium text-[#C9A94E]/70 transition-colors hover:text-[#C9A94E]"
   790→            >
   791→              Limpiar filtros
   792→            </button>
   793→          )}
   794→        </div>
   795→
   796→        {/* ---------- Content ---------- */}
   797→        {loading ? (
   798→          viewMode === 'grid' ? (
   799→            <div className="grid grid-cols-1 gap-4 md:grid-cols-2 xl:grid-cols-3">
   800→              {Array.from({ length: 6 }).map((_, i) => (
   801→                <SkeletonCard key={i} />
   802→              ))}
   803→            </div>
   804→          ) : (
   805→            <div className="flex flex-col gap-3">
   806→              {Array.from({ length: 6 }).map((_, i) => (
   807→                <SkeletonRow key={i} />
   808→              ))}
   809→            </div>
   810→          )
   811→        ) : templates.length === 0 ? (
   812→          <EmptyState hasFilters={hasActiveFilters} />
   813→        ) : viewMode === 'grid' ? (
   814→          <div className="grid grid-cols-1 gap-4 md:grid-cols-2 xl:grid-cols-3">
   815→            {templates.map((t) => (
   816→              <TemplateCard key={t.id} template={t} onSelect={handleSelectTemplate} />
   817→            ))}
   818→          </div>
   819→        ) : (
   820→          <div className="flex flex-col gap-3">
   821→            {templates.map((t) => (
   822→              <TemplateRow key={t.id} template={t} onSelect={handleSelectTemplate} />
   823→            ))}
   824→          </div>
   825→        )}
   826→
   827→        {/* ---------- Bottom CTA ---------- */}
   828→        <div className="flex items-center justify-center pt-4">
   829→          <button
   830→            onClick={() => {
   831→              // Navigate to a request page or open a dialog for custom document
   832→              // For now, we'll set page to documents where a request flow might exist
   833→            }}
   834→            className="group flex items-center gap-3 rounded-xl border border-[#C9A94E]/15 bg-[#0F1D32]/60 px-6 py-4 text-left backdrop-blur-sm transition-all hover:border-[#C9A94E]/30 hover:bg-[#C9A94E]/5 hover:shadow-lg hover:shadow-[#C9A94E]/5"
   835→          >
   836→            <div className="flex h-11 w-11 shrink-0 items-center justify-center rounded-lg bg-[#C9A94E]/15 text-[#C9A94E] transition-colors group-hover:bg-[#C9A94E]/25">
   837→              <Plus className="h-5 w-5" />
   838→            </div>
   839→            <div className="min-w-0 flex-1">
   840→              <p className="text-sm font-semibold text-white/80 group-hover:text-white">
   841→                ¿No encuentras lo que buscas?
   842→              </p>
   843→              <p className="text-xs text-white/40">
   844→                Solicita un documento personalizado y nuestro equipo lo preparará para ti.
   845→              </p>
   846→            </div>
   847→            <ArrowRight className="h-4 w-4 shrink-0 text-white/20 transition-transform group-hover:translate-x-0.5 group-hover:text-[#C9A94E]" />
   848→          </button>
   849→        </div>
   850→      </div>
   851→
   852→      {/* ---------- Detail Dialog ---------- */}
   853→      <TemplateDetailDialog
   854→        template={selectedTemplate}
   855→        open={dialogOpen}
   856→        onOpenChange={handleDialogClose}
   857→      />
   858→    </main>
   859→  )
   860→}
