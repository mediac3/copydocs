     9→  Building,

    31→} from '@/components/ui/dialog'
    32→import { Input } from '@/components/ui/input'

    44→import {

    54→/* -------------------------------------------------------------------------- */
    55→/*  Types                                                                    */
    56→/* -------------------------------------------------------------------------- */
    57→
    58→interface Payment {
    59→  id: string
    60→  amount: number
    61→  currency: string
    62→  paymentMethod: string
    63→  paymentGateway: string
    64→  transactionRef: string
    65→  status: string
    66→  planName: string
    67→  createdAt: string
    68→}
    69→
    70→interface Plan {
    71→  id: string
    72→  name: string
    73→  price: number
    74→  interval: string
    75→  maxDocuments: number
    76→  features: string
    77→}
    78→
    79→interface PaymentsResponse {
    80→  payments: Payment[]
    81→  plans: Plan[]
    82→}
    83→
    84→/* -------------------------------------------------------------------------- */
    85→/*  Helpers                                                                  */
    86→/* -------------------------------------------------------------------------- */
    87→
    88→const formatCOP = (amount: number) =>
    89→  new Intl.NumberFormat('es-CO', {
    90→    style: 'currency',
    91→    currency: 'COP',
    92→    minimumFractionDigits: 0,
    93→  }).format(amount)
    94→
    95→const daysUntil = (dateStr: string) => {
    96→  const now = new Date()
    97→  const target = new Date(dateStr)
    98→  const diff = target.getTime() - now.getTime()
    99→  return Math.ceil(diff / (1000 * 60 * 60 * 24))
   100→}
   101→
   102→const formatDate = (dateStr: string) => {
   103→  return new Date(dateStr).toLocaleDateString('es-CO', {
   104→    year: 'numeric',
   105→    month: 'short',
   106→    day: 'numeric',
   107→  })
   108→}
   109→
   110→const PLAN_ICONS: Record<string, React.ReactNode> = {
   111→  Básico: <Zap className="h-6 w-6" />,
   112→  Profesional: <Crown className="h-6 w-6" />,
   113→  Empresarial: <Building className="h-6 w-6" />,
   114→}
   115→
   116→const GATEWAY_LABELS: Record<string, string> = {
   117→  epayco: 'ePayco',
   118→  payu: 'PayU',
   119→  wompi: 'Wompi',
   120→}
   121→
   122→/* -------------------------------------------------------------------------- */
   123→/*  Component                                                                */
   124→/* -------------------------------------------------------------------------- */
   125→
   126→export default function PaymentsPage() {
   127→  const { user, token } = useAppStore()
   128→
   129→  const [payments, setPayments] = useState<Payment[]>([])
   130→  const [plans, setPlans] = useState<Plan[]>([])
   131→  const [loading, setLoading] = useState(true)
   132→  const [submitting, setSubmitting] = useState(false)
   133→
   134→  // Dialog state
   135→  const [dialogOpen, setDialogOpen] = useState(false)
   136→  const [selectedPlan, setSelectedPlan] = useState<Plan | null>(null)
   137→  const [paymentMethod, setPaymentMethod] = useState('credit_card')
   138→  const [paymentGateway, setPaymentGateway] = useState('epayco')
   139→  const [cardNumber, setCardNumber] = useState('')
   140→  const [cardExpiry, setCardExpiry] = useState('')
   141→  const [cardCvv, setCardCvv] = useState('')
   142→
   143→  /* ---------------------------------------------------------------------- */
   144→  /*  Fetch data                                                             */
   145→  /* ---------------------------------------------------------------------- */
   146→
   147→  const fetchData = useCallback(async () => {
   148→    if (!user?.id || !token) {
   149→      setLoading(false)
   150→      return
   151→    }
   152→    try {
   153→      const res = await fetch('/api/payments', {
   154→        headers: { 'x-user-id': user.id, Authorization: `Bearer ${token}` },
   155→      })
   156→      if (!res.ok) throw new Error('Error cargando datos de pagos')
   157→      const data: PaymentsResponse = await res.json()
   158→      setPayments(data.payments ?? [])
   159→      setPlans(data.plans ?? [])
   160→    } catch (err) {
   161→      console.error(err)
   162→      toast.error('No se pudieron cargar los datos de pagos')
   163→    } finally {
   164→      setLoading(false)
   165→    }
   166→  }, [user?.id, token])
   167→
   168→  useEffect(() => {
   169→    fetchData()
   170→  }, [fetchData])
   171→
   172→  /* ---------------------------------------------------------------------- */
   173→  /*  Current plan helpers                                                   */
   174→  /* ---------------------------------------------------------------------- */
   175→
   176→  const currentPlanName = user?.subscriptionPlan ?? null
   177→  const currentPlan = plans.find((p) => p.name === currentPlanName)
   178→  const subscriptionEnd = user?.subscriptionEnd
   179→
   180→  const isRenewalAlert =
   181→    subscriptionEnd && daysUntil(subscriptionEnd) <= 7 && daysUntil(subscriptionEnd) >= 0
   182→
   183→  // Count documents used from completed payments in current period
   184→  const docsUsed = payments.filter(
   185→    (p) =>
   186→      p.status === 'Completado' &&
   187→      p.planName === currentPlanName &&
   188→      subscriptionEnd &&
   189→      new Date(p.createdAt) > new Date(new Date(subscriptionEnd).getTime() - 30 * 24 * 60 * 60 * 1000),
   190→  ).length
   191→
   192→  const docLimit = currentPlan?.maxDocuments ?? 0
   193→  const docProgress = docLimit > 0 ? Math.min((docsUsed / docLimit) * 100, 100) : 0
   194→
   195→  /* ---------------------------------------------------------------------- */
   196→  /*  Open payment dialog                                                    */
   197→  /* ---------------------------------------------------------------------- */
   198→
   199→  const openPaymentDialog = (plan: Plan) => {
   200→    if (plan.name === currentPlanName) return
   201→    setSelectedPlan(plan)
   202→    setPaymentMethod('credit_card')
   203→    setPaymentGateway('epayco')
   204→    setCardNumber('')
   205→    setCardExpiry('')
   206→    setCardCvv('')
   207→    setDialogOpen(true)
   208→  }
   209→
   210→  /* ---------------------------------------------------------------------- */
   211→  /*  Submit payment                                                         */
   212→  /* ---------------------------------------------------------------------- */
   213→
   214→  const handleSubmitPayment = async () => {
   215→    if (!selectedPlan || !user?.id || !token) return
   216→
   217→    if (!cardNumber.trim() || !cardExpiry.trim() || !cardCvv.trim()) {
   218→      toast.error('Por favor completa todos los campos de la tarjeta')
   219→      return
   220→    }
   221→
   222→    setSubmitting(true)
   223→    try {
   224→      const subscriptionEnd = new Date()
   225→      subscriptionEnd.setMonth(subscriptionEnd.getMonth() + 1)
   226→
   227→      const res = await fetch('/api/payments', {
   228→        method: 'POST',
   229→        headers: {
   230→          'Content-Type': 'application/json',
   231→          'x-user-id': user.id,
   232→          Authorization: `Bearer ${token}`,
   233→        },
   234→        body: JSON.stringify({
   235→          amount: selectedPlan.price,
   236→          planName: selectedPlan.name,
   237→          paymentMethod,
   238→          paymentGateway,
   239→          subscriptionEnd: subscriptionEnd.toISOString(),
   240→        }),
   241→      })
   242→
   243→      if (!res.ok) throw new Error('Error procesando el pago')
   244→
   245→      toast.success(`¡Suscripción a ${selectedPlan.name} exitosa!`)
   246→      setDialogOpen(false)
   247→      fetchData()
   248→    } catch (err) {
   249→      console.error(err)
   250→      toast.error('No se pudo procesar el pago. Intenta de nuevo.')
   251→    } finally {
   252→      setSubmitting(false)
   253→    }
   254→  }
   255→
   256→  /* ---------------------------------------------------------------------- */
   257→  /*  Status badge                                                           */
   258→  /* ---------------------------------------------------------------------- */
   259→
   260→  const StatusBadge = ({ status }: { status: string }) => {
   261→    const styles: Record<string, string> = {
   262→      Completado: 'bg-emerald-500/20 text-emerald-400 border border-emerald-500/30',
   263→      Pendiente: 'bg-yellow-500/20 text-yellow-400 border border-yellow-500/30',
   264→      Fallido: 'bg-red-500/20 text-red-400 border border-red-500/30',
   265→    }
   266→    return (
   267→      <Badge
   268→        className={
   269→          styles[status] ?? 'bg-gray-500/20 text-gray-400 border border-gray-500/30'
   270→        }
   271→        variant="outline"
   272→      >
   273→        {status}
   274→      </Badge>
   275→    )
   276→  }
   277→
   278→  /* ---------------------------------------------------------------------- */
   279→  /*  Render: Loading                                                        */
   280→  /* ---------------------------------------------------------------------- */
   281→
   282→  if (loading) {
   283→    return (
   284→      <div className="flex items-center justify-center h-96">
   285→        <div className="animate-spin rounded-full h-10 w-10 border-2 border-[#C9A94E] border-t-transparent" />
   286→      </div>
   287→    )
   288→  }
   289→
   290→  /* ---------------------------------------------------------------------- */
   291→  /*  Render                                                                 */
   292→  /* ---------------------------------------------------------------------- */
   293→
   294→  return (
   295→    <div className="space-y-8 p-6 max-w-7xl mx-auto">
   296→      {/* Page Header */}
   297→      <div>
   298→        <h1 className="text-3xl font-bold text-white flex items-center gap-3">
   299→          <CreditCard className="h-8 w-8 text-[#C9A94E]" />
   300→          Pagos y Suscripciones
   301→        </h1>
   302→        <p className="text-gray-400 mt-1">
   303→          Gestiona tu plan, realiza pagos y consulta tu historial.
   304→        </p>
   305→      </div>
   306→
   307→      {/* ------------------------------------------------------------------ */}
   308→      {/*  Renewal Alert                                                     */}
   309→      {/* ------------------------------------------------------------------ */}
   310→      {isRenewalAlert && subscriptionEnd && (
   311→        <Alert className="border-amber-500/50 bg-amber-500/10">
   312→          <AlertTriangle className="h-5 w-5 text-amber-400" />
   313→          <AlertDescription className="text-amber-200">
   314→            <span className="font-semibold">¡Atención!</span> Tu suscripción{' '}
   315→            <span className="font-semibold text-amber-300">{currentPlanName}</span> vence en{' '}
   316→            <span className="font-semibold text-amber-300">
   317→              {daysUntil(subscriptionEnd)} día{daysUntil(subscriptionEnd) !== 1 ? 's' : ''}
   318→            </span>{' '}
   319→            ({formatDate(subscriptionEnd)}). Renueva para no perder acceso a tus documentos.
   320→          </AlertDescription>
   321→        </Alert>
   322→      )}
   323→
   324→      {/* ------------------------------------------------------------------ */}
   325→      {/*  Current Plan Card                                                 */}
   326→      {/* ------------------------------------------------------------------ */}
   327→      <Card className="bg-gradient-to-br from-[#0F1F3D] to-[#0A1628] border border-[#C9A94E]/30">
   328→        <CardHeader className="pb-4">
   329→          <div className="flex items-center justify-between flex-wrap gap-4">
   330→            <div className="flex items-center gap-3">
   331→              <div className="p-3 rounded-xl bg-[#C9A94E]/10 border border-[#C9A94E]/20">
   332→                <Star className="h-6 w-6 text-[#C9A94E]" />
   333→              </div>
   334→              <div>
   335→                <CardTitle className="text-white text-xl">
   336→                  {currentPlan ? currentPlan.name : 'Sin suscripción'}
   337→                </CardTitle>
   338→                <CardDescription className="text-gray-400">
   339→                  {currentPlan
   340→                    ? `Plan ${currentPlan.name} · ${formatCOP(currentPlan.price)}/${currentPlan.interval === 'month' ? 'mes' : currentPlan.interval}`
   341→                    : 'No tienes una suscripción activa'}
   342→                </CardDescription>
   343→              </div>
   344→            </div>
   345→            <div className="flex items-center gap-4">
   346→              {subscriptionEnd && (
   347→                <div className="flex items-center gap-2 text-sm text-gray-400">
   348→                  <Calendar className="h-4 w-4" />
   349→                  Renovación: {formatDate(subscriptionEnd)}
   350→                </div>
   351→              )}
   352→              <Button
   353→                onClick={() => {
   354→                  const el = document.getElementById('plans-section')
   355→                  el?.scrollIntoView({ behavior: 'smooth' })
   356→                }}
   357→                className="bg-[#C9A94E] hover:bg-[#B8963F] text-[#0A1628] font-semibold"
   358→              >
   359→                Cambiar plan
   360→                <ArrowRight className="ml-2 h-4 w-4" />
   361→              </Button>
   362→            </div>
   363→          </div>
   364→        </CardHeader>
   365→
   366→        {currentPlan && (
   367→          <CardContent>
   368→            <div className="space-y-2">
   369→              <div className="flex items-center justify-between text-sm">
   370→                <span className="text-gray-400">
   371→                  Documentos generados: <span className="text-white font-medium">{docsUsed}</span> /{' '}
   372→                  <span className="text-white font-medium">{docLimit}</span>
   373→                </span>
   374→                <span className="text-[#C9A94E] font-medium">{Math.round(docProgress)}%</span>
   375→              </div>
   376→              <div className="h-3 w-full rounded-full bg-gray-700/50 overflow-hidden">
   377→                <div
   378→                  className="h-full rounded-full transition-all duration-500"
   379→                  style={{
   380→                    width: `${docProgress}%`,
   381→                    background:
   382→                      docProgress > 90
   383→                        ? 'linear-gradient(90deg, #ef4444, #f87171)'
   384→                        : docProgress > 70
   385→                          ? 'linear-gradient(90deg, #f59e0b, #fbbf24)'
   386→                          : 'linear-gradient(90deg, #C9A94E, #E5CC7F)',
   387→                  }}
   388→                />
   389→              </div>
   390→            </div>
   391→          </CardContent>
   392→        )}
   393→      </Card>
   394→
   395→      {/* ------------------------------------------------------------------ */}
   396→      {/*  Subscription Plans                                                */}
   397→      {/* ------------------------------------------------------------------ */}
   398→      <section id="plans-section">
   399→        <h2 className="text-xl font-bold text-white mb-4 flex items-center gap-2">
   400→          <Crown className="h-5 w-5 text-[#C9A94E]" />
   401→          Planes de Suscripción
   402→        </h2>
   403→        <div className="grid grid-cols-1 md:grid-cols-3 gap-6">
   404→          {plans.map((plan) => {
   405→            const isCurrent = plan.name === currentPlanName
   406→            const isPopular = plan.name === 'Profesional'
   407→            const features = plan.features.split(',').map((f) => f.trim()).filter(Boolean)
   408→
   409→            return (
   410→              <Card
   411→                key={plan.id}
   412→                className={`relative bg-[#0F1F3D] transition-all duration-200 hover:shadow-lg hover:shadow-[#C9A94E]/5 ${
   413→                  isCurrent
   414→                    ? 'border-2 border-[#C9A94E] shadow-lg shadow-[#C9A94E]/10'
   415→                    : 'border border-gray-700/50'
   416→                }`}
   417→              >
   418→                {/* Popular badge */}
   419→                {isPopular && (
   420→                  <div className="absolute -top-3 left-1/2 -translate-x-1/2">
   421→                    <Badge className="bg-[#C9A94E] text-[#0A1628] font-bold px-4 py-1 text-xs shadow-md">
   422→                      ★ POPULAR
   423→                    </Badge>
   424→                  </div>
   425→                )}
   426→
   427→                <CardHeader className="pt-6">
   428→                  <div className="flex items-center gap-3">
   429→                    <div
   430→                      className={`p-2.5 rounded-lg ${
   431→                        isCurrent ? 'bg-[#C9A94E]/20 text-[#C9A94E]' : 'bg-gray-700/50 text-gray-400'
   432→                      }`}
   433→                    >
   434→                      {PLAN_ICONS[plan.name] ?? <Zap className="h-6 w-6" />}
   435→                    </div>
   436→                    <div>
   437→                      <CardTitle className="text-white text-lg">{plan.name}</CardTitle>
   438→                      <CardDescription className="text-gray-500 text-xs">
   439→                        Hasta {plan.maxDocuments} docs/{plan.interval === 'month' ? 'mes' : plan.interval}
   440→                      </CardDescription>
   441→                    </div>
   442→                  </div>
   443→                </CardHeader>
   444→
   445→                <CardContent className="space-y-5">
   446→                  {/* Price */}
   447→                  <div>
   448→                    <span className="text-3xl font-bold text-white">{formatCOP(plan.price)}</span>
   449→                    <span className="text-gray-500 text-sm ml-1">/mes</span>
   450→                  </div>
   451→
   452→                  {/* Features */}
   453→                  <ul className="space-y-2.5">
   454→                    {features.map((feature, i) => (
   455→                      <li key={i} className="flex items-start gap-2 text-sm text-gray-300">
   456→                        <Check className="h-4 w-4 text-[#C9A94E] mt-0.5 shrink-0" />
   457→                        <span>{feature}</span>
   458→                      </li>
   459→                    ))}
   460→                  </ul>
   461→
   462→                  {/* CTA Button */}
   463→                  {isCurrent ? (
   464→                    <Button
   465→                      disabled
   466→                      variant="outline"
   467→                      className="w-full border-[#C9A94E]/30 text-[#C9A94E] bg-[#C9A94E]/5 cursor-default"
   468→                    >
   469→                      Plan actual
   470→                    </Button>
   471→                  ) : (
   472→                    <Button
   473→                      onClick={() => openPaymentDialog(plan)}
   474→                      className={`w-full font-semibold ${
   475→                        isPopular
   476→                          ? 'bg-[#C9A94E] hover:bg-[#B8963F] text-[#0A1628]'
   477→                          : 'bg-[#162544] hover:bg-[#1E3358] text-white border border-gray-600/50 hover:border-[#C9A94E]/50'
   478→                      }`}
   479→                    >
   480→                      Suscribirse
   481→                      <ArrowRight className="ml-2 h-4 w-4" />
   482→                    </Button>
   483→                  )}
   484→                </CardContent>
   485→              </Card>
   486→            )
   487→          })}
   488→        </div>
   489→      </section>
   490→
   491→      {/* ------------------------------------------------------------------ */}
   492→      {/*  Payment History                                                   */}
   493→      {/* ------------------------------------------------------------------ */}
   494→      <section>
   495→        <h2 className="text-xl font-bold text-white mb-4 flex items-center gap-2">
   496→          <Receipt className="h-5 w-5 text-[#C9A94E]" />
   497→          Historial de Pagos
   498→        </h2>
   499→        <Card className="bg-[#0F1F3D] border border-gray-700/50">
   500→          <CardContent className="p-0">
   501→            {payments.length === 0 ? (
   502→              <div className="flex flex-col items-center justify-center py-16 text-gray-500">
   503→                <Receipt className="h-12 w-12 mb-3 opacity-40" />
   504→                <p className="text-sm">No hay pagos registrados.</p>
   505→              </div>
   506→            ) : (
   507→              <Table>
   508→                <TableHeader>
   509→                  <TableRow className="border-gray-700/50 hover:bg-transparent">
   510→                    <TableHead className="text-gray-400 font-semibold">Fecha</TableHead>
   511→                    <TableHead className="text-gray-400 font-semibold">Descripción</TableHead>
   512→                    <TableHead className="text-gray-400 font-semibold text-right">Monto</TableHead>
   513→                    <TableHead className="text-gray-400 font-semibold text-center">Estado</TableHead>
   514→                    <TableHead className="text-gray-400 font-semibold">Método</TableHead>
   515→                  </TableRow>
   516→                </TableHeader>
   517→                <TableBody>
   518→                  {payments.map((payment) => (
   519→                    <TableRow
   520→                      key={payment.id}
   521→                      className="border-gray-700/30 hover:bg-white/[0.02]"
   522→                    >
   523→                      <TableCell className="text-gray-300 text-sm">
   524→                        {formatDate(payment.createdAt)}
   525→                      </TableCell>
   526→                      <TableCell className="text-white text-sm font-medium">
   527→                        {payment.planName}
   528→                      </TableCell>
   529→                      <TableCell className="text-white text-sm text-right font-mono">
   530→                        {formatCOP(payment.amount)}
   531→                      </TableCell>
   532→                      <TableCell className="text-center">
   533→                        <StatusBadge status={payment.status} />
   534→                      </TableCell>
   535→                      <TableCell className="text-gray-400 text-sm">
   536→                        <div className="flex items-center gap-1.5">
   537→                          <CreditCard className="h-3.5 w-3.5" />
   538→                          <span>{GATEWAY_LABELS[payment.paymentGateway] ?? payment.paymentGateway}</span>
   539→                        </div>
   540→                      </TableCell>
   541→                    </TableRow>
   542→                  ))}
   543→                </TableBody>
   544→              </Table>
   545→            )}
   546→          </CardContent>
   547→        </Card>
   548→      </section>
   549→
   550→      {/* ------------------------------------------------------------------ */}
   551→      {/*  Payment Dialog                                                    */}
   552→      {/* ------------------------------------------------------------------ */}
   553→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   554→        <DialogContent className="bg-[#0F1F3D] border border-gray-700/50 text-white sm:max-w-lg">
   555→          <DialogHeader>
   556→            <DialogTitle className="text-xl flex items-center gap-2">
   557→              <CreditCard className="h-5 w-5 text-[#C9A94E]" />
   558→              Suscribirse a {selectedPlan?.name}
   559→            </DialogTitle>
   560→          </DialogHeader>
   561→
   562→          <div className="space-y-5 py-2">
   563→            {/* Summary */}
   564→            <div className="rounded-lg bg-[#0A1628] p-4 border border-gray-700/50">
   565→              <div className="flex items-center justify-between">
   566→                <span className="text-gray-400">Plan seleccionado</span>
   567→                <span className="text-white font-semibold">{selectedPlan?.name}</span>
   568→              </div>
   569→              <div className="flex items-center justify-between mt-1">
   570→                <span className="text-gray-400">Total a pagar</span>
   571→                <span className="text-[#C9A94E] text-2xl font-bold">
   572→                  {selectedPlan ? formatCOP(selectedPlan.price) : ''}
   573→                </span>
   574→              </div>
   575→            </div>
   576→
   577→            {/* Payment Gateway */}
