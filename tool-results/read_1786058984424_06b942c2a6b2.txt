   771→/* -------------------------------------------------------------------------- */
   772→/*  Field Renderer                                                           */
   773→/* -------------------------------------------------------------------------- */
   774→
   775→interface FieldRendererProps {
   776→  field: WizardField
   777→  value: string | number | boolean | undefined
   778→  onChange: (value: string | number | boolean) => void
   779→  onFocus: () => void
   780→  contacts: Contact[]
   781→  onApplyContact: (contact: Contact) => void
   782→  onDismissContacts: () => void
   783→}
   784→
   785→function FieldRenderer({
   786→  field,
   787→  value,
   788→  onChange,
   789→  onFocus,
   790→  contacts,
   791→  onApplyContact,
   792→  onDismissContacts,
   793→}: FieldRendererProps) {
   794→  const fieldRef = useRef<HTMLDivElement>(null)
   795→
   796→  // Close contacts dropdown on outside click
   797→  useEffect(() => {
   798→    if (contacts.length === 0) return
   799→    function handleClick(e: MouseEvent) {
   800→      if (fieldRef.current && !fieldRef.current.contains(e.target as Node)) {
   801→        onDismissContacts()
   802→      }
   803→    }
   804→    document.addEventListener('mousedown', handleClick)
   805→    return () => document.removeEventListener('mousedown', handleClick)
   806→  }, [contacts.length, onDismissContacts])
   807→
   808→  const labelElement = (
   809→    <div className="mb-1.5 flex items-center gap-1.5">
   810→      <Label className="text-xs font-medium text-[#CBD5E1]">{field.label}</Label>
   811→      {field.tooltip && (
   812→        <Tooltip>
   813→          <TooltipTrigger asChild>
   814→            <button type="button" className="text-[#475569] hover:text-[#C9A94E] transition-colors">
   815→              <Info className="h-3.5 w-3.5" />
   816→            </button>
   817→          </TooltipTrigger>
   818→          <TooltipContent
   819→            side="right"
   820→            className="max-w-[280px] border-[#1E3A5F] bg-[#152A4A] text-[#CBD5E1] text-xs shadow-xl"
   821→          >
   822→            <p className="leading-relaxed">{field.tooltip}</p>
   823→          </TooltipContent>
   824→        </Tooltip>
   825→      )}
   826→    </div>
   827→  )
   828→
   829→  return (
   830→    <div ref={fieldRef} className="relative animate-fade-in-up">
   831→      {labelElement}
   832→
   833→      {/* Contact autocomplete dropdown */}
   834→      {contacts.length > 0 && (
   835→        <div className="absolute bottom-full left-0 right-0 mb-1 z-50 rounded-lg border border-[#C9A94E]/30 bg-[#152A4A] shadow-xl animate-fade-in-up">
   836→          <div className="px-3 py-1.5 text-[10px] font-medium text-[#C9A94E] uppercase tracking-wider">
   837→            Contactos guardados
   838→          </div>
   839→          {contacts.slice(0, 5).map((contact) => (
   840→            <button
   841→              key={contact.id}
   842→              type="button"
   843→              className="flex w-full items-center gap-3 px-3 py-2 text-left text-xs text-[#CBD5E1] hover:bg-[#1E3A5F] transition-colors first:rounded-t-lg last:rounded-b-lg"
   844→              onClick={() => onApplyContact(contact)}
   845→            >
   846→              <div className="flex h-7 w-7 items-center justify-center rounded-full bg-[#C9A94E]/10 text-[#C9A94E]">
   847→                {contact.name.charAt(0).toUpperCase()}
   848→              </div>
   849→              <div className="flex-1 min-w-0">
   850→                <p className="font-medium truncate">{contact.name}</p>
   851→                <p className="text-[10px] text-[#475569] truncate">
   852→                  {[contact.documentType, contact.documentNumber].filter(Boolean).join(': ') ||
   853→                    contact.companyName ||
   854→                    ''}
   855→                </p>
   856→              </div>
   857→            </button>
   858→          ))}
   859→        </div>
   860→      )}
   861→
   862→      {/* Field input */}
   863→      {field.type === 'text' && (
   864→        <Input
   865→          value={String(value ?? '')}
   866→          onChange={(e) => onChange(e.target.value)}
   867→          onFocus={onFocus}
   868→          placeholder={field.label}
   869→          className="border-[#1E3A5F] bg-[#0F1F38] text-white placeholder:text-[#475569] focus:border-[#C9A94E] focus:ring-[#C9A94E]/20 h-10 text-sm"
   870→        />
   871→      )}
   872→
   873→      {field.type === 'number' && (
   874→        <Input
   875→          type="number"
   876→          value={value !== undefined ? String(value) : ''}
   877→          onChange={(e) => onChange(e.target.value === '' ? '' : Number(e.target.value))}
   878→          placeholder={field.label}
   879→          className="border-[#1E3A5F] bg-[#0F1F38] text-white placeholder:text-[#475569] focus:border-[#C9A94E] focus:ring-[#C9A94E]/20 h-10 text-sm"
   880→        />
   881→      )}
   882→
   883→      {field.type === 'date' && (
   884→        <Input
   885→          type="date"
   886→          value={String(value ?? '')}
   887→          onChange={(e) => onChange(e.target.value)}
   888→          className="border-[#1E3A5F] bg-[#0F1F38] text-white placeholder:text-[#475569] focus:border-[#C9A94E] focus:ring-[#C9A94E]/20 h-10 text-sm [color-scheme:dark]"
   889→        />
   890→      )}
   891→
   892→      {field.type === 'textarea' && (
   893→        <Textarea
   894→          value={String(value ?? '')}
   895→          onChange={(e) => onChange(e.target.value)}
   896→          placeholder={field.label}
   897→          rows={3}
   898→          className="border-[#1E3A5F] bg-[#0F1F38] text-white placeholder:text-[#475569] focus:border-[#C9A94E] focus:ring-[#C9A94E]/20 text-sm resize-none"
   899→        />
   900→      )}
   901→
   902→      {field.type === 'select' && field.options && (
   903→        <Select
   904→          value={value !== undefined ? String(value) : undefined}
   905→          onValueChange={(val) => onChange(val)}
   906→        >
   907→          <SelectTrigger className="border-[#1E3A5F] bg-[#0F1F38] text-white focus:border-[#C9A94E] focus:ring-[#C9A94E]/20 h-10 text-sm">
   908→            <SelectValue placeholder={`Seleccionar ${field.label.toLowerCase()}`} />
   909→          </SelectTrigger>
   910→          <SelectContent className="border-[#1E3A5F] bg-[#152A4A] text-[#CBD5E1]">
   911→            {field.options.map((opt) => (
   912→              <SelectItem key={opt} value={opt} className="text-sm focus:bg-[#1E3A5F] focus:text-white">
   913→                {opt}
   914→              </SelectItem>
   915→            ))}
   916→          </SelectContent>
   917→        </Select>
   918→      )}
   919→
   920→      {field.type === 'boolean' && (
   921→        <RadioGroup
   922→          value={value !== undefined ? String(value) : undefined}
   923→          onValueChange={(val) => onChange(val === 'true')}
   924→          className="flex gap-4"
   925→        >
   926→          <div className="flex items-center gap-2">
   927→            <RadioGroupItem
   928→              value="true"
   929→              id={`${field.key}-yes`}
   930→              className="border-[#1E3A5F] text-[#C9A94E] data-[state=checked]:border-[#C9A94E] data-[state=checked]:bg-[#C9A94E]"
   931→            />
   932→            <Label
   933→              htmlFor={`${field.key}-yes`}
   934→              className="text-sm text-[#CBD5E1] cursor-pointer"
   935→            >
   936→              Sí
   937→            </Label>
   938→          </div>
   939→          <div className="flex items-center gap-2">
   940→            <RadioGroupItem
   941→              value="false"
   942→              id={`${field.key}-no`}
   943→              className="border-[#1E3A5F] text-[#C9A94E] data-[state=checked]:border-[#C9A94E] data-[state=checked]:bg-[#C9A94E]"
   944→            />
   945→            <Label
   946→              htmlFor={`${field.key}-no`}
   947→              className="text-sm text-[#CBD5E1] cursor-pointer"
   948→            >
   949→              No
   950→            </Label>
   951→          </div>
   952→        </RadioGroup>
   953→      )}
   954→    </div>
   955→  )
   956→}

   958→/* -------------------------------------------------------------------------- */
   959→/*  Preview formatting                                                       */
   960→/* -------------------------------------------------------------------------- */
   961→
   962→function formatPreviewContent(content: string): string {
   963→  // Replace newlines with paragraph breaks
   964→  let html = content
   965→    .split('\n')
   966→    .filter((line) => line.trim() !== '')
   967→    .map((line) => {
   968→      // Highlight unresolved {{variables}} in gold
   969→      const highlighted = line.replace(
   970→        /\{\{([^}]+)\}\}/g,
   971→        '<span class="inline-block rounded bg-[#C9A94E]/10 px-1.5 py-0.5 text-[#C9A94E] font-semibold text-xs mx-0.5">{{$1}}</span>'
   972→      )
   973→      return `<p>${highlighted}</p>`
   974→    })
   975→    .join('')
   976→  return html
   977→}
