   701→                )}
   702→                {isReviewStep && (
   703→                  <div className="flex items-center justify-between border-t border-[#1E3A5F] px-6 py-4">
   704→                    <Button
   705→                      variant="outline"
   706→                      className="border-[#1E3A5F] text-[#CBD5E1] hover:bg-[#152A4A] hover:text-white"
   707→                      onClick={goPrev}
   708→                    >
   709→                      <ArrowLeft className="mr-1.5 h-4 w-4" />
   710→                      Anterior
   711→                    </Button>
   712→                    <Button
   713→                      variant="outline"
   714→                      className="border-[#C9A94E]/30 text-[#C9A94E] hover:bg-[#C9A94E]/10"
   715→                      onClick={saveDraft}

   723→              </div>
   724→            </ResizablePanel>
   725→
   726→            {/* ---- Handle ---- */}
   727→            <ResizableHandle withHandle className="bg-[#1E3A5F] hover:bg-[#C9A94E]/50 transition-colors" />
   728→
   729→            {/* =============== RIGHT PANEL - PREVIEW =============== */}

   732→                {/* Preview header */}

   751→                    {/* Document body */}

   762→          </ResizablePanelGroup>

   765→    </TooltipProvider>

   770→/*  Field Renderer                                                           */
   771→/* -------------------------------------------------------------------------- */
   772→
   773→interface FieldRendererProps {
   774→  field: WizardField
   775→  value: string | number | boolean | undefined
   776→  onChange: (value: string | number | boolean) => void
   777→  onFocus: () => void
   778→  contacts: Contact[]
   779→  onApplyContact: (contact: Contact) => void
   780→  onDismissContacts: () => void
   781→}
   782→
   783→function FieldRenderer({
   784→  field,
   785→  value,
   786→  onChange,
   787→  onFocus,
   788→  contacts,
   789→  onApplyContact,
   790→  onDismissContacts,
   791→}: FieldRendererProps) {
   792→  const fieldRef = useRef<HTMLDivElement>(null)
   793→
   794→  // Close contacts dropdown on outside click
   795→  useEffect(() => {
   796→    if (contacts.length === 0) return
   797→    function handleClick(e: MouseEvent) {
   798→      if (fieldRef.current && !fieldRef.current.contains(e.target as Node)) {
   799→        onDismissContacts()
   800→      }
   801→    }
   802→    document.addEventListener('mousedown', handleClick)
   803→    return () => document.removeEventListener('mousedown', handleClick)
   804→  }, [contacts.length, onDismissContacts])
   805→
   806→  const labelElement = (
   807→    <div className="mb-1.5 flex items-center gap-1.5">
   808→      <Label className="text-xs font-medium text-[#CBD5E1]">{field.label}</Label>
   809→      {field.tooltip && (
   810→        <Tooltip>
   811→          <TooltipTrigger asChild>
   812→            <button type="button" className="text-[#475569] hover:text-[#C9A94E] transition-colors">
   813→              <Info className="h-3.5 w-3.5" />
   814→            </button>
   815→          </TooltipTrigger>
   816→          <TooltipContent
   817→            side="right"
   818→            className="max-w-[280px] border-[#1E3A5F] bg-[#152A4A] text-[#CBD5E1] text-xs shadow-xl"
   819→          >
   820→            <p className="leading-relaxed">{field.tooltip}</p>
   821→          </TooltipContent>
   822→        </Tooltip>
   823→      )}
   824→    </div>
   825→  )
   826→
   827→  return (
   828→    <div ref={fieldRef} className="relative animate-fade-in-up">
   829→      {labelElement}
   830→
   831→      {/* Contact autocomplete dropdown */}
   832→      {contacts.length > 0 && (
   833→        <div className="absolute bottom-full left-0 right-0 mb-1 z-50 rounded-lg border border-[#C9A94E]/30 bg-[#152A4A] shadow-xl animate-fade-in-up">
   834→          <div className="px-3 py-1.5 text-[10px] font-medium text-[#C9A94E] uppercase tracking-wider">
   835→            Contactos guardados
   836→          </div>
   837→          {contacts.slice(0, 5).map((contact) => (
   838→            <button
   839→              key={contact.id}
   840→              type="button"
   841→              className="flex w-full items-center gap-3 px-3 py-2 text-left text-xs text-[#CBD5E1] hover:bg-[#1E3A5F] transition-colors first:rounded-t-lg last:rounded-b-lg"
   842→              onClick={() => onApplyContact(contact)}
   843→            >
   844→              <div className="flex h-7 w-7 items-center justify-center rounded-full bg-[#C9A94E]/10 text-[#C9A94E]">
   845→                {contact.name.charAt(0).toUpperCase()}
   846→              </div>
   847→              <div className="flex-1 min-w-0">
   848→                <p className="font-medium truncate">{contact.name}</p>
   849→                <p className="text-[10px] text-[#475569] truncate">
   850→                  {[contact.documentType, contact.documentNumber].filter(Boolean).join(': ') ||
   851→                    contact.companyName ||
   852→                    ''}
   853→                </p>
   854→              </div>
   855→            </button>
   856→          ))}
   857→        </div>
   858→      )}
   859→
   860→      {/* Field input */}
   861→      {field.type === 'text' && (
   862→        <Input
   863→          value={String(value ?? '')}
   864→          onChange={(e) => onChange(e.target.value)}
   865→          onFocus={onFocus}
   866→          placeholder={field.label}
   867→          className="border-[#1E3A5F] bg-[#0F1F38] text-white placeholder:text-[#475569] focus:border-[#C9A94E] focus:ring-[#C9A94E]/20 h-10 text-sm"
   868→        />
   869→      )}
   870→
   871→      {field.type === 'number' && (
   872→        <Input
   873→          type="number"
   874→          value={value !== undefined ? String(value) : ''}
   875→          onChange={(e) => onChange(e.target.value === '' ? '' : Number(e.target.value))}
   876→          placeholder={field.label}
   877→          className="border-[#1E3A5F] bg-[#0F1F38] text-white placeholder:text-[#475569] focus:border-[#C9A94E] focus:ring-[#C9A94E]/20 h-10 text-sm"
   878→        />
   879→      )}
   880→
   881→      {field.type === 'date' && (
   882→        <Input
   883→          type="date"
   884→          value={String(value ?? '')}
   885→          onChange={(e) => onChange(e.target.value)}
   886→          className="border-[#1E3A5F] bg-[#0F1F38] text-white placeholder:text-[#475569] focus:border-[#C9A94E] focus:ring-[#C9A94E]/20 h-10 text-sm [color-scheme:dark]"
   887→        />
   888→      )}
   889→
   890→      {field.type === 'textarea' && (
   891→        <Textarea
   892→          value={String(value ?? '')}
   893→          onChange={(e) => onChange(e.target.value)}
   894→          placeholder={field.label}
   895→          rows={3}
   896→          className="border-[#1E3A5F] bg-[#0F1F38] text-white placeholder:text-[#475569] focus:border-[#C9A94E] focus:ring-[#C9A94E]/20 text-sm resize-none"
   897→        />
   898→      )}
   899→
   900→      {field.type === 'select' && field.options && (
   901→        <Select
   902→          value={value !== undefined ? String(value) : undefined}
   903→          onValueChange={(val) => onChange(val)}
   904→        >
   905→          <SelectTrigger className="border-[#1E3A5F] bg-[#0F1F38] text-white focus:border-[#C9A94E] focus:ring-[#C9A94E]/20 h-10 text-sm">
   906→            <SelectValue placeholder={`Seleccionar ${field.label.toLowerCase()}`} />
   907→          </SelectTrigger>
   908→          <SelectContent className="border-[#1E3A5F] bg-[#152A4A] text-[#CBD5E1]">
   909→            {field.options.map((opt) => (
   910→              <SelectItem key={opt} value={opt} className="text-sm focus:bg-[#1E3A5F] focus:text-white">
   911→                {opt}
   912→              </SelectItem>
   913→            ))}
   914→          </SelectContent>
   915→        </Select>
   916→      )}
   917→
   918→      {field.type === 'boolean' && (
   919→        <RadioGroup
   920→          value={value !== undefined ? String(value) : undefined}
   921→          onValueChange={(val) => onChange(val === 'true')}
   922→          className="flex gap-4"
   923→        >
   924→          <div className="flex items-center gap-2">
   925→            <RadioGroupItem
   926→              value="true"
   927→              id={`${field.key}-yes`}
   928→              className="border-[#1E3A5F] text-[#C9A94E] data-[state=checked]:border-[#C9A94E] data-[state=checked]:bg-[#C9A94E]"
   929→            />
   930→            <Label
   931→              htmlFor={`${field.key}-yes`}
   932→              className="text-sm text-[#CBD5E1] cursor-pointer"
   933→            >
   934→              Sí
   935→            </Label>
   936→          </div>
   937→          <div className="flex items-center gap-2">
   938→            <RadioGroupItem
   939→              value="false"
   940→              id={`${field.key}-no`}
   941→              className="border-[#1E3A5F] text-[#C9A94E] data-[state=checked]:border-[#C9A94E] data-[state=checked]:bg-[#C9A94E]"
   942→            />
   943→            <Label
   944→              htmlFor={`${field.key}-no`}
   945→              className="text-sm text-[#CBD5E1] cursor-pointer"
   946→            >
   947→              No
   948→            </Label>
   949→          </div>
   950→        </RadioGroup>
   951→      )}
   952→    </div>
   953→  )
   954→}

   956→/* -------------------------------------------------------------------------- */
   957→/*  Preview formatting                                                       */
   958→/* -------------------------------------------------------------------------- */
   959→
   960→function formatPreviewContent(content: string): string {
   961→  // Replace newlines with paragraph breaks
   962→  let html = content
   963→    .split('\n')
   964→    .filter((line) => line.trim() !== '')
   965→    .map((line) => {
   966→      // Highlight unresolved {{variables}} in gold
   967→      const highlighted = line.replace(
   968→        /\{\{([^}]+)\}\}/g,
   969→        '<span class="inline-block rounded bg-[#C9A94E]/10 px-1.5 py-0.5 text-[#C9A94E] font-semibold text-xs mx-0.5">{{$1}}</span>'
   970→      )
   971→      return `<p>${highlighted}</p>`
   972→    })
   973→    .join('')
   974→  return html
   975→}
