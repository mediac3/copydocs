   159→function formatCurrency(value: number): string {
   160→  return new Intl.NumberFormat('es-CO', {
   161→    style: 'currency',
   162→    currency: 'COP',
   163→    minimumFractionDigits: 0,
   164→  }).format(value)
   165→}
   166→
   167→function formatDate(dateStr: string): string {
   168→  return new Date(dateStr).toLocaleDateString('es-CO', {
   169→    day: '2-digit',
   170→    month: 'short',
   171→    year: 'numeric',
   172→  })
   173→}
   174→
   175→function formatDateTime(dateStr: string): string {
   176→  return new Date(dateStr).toLocaleDateString('es-CO', {
   177→    day: '2-digit',
   178→    month: 'short',
   179→    year: 'numeric',
   180→    hour: '2-digit',
   181→    minute: '2-digit',
   182→  })
   183→}

   228→function ChartTooltip({ active, payload, label }: { active?: boolean; payload?: Array<{ value: number; name: string; color: string }>; label?: string }) {
   229→  if (!active || !payload?.length) return null
   230→  return (
   231→    <div className="rounded-lg border border-[#C9A94E]/20 bg-[#0A1628]/95 px-3 py-2 shadow-xl backdrop-blur-sm">
   232→      <p className="mb-1 text-xs font-semibold text-white/80 capitalize">{label}</p>
   233→      {payload.map((p, i) => (
   234→        <div key={i} className="flex items-center gap-2 text-xs">
   235→          <span className="inline-block h-2 w-2 rounded-full" style={{ backgroundColor: p.color }} />
   236→          <span className="text-white/60">{p.name}:</span>
   237→          <span className="font-medium text-white">{p.value}</span>
   238→        </div>
   239→      ))}
   240→    </div>
   241→  )
   242→}
