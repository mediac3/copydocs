   168→function formatCurrency(value: number): string {
   169→  return new Intl.NumberFormat('es-CO', {
   170→    style: 'currency',
   171→    currency: 'COP',
   172→    minimumFractionDigits: 0,
   173→  }).format(value)
   174→}
   175→
   176→function formatDate(dateStr: string): string {
   177→  return new Date(dateStr).toLocaleDateString('es-CO', {
   178→    day: '2-digit',
   179→    month: 'short',
   180→    year: 'numeric',
   181→  })
   182→}
   183→
   184→function formatDateTime(dateStr: string): string {
   185→  return new Date(dateStr).toLocaleDateString('es-CO', {
   186→    day: '2-digit',
   187→    month: 'short',
   188→    year: 'numeric',
   189→    hour: '2-digit',
   190→    minute: '2-digit',
   191→  })
   192→}

   237→function ChartTooltip({ active, payload, label }: { active?: boolean; payload?: Array<{ value: number; name: string; color: string }>; label?: string }) {
   238→  if (!active || !payload?.length) return null
   239→  return (
   240→    <div className="rounded-lg border border-[#C9A94E]/20 bg-[#0A1628]/95 px-3 py-2 shadow-xl backdrop-blur-sm">
   241→      <p className="mb-1 text-xs font-semibold text-white/80 capitalize">{label}</p>
   242→      {payload.map((p, i) => (
   243→        <div key={i} className="flex items-center gap-2 text-xs">
   244→          <span className="inline-block h-2 w-2 rounded-full" style={{ backgroundColor: p.color }} />
   245→          <span className="text-white/60">{p.name}:</span>
   246→          <span className="font-medium text-white">{p.value}</span>
   247→        </div>
   248→      ))}
   249→    </div>
   250→  )
   251→}
