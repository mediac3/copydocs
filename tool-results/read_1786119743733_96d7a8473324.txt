   235→   235→          <span className="inline-block h-2 w-2 rounded-full" style={{ backgroundColor: p.color }} />
   236→   236→          <span className="text-white/60">{p.name}:</span>
   237→   237→          <span className="font-medium text-white">{p.value}</span>
   238→   238→        </div>
