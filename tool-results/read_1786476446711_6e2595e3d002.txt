   168→/* ---- Media helper: stores type + dimensions alongside base64 data ---- */
   169→interface MediaContent {
   170→  type: 'text' | 'image'
   171→  text?: string
   172→  dataUrl?: string
   173→  width?: number
   174→  height?: number
   175→}
   176→
   177→function parseMedia(raw: string | null | undefined): MediaContent {
   178→  if (!raw) return { type: 'text', text: '' }
   179→  if (raw.startsWith('data:image')) {
   180→    try {
   181→      const parsed = JSON.parse(raw)
   182→      if (parsed && parsed.type === 'image' && parsed.dataUrl) return parsed
   183→    } catch { /* not JSON, treat as raw base64 */ }
   184→    return { type: 'image', dataUrl: raw, width: 468, height: 60 }
   185→  }
   186→  return { type: 'text', text: raw }
   187→}
   188→
   189→function encodeMedia(m: MediaContent): string {
   190→  if (m.type === 'text') return m.text || ''
   191→  return JSON.stringify({ type: 'image', dataUrl: m.dataUrl, width: m.width, height: m.height })
   192→}
   193→
   194→function isMediaImage(m: MediaContent): boolean {
   195→  return m.type === 'image' && !!m.dataUrl
   196→}

   372→
   373→function MediaFieldEditor({
   374→  label, rawValue, onChange, defaultW, defaultH, maxH, hint
   375→}: {
   376→  label: string
   377→  rawValue: string
   378→  onChange: (encoded: string) => void
   379→  defaultW: number
   380→  defaultH: number
   381→  maxH: number
   382→  hint: string
   383→}) {
   384→  const [mode, setMode] = useState<'text' | 'image'>(() => {
   385→    if (!rawValue) return 'text'
   386→    if (rawValue.startsWith('data:image')) return 'image'
   387→    try { const p = JSON.parse(rawValue); if (p?.type === 'image') return 'image' } catch {}
   388→    return 'text'
   389→  })
   390→  const [imgData, setImgData] = useState<string>('')
   391→  const [imgW, setImgW] = useState(defaultW)
   392→  const [imgH, setImgH] = useState(defaultH)
   393→  const [textVal, setTextVal] = useState('')
   394→  const initialized = useRef(false)
   395→
   396→  // Init from rawValue on mount
   397→  useEffect(() => {
   398→    if (initialized.current) return
   399→    initialized.current = true
   400→    if (!rawValue) return
   401→    if (rawValue.startsWith('data:image')) {
   402→      setImgData(rawValue)
   403→    } else {
   404→      try {
   405→        const p = JSON.parse(rawValue)
   406→        if (p?.type === 'image') {
   407→          setImgData(p.dataUrl || '')
   408→          if (p.width) setImgW(p.width)
   409→          if (p.height) setImgH(p.height)
   410→        } else {
   411→          setTextVal(typeof p?.text === 'string' ? p.text : rawValue)
   412→        }
   413→      } catch {
   414→        setTextVal(rawValue)
   415→      }
   416→    }
   417→  }, [])
   418→
   419→  const switchMode = (t: 'text' | 'image') => {
   420→    setMode(t)
   421→    if (t === 'text') {
   422→      setTextVal('')
   423→      onChange('')
   424→    } else {
   425→      setImgData('')
   426→      setImgW(defaultW)
   427→      setImgH(defaultH)
   428→      onChange(JSON.stringify({ type: 'image', dataUrl: '', width: defaultW, height: defaultH }))
   429→    }
   430→  }
   431→
   432→  const handleFile = (e: React.ChangeEvent<HTMLInputElement>) => {
   433→    const file = e.target.files?.[0]
   434→    if (!file) return
   435→    if (file.size > 2 * 1024 * 1024) { toast.error('La imagen no puede superar 2 MB'); return }
   436→    const reader = new FileReader()
   437→    reader.onload = () => {
   438→      const dataUrl = reader.result as string
   439→      const img = new window.Image()
   440→      img.onload = () => {
   441→        const ratio = img.width / img.height
   442→        let h = maxH
   443→        let w = Math.round(h * ratio)
   444→        if (w > 468) { w = 468; h = Math.round(w / ratio) }
   445→        setImgData(dataUrl)
   446→        setImgW(w)
   447→        setImgH(h)
   448→        onChange(JSON.stringify({ type: 'image', dataUrl, width: w, height: h }))
   449→      }
   450→      img.onerror = () => {
   451→        setImgData(dataUrl)
   452→        setImgW(defaultW)
   453→        setImgH(defaultH)
   454→        onChange(JSON.stringify({ type: 'image', dataUrl, width: defaultW, height: defaultH }))
   455→      }
   456→      img.src = dataUrl
   457→    }
   458→    reader.readAsDataURL(file)
   459→    e.target.value = ''
   460→  }
   461→
   462→  const removeImage = () => {
   463→    setImgData('')
   464→    setImgW(defaultW)
   465→    setImgH(defaultH)
   466→    onChange(JSON.stringify({ type: 'image', dataUrl: '', width: defaultW, height: defaultH }))
   467→  }
   468→
   469→  return (
   470→    <div className="grid gap-2">
   471→      <Label className="text-white/70">{label}</Label>
   472→      <div className="flex gap-1 mb-1">
   473→        {(['text', 'image'] as const).map(t => (
   474→          <button
   475→            key={t}
   476→            type="button"
   477→            onClick={() => switchMode(t)}
   478→            className={`flex-1 rounded-md px-3 py-1.5 text-[11px] font-medium transition-colors ${
   479→              mode === t
   480→                ? 'bg-[#28A745]/20 text-[#28A745] border border-[#28A745]/30'
   481→                : 'bg-white/5 text-white/40 border border-white/10 hover:bg-white/10'
   482→            }`}
   483→          >
   484→            {t === 'text' ? 'Texto' : 'Imagen'}
   485→          </button>
   486→        ))}
   487→      </div>
   488→      {mode === 'text' ? (
   489→        <Textarea
   490→          value={textVal}
   491→          onChange={(e) => { setTextVal(e.target.value); onChange(e.target.value) }}
   492→          placeholder={`Texto del ${label.toLowerCase()}...`}
   493→          rows={3}
   494→          className="border-white/10 bg-white/5 text-white placeholder:text-white/30"
   495→        />
   496→      ) : (
   497→        <div className="space-y-2">
   498→          {imgData && (
   499→            <div className="flex items-center justify-center rounded-lg border border-dashed border-white/15 bg-white/[0.02] p-3">
   500→              <img src={imgData} alt="Preview" className="max-h-24 max-w-full object-contain rounded" />
   501→            </div>
   502→          )}
   503→          <label className="flex cursor-pointer items-center justify-center gap-2 rounded-md border border-white/10 bg-white/5 px-3 py-2.5 text-xs text-white/50 hover:bg-white/10 hover:text-white/70 transition-colors">
   504→            <Upload className="h-3.5 w-3.5" />
   505→            {imgData ? 'Cambiar imagen' : 'Subir imagen (PNG, JPG, WebP)'}
   506→            <input type="file" accept="image/png,image/jpeg,image/webp" className="hidden" onChange={handleFile} />
   507→          </label>
   508→          <div className="grid grid-cols-2 gap-2">
   509→            <div className="grid gap-1">
   510→              <span className="text-[9px] uppercase tracking-wider text-white/25">Ancho (px)</span>
   511→              <Input
   512→                type="number"
   513→                value={imgW}
   514→                onChange={(e) => {
   515→                  const w = parseInt(e.target.value) || defaultW
   516→                  setImgW(w)
   517→                  onChange(JSON.stringify({ type: 'image', dataUrl: imgData, width: w, height: imgH }))
   518→                }}
   519→                className="h-7 border-white/10 bg-white/5 text-white text-xs"
   520→              />
   521→            </div>
   522→            <div className="grid gap-1">
   523→              <span className="text-[9px] uppercase tracking-wider text-white/25">Alto (px)</span>
   524→              <Input
   525→                type="number"
   526→                value={imgH}
   527→                onChange={(e) => {
   528→                  const h = parseInt(e.target.value) || defaultH
   529→                  setImgH(h)
   530→                  onChange(JSON.stringify({ type: 'image', dataUrl: imgData, width: imgW, height: h }))
   531→                }}
   532→                className="h-7 border-white/10 bg-white/5 text-white text-xs"
   533→              />
   534→            </div>
   535→          </div>
   536→          <p className="text-[10px] text-white/25 text-center">{hint}. Max 2 MB.</p>
   537→          {imgData && (
   538→            <button
   539→              type="button"
   540→              onClick={removeImage}
   541→              className="w-full text-center text-[11px] text-red-400/60 hover:text-red-400 transition-colors"
   542→            >
   543→              Eliminar imagen
   544→            </button>
   545→          )}
   546→        </div>
   547→      )}
   548→    </div>
   549→  )
   550→}

   556→const FIELD_TYPE_OPTIONS: Array<{ value: WizardFieldDef['type']; label: string; icon: React.ElementType }> = [
   557→  { value: 'text', label: 'Texto', icon: Type },
   558→  { value: 'number', label: 'Número', icon: Hash },

   564→
   565→function WizardStepEditor({
   566→  step, stepIndex, allSteps, onChange, onRemove
   567→}: {
   568→  step: WizardStepDef
   569→  stepIndex: number
   570→  allSteps: WizardStepDef[]
   571→  onChange: (updated: WizardStepDef) => void
   572→  onRemove: () => void
   573→}) {
   574→  const [expanded, setExpanded] = useState(true)
   575→
   576→  const updateField = (fi: number, patch: Partial<WizardFieldDef>) => {
   577→    const fields = step.fields.map((f, i) => i === fi ? { ...f, ...patch } : f)
   578→    onChange({ ...step, fields })
   579→  }
   580→
   581→  const removeField = (fi: number) => {
   582→    onChange({ ...step, fields: step.fields.filter((_, i) => i !== fi) })
   583→  }
   584→
   585→  const addField = () => {
   586→    onChange({
   587→      ...step,
   588→      fields: [...step.fields, { key: '', label: '', type: 'text' }]
