   332→const FIELD_TYPE_OPTIONS: Array<{ value: WizardFieldDef['type']; label: string; icon: React.ElementType }> = [
   333→  { value: 'text', label: 'Texto', icon: Type },
   334→  { value: 'number', label: 'Número', icon: Hash },
   335→  { value: 'date', label: 'Fecha', icon: CalendarDays },
   336→  { value: 'textarea', label: 'Área de texto', icon: AlignLeft },
   337→  { value: 'select', label: 'Selección', icon: List },
   338→  { value: 'boolean', label: 'Sí/No', icon: ToggleLeft },
   339→]
   340→
   341→function WizardStepEditor({
   342→  step, stepIndex, allSteps, onChange, onRemove
   343→}: {
   344→  step: WizardStepDef
   345→  stepIndex: number
   346→  allSteps: WizardStepDef[]
   347→  onChange: (updated: WizardStepDef) => void
   348→  onRemove: () => void
   349→}) {
   350→  const [expanded, setExpanded] = useState(true)
   351→
   352→  const updateField = (fi: number, patch: Partial<WizardFieldDef>) => {
   353→    const fields = step.fields.map((f, i) => i === fi ? { ...f, ...patch } : f)
   354→    onChange({ ...step, fields })
   355→  }
   356→
   357→  const removeField = (fi: number) => {
   358→    onChange({ ...step, fields: step.fields.filter((_, i) => i !== fi) })
   359→  }
   360→
   361→  const addField = () => {
   362→    onChange({
   363→      ...step,
   364→      fields: [...step.fields, { key: '', label: '', type: 'text' }]
   365→    })
   366→  }
   367→
   368→  /* Collect boolean fields for condition dropdown */
   369→  const booleanFields = allSteps.flatMap((s, si) =>
   370→    s.fields.filter(f => f.type === 'boolean').map(f => ({ key: f.key, label: `[Paso ${si + 1}] ${f.label}`, value: true }))
   371→  )
   372→
   373→  return (
   374→    <div className="rounded-lg border border-white/10 bg-white/[0.02] overflow-hidden">
   375→      <div
   376→        className="flex items-center gap-2 px-3 py-2.5 cursor-pointer hover:bg-white/[0.03] transition-colors"
   377→        onClick={() => setExpanded(!expanded)}
   378→      >
   379→        {expanded ? <ChevronDown className="h-3.5 w-3.5 text-white/30" /> : <ChevronRight className="h-3.5 w-3.5 text-white/30" />}
   380→        <Input
   381→          value={step.title}
   382→          onChange={(e) => onChange({ ...step, title: e.target.value })}
   383→          onClick={(e) => e.stopPropagation()}
   384→          className="h-6 border-0 bg-transparent p-0 text-sm font-medium text-white focus-visible:ring-0"
   385→          placeholder={`Paso ${stepIndex + 1}`}
   386→        />
   387→        <Badge variant="outline" className="ml-auto border-white/10 text-white/30 text-[10px]">
   388→          {step.fields.length} campo{step.fields.length !== 1 ? 's' : ''}
   389→        </Badge>
   390→        <Button
   391→          type="button"
   392→          variant="ghost"
   393→          size="sm"
   394→          className="h-6 w-6 p-0 text-white/20 hover:text-red-400 hover:bg-red-400/10"
   395→          onClick={(e) => { e.stopPropagation(); onRemove() }}
   396→        >
   397→          <X className="h-3.5 w-3.5" />
   398→        </Button>
   399→      </div>
   400→
   401→      {expanded && (
   402→        <div className="border-t border-white/5 p-3 space-y-2">
   403→          {step.fields.length === 0 && (
   404→            <p className="text-center text-[11px] text-white/20 py-3">Sin campos. Haz clic en &quot;Agregar campo&quot;.</p>
   405→          )}
   406→          {step.fields.map((field, fi) => (
   407→            <div key={fi} className="grid gap-2 rounded-md border border-white/5 bg-white/[0.02] p-2.5 sm:grid-cols-[1fr_1fr_auto_auto_auto]">
   408→              <div className="flex flex-col gap-1">
   409→                <span className="text-[9px] uppercase tracking-wider text-white/20">Variable (key)</span>
   410→                <Input
   411→                  value={field.key}
   412→                  onChange={(e) => updateField(fi, { key: e.target.value.replace(/[^a-zA-Z0-9_]/g, '').toLowerCase() })}
   413→                  placeholder="nombre_variable"
   414→                  className="h-7 border-white/10 bg-white/5 text-white text-xs placeholder:text-white/20"
   415→                />
   416→              </div>
   417→              <div className="flex flex-col gap-1">
   418→                <span className="text-[9px] uppercase tracking-wider text-white/20">Etiqueta</span>
   419→                <Input
   420→                  value={field.label}
   421→                  onChange={(e) => updateField(fi, { label: e.target.value })}
   422→                  placeholder="Texto visible"
   423→                  className="h-7 border-white/10 bg-white/5 text-white text-xs placeholder:text-white/20"
   424→                />
   425→              </div>
   426→              <div className="flex flex-col gap-1">
   427→                <span className="text-[9px] uppercase tracking-wider text-white/20">Tipo</span>
   428→                <Select
   429→                  value={field.type}
   430→                  onValueChange={(v) => updateField(fi, { type: v as WizardFieldDef['type'], options: v === 'select' ? field.options || [''] : undefined })}
   431→                >
   432→                  <SelectTrigger className="h-7 w-[110px] border-white/10 bg-white/5 text-white text-xs">
   433→                    <SelectValue />
   434→                  </SelectTrigger>
   435→                  <SelectContent className="border-white/10 bg-[#0F1D32]">
   436→                    {FIELD_TYPE_OPTIONS.map(ft => (
   437→                      <SelectItem key={ft.value} value={ft.value} className="text-white focus:bg-white/5 focus:text-white text-xs">
   438→                        {ft.label}
   439→                      </SelectItem>
   440→                    ))}
   441→                  </SelectContent>
   442→                </Select>
   443→              </div>
   444→              <div className="flex flex-col gap-1">
   445→                <span className="text-[9px] uppercase tracking-wider text-white/20">Cond.</span>
   446→                <Select
   447→                  value={field.condition?.field || '_none'}
   448→                  onValueChange={(v) => {
   449→                    if (v === '_none') {
   450→                      const { condition, ...rest } = field
   451→                      const cleaned = { ...rest } as WizardFieldDef
   452→                      const fields = step.fields.map((f, i) => i === fi ? cleaned : f)
   453→                      onChange({ ...step, fields })
   454→                    } else {
   455→                      const bf = booleanFields.find(b => b.key === v)
   456→                      updateField(fi, { condition: { field: v, value: bf?.value ?? true } })
   457→                    }
   458→                  }}
   459→                >
   460→                  <SelectTrigger className="h-7 w-[110px] border-white/10 bg-white/5 text-white/20 text-xs">
   461→                    <SelectValue placeholder="Ninguna" />
   462→                  </SelectTrigger>
   463→                  <SelectContent className="border-white/10 bg-[#0F1D32]">
   464→                    <SelectItem value="_none" className="text-white/50 text-xs">Ninguna</SelectItem>
   465→                    {booleanFields.map(bf => (
   466→                      <SelectItem key={bf.key} value={bf.key} className="text-white focus:bg-white/5 focus:text-white text-xs">
   467→                        {bf.label}
   468→                      </SelectItem>
   469→                    ))}
   470→                  </SelectContent>
   471→                </Select>
   472→              </div>
   473→              <div className="flex items-end pb-0.5">
   474→                <Button
   475→                  type="button"
   476→                  variant="ghost"
   477→                  size="sm"
   478→                  className="h-7 w-7 p-0 text-white/20 hover:text-red-400 hover:bg-red-400/10"
   479→                  onClick={() => removeField(fi)}
   480→                >
   481→                  <TrashIcon className="h-3.5 w-3.5" />
   482→                </Button>
   483→              </div>
   484→              {/* Options editor for select type */}
   485→              {field.type === 'select' && (
   486→                <div className="sm:col-span-5 mt-1">
   487→                  <span className="text-[9px] uppercase tracking-wider text-white/20">Opciones (separadas por coma)</span>
   488→                  <Input
   489→                    value={(field.options || []).join(', ')}
   490→                    onChange={(e) => updateField(fi, { options: e.target.value.split(',').map(o => o.trim()).filter(Boolean) })}
   491→                    placeholder="Opción 1, Opción 2, Opción 3"
   492→                    className="h-7 mt-0.5 border-white/10 bg-white/5 text-white text-xs placeholder:text-white/20"
   493→                  />
   494→                </div>
   495→              )}
   496→              {/* Tooltip */}
   497→              <div className="sm:col-span-5">
   498→                <Input
   499→                  value={field.tooltip || ''}
   500→                  onChange={(e) => updateField(fi, { tooltip: e.target.value || undefined })}
   501→                  placeholder="Tooltip o ayuda (opcional)"
   502→                  className="h-6 border-0 bg-transparent text-white/30 text-[11px] placeholder:text-white/15 focus-visible:ring-0"
   503→                />
   504→              </div>
   505→            </div>
   506→          ))}
   507→          <Button
   508→            type="button"
   509→            variant="outline"
   510→            size="sm"
   511→            className="w-full border-dashed border-white/10 bg-transparent text-white/40 hover:bg-white/5 hover:text-white/70 text-xs"
   512→            onClick={addField}
   513→          >
   514→            <Plus className="mr-1 h-3 w-3" />
   515→            Agregar campo
   516→          </Button>
   517→        </div>
   518→      )}
   519→    </div>
   520→  )
   521→}

   526→
   527→function InlineTerms() {
   528→  const { authHeaders } = useAppStore() as { authHeaders: () => Record<string, string> }
   529→  const [terms, setTerms] = useState('')
   530→  const [loading, setLoading] = useState(true)
   531→  const [saving, setSaving] = useState(false)
   532→
   533→  useEffect(() => {
   534→    setLoading(true)
   535→    fetch('/api/settings').then(r => r.json()).then(data => {
   536→      setTerms(data.terms_and_conditions || '')
   537→    }).catch(() => {}).finally(() => setLoading(false))
   538→  }, [])
   539→
   540→  const handleSave = async () => {
   541→    setSaving(true)
   542→    try {
   543→      const res = await fetch('/api/settings', {
   544→        method: 'PUT',
   545→        headers: authHeaders(),
   546→        body: JSON.stringify({ terms_and_conditions: terms }),
   547→      })
   548→      if (res.ok) toast.success('Términos y condiciones actualizados')
   549→      else toast.error('Error al guardar')
   550→    } catch { toast.error('Error de conexión') }
   551→    finally { setSaving(false) }
   552→  }
   553→
   554→  return (
   555→    <div className="space-y-6">
   556→      <div>
   557→        <h2 className="text-lg font-semibold text-white">Términos y Condiciones</h2>
   558→        <p className="text-xs text-white/40">Configura los términos que los visitantes deben aceptar antes de generar un documento.</p>
