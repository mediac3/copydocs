   169→   169→interface MediaContent {
   170→   170→  type: 'text' | 'image'
   171→   171→  text?: string
   172→   172→  dataUrl?: string
   173→   173→  width?: number
   174→   174→  height?: number
   175→   175→}
   176→   176→
   177→   177→function parseMedia(raw: string | null | undefined): MediaContent {
   178→   178→  if (!raw) return { type: 'text', text: '' }
   179→   179→  if (raw.startsWith('data:image')) {
   180→   180→    try {
   181→   181→      const parsed = JSON.parse(raw)
   182→   182→      if (parsed && parsed.type === 'image' && parsed.dataUrl) return parsed
   183→   183→    } catch { /* not JSON, treat as raw base64 */ }
   184→   184→    return { type: 'image', dataUrl: raw, width: 468, height: 60 }
   185→   185→  }
   186→   186→  return { type: 'text', text: raw }
   187→   187→}
   188→   188→
   189→   189→function encodeMedia(m: MediaContent): string {
   190→   190→  if (m.type === 'text') return m.text || ''
   191→   191→  return JSON.stringify({ type: 'image', dataUrl: m.dataUrl, width: m.width, height: m.height })
   192→   192→}
   193→   193→
   194→   194→function isMediaImage(m: MediaContent): boolean {
   195→   195→  return m.type === 'image' && !!m.dataUrl

   369→   369→/* -------------------------------------------------------------------------- */
   370→   370→/*  Media Field Editor (Header / Footer with Image+Dimensions)                 */
   371→   371→/* -------------------------------------------------------------------------- */
   372→   372→
   373→   373→function MediaFieldEditor({
   374→   374→  label, rawValue, onChange, defaultW, defaultH, maxH, hint
   375→   375→}: {
   376→   376→  label: string
   377→   377→  rawValue: string
   378→   378→  onChange: (encoded: string) => void
   379→   379→  defaultW: number
   380→   380→  defaultH: number
   381→   381→  maxH: number
   382→   382→  hint: string
   383→   383→}) {
   384→   384→  const [mode, setMode] = useState<'text' | 'image'>(() => {
   385→   385→    if (!rawValue) return 'text'
   386→   386→    if (rawValue.startsWith('data:image')) return 'image'
   387→   387→    try { const p = JSON.parse(rawValue); if (p?.type === 'image') return 'image' } catch {}
   388→   388→    return 'text'
   389→   389→  })
   390→   390→  const [imgData, setImgData] = useState<string>('')
   391→   391→  const [imgW, setImgW] = useState(defaultW)
   392→   392→  const [imgH, setImgH] = useState(defaultH)
   393→   393→  const [textVal, setTextVal] = useState('')
   394→   394→  const initialized = useRef(false)
   395→   395→
   396→   396→  // Init from rawValue on mount
   397→   397→  useEffect(() => {
   398→   398→    if (initialized.current) return
   399→   399→    initialized.current = true
   400→   400→    if (!rawValue) return
   401→   401→    if (rawValue.startsWith('data:image')) {
   402→   402→      setImgData(rawValue)
   403→   403→    } else {
   404→   404→      try {
   405→   405→        const p = JSON.parse(rawValue)
   406→   406→        if (p?.type === 'image') {
   407→   407→          setImgData(p.dataUrl || '')
   408→   408→          if (p.width) setImgW(p.width)
   409→   409→          if (p.height) setImgH(p.height)
   410→   410→        } else {
   411→   411→          setTextVal(typeof p?.text === 'string' ? p.text : rawValue)
   412→   412→        }
   413→   413→      } catch {
   414→   414→        setTextVal(rawValue)
   415→   415→      }
   416→   416→    }
   417→   417→  }, [])
   418→   418→
   419→   419→  const switchMode = (t: 'text' | 'image') => {
   420→   420→    setMode(t)
   421→   421→    if (t === 'text') {
   422→   422→      setTextVal('')
   423→   423→      onChange('')
   424→   424→    } else {
   425→   425→      setImgData('')
   426→   426→      setImgW(defaultW)
   427→   427→      setImgH(defaultH)
   428→   428→      onChange(JSON.stringify({ type: 'image', dataUrl: '', width: defaultW, height: defaultH }))
   429→   429→    }
   430→   430→  }
   431→   431→
   432→   432→  const handleFile = (e: React.ChangeEvent<HTMLInputElement>) => {
   433→   433→    const file = e.target.files?.[0]
   434→   434→    if (!file) return
   435→   435→    if (file.size > 2 * 1024 * 1024) { toast.error('La imagen no puede superar 2 MB'); return }
   436→   436→    const reader = new FileReader()
   437→   437→    reader.onload = () => {
   438→   438→      const dataUrl = reader.result as string
   439→   439→      const img = new window.Image()
   440→   440→      img.onload = () => {
   441→   441→        const ratio = img.width / img.height
   442→   442→        let h = maxH
   443→   443→        let w = Math.round(h * ratio)
   444→   444→        if (w > 468) { w = 468; h = Math.round(w / ratio) }
   445→   445→        setImgData(dataUrl)
   446→   446→        setImgW(w)
   447→   447→        setImgH(h)
   448→   448→        onChange(JSON.stringify({ type: 'image', dataUrl, width: w, height: h }))
   449→   449→      }
   450→   450→      img.onerror = () => {
   451→   451→        setImgData(dataUrl)
   452→   452→        setImgW(defaultW)
   453→   453→        setImgH(defaultH)
   454→   454→        onChange(JSON.stringify({ type: 'image', dataUrl, width: defaultW, height: defaultH }))
   455→   455→      }
   456→   456→      img.src = dataUrl
   457→   457→    }
   458→   458→    reader.readAsDataURL(file)
   459→   459→    e.target.value = ''
   460→   460→  }
   461→   461→
   462→   462→  const removeImage = () => {
   463→   463→    setImgData('')
   464→   464→    setImgW(defaultW)
   465→   465→    setImgH(defaultH)
   466→   466→    onChange(JSON.stringify({ type: 'image', dataUrl: '', width: defaultW, height: defaultH }))
   467→   467→  }
   468→   468→
   469→   469→  return (
   470→   470→    <div className="grid gap-2">
   471→   471→      <Label className="text-white/70">{label}</Label>
   472→   472→      <div className="flex gap-1 mb-1">
   473→   473→        {(['text', 'image'] as const).map(t => (
   474→   474→          <button
   475→   475→            key={t}
   476→   476→            type="button"
   477→   477→            onClick={() => switchMode(t)}
   478→   478→            className={`flex-1 rounded-md px-3 py-1.5 text-[11px] font-medium transition-colors ${
   479→   479→              mode === t
   480→   480→                ? 'bg-[#28A745]/20 text-[#28A745] border border-[#28A745]/30'
   481→   481→                : 'bg-white/5 text-white/40 border border-white/10 hover:bg-white/10'
   482→   482→            }`}
   483→   483→          >
   484→   484→            {t === 'text' ? 'Texto' : 'Imagen'}
   485→   485→          </button>
   486→   486→        ))}
   487→   487→      </div>
   488→   488→      {mode === 'text' ? (
   489→   489→        <Textarea
   490→   490→          value={textVal}
   491→   491→          onChange={(e) => { setTextVal(e.target.value); onChange(e.target.value) }}
   492→   492→          placeholder={`Texto del ${label.toLowerCase()}...`}
   493→   493→          rows={3}
   494→   494→          className="border-white/10 bg-white/5 text-white placeholder:text-white/30"
   495→   495→        />
   496→   496→      ) : (
