    76→    76→/*  Contact type mapping - field name prefix → contact type                   */
    77→    77→/* -------------------------------------------------------------------------- */
    78→    78→
    79→    79→const CONTACT_TYPE_MAP: Record<string, string[]> = {
    80→    80→  arrendador: ['arrendador', 'propietario'],
    81→    81→  arrendatario: ['arrendatario', 'inquilino'],
    82→    82→  fiador: ['fiador', 'codeudor'],
    83→    83→  comprador: ['comprador', 'adquirente'],
    84→    84→  vendedor: ['vendedor', 'enajenante'],
    85→    85→  empleador: ['empleador', 'patrono', 'empresa'],
    86→    86→  trabajador: ['trabajador', 'empleado'],
    87→    87→  demandante: ['demandante', 'actor'],
    88→    88→  demandado: ['demandado'],
    89→    89→  deudor: ['deudor'],
    90→    90→  acreedor: ['acreedor'],
    91→    91→}
    92→    92→
    93→    93→function detectContactType(fieldKey: string): string | null {
    94→    94→  const lower = fieldKey.toLowerCase()
    95→    95→  for (const [contactType, prefixes] of Object.entries(CONTACT_TYPE_MAP)) {
    96→    96→    for (const prefix of prefixes) {
    97→    97→      if (lower.includes(prefix)) return contactType
    98→    98→    }
    99→    99→  }
   100→   100→  return null
   101→   101→}

   107→   107→export default function WizardPage() {
   108→   108→  const { user, wizardTemplateId, wizardDocumentId, setCurrentPage } = useAppStore()
   109→   109→
   110→   110→  /* ---- state ---- */
   111→   111→  const [template, setTemplate] = useState<Template | null>(null)
   112→   112→  const [wizardConfig, setWizardConfig] = useState<WizardConfig | null>(null)
   113→   113→  const [currentStep, setCurrentStep] = useState(0)
   114→   114→  const [answers, setAnswers] = useState<Record<string, string | number | boolean>>({})
   115→   115→  const [loading, setLoading] = useState(true)
   116→   116→  const [generating, setGenerating] = useState(false)
   117→   117→  const [saving, setSaving] = useState(false)
   118→   118→  const [docId, setDocId] = useState<string | null>(wizardDocumentId)
   119→   119→  const [contactsCache, setContactsCache] = useState<Record<string, Contact[]>>({})
   120→   120→  const [showContacts, setShowContacts] = useState<string | null>(null)
   121→   121→  const previewRef = useRef<HTMLDivElement>(null)
   122→   122→  const hasLoadedDraft = useRef(false)
   123→   123→
   124→   124→  /* ---- fetch template ---- */
   125→   125→  useEffect(() => {
   126→   126→    if (!wizardTemplateId) {
   127→   127→      setCurrentPage('catalog')
   128→   128→      return
   129→   129→    }
   130→   130→    async function load() {
   131→   131→      try {
   132→   132→        const res = await fetch('/api/templates')
   133→   133→        const data = await res.json()
   134→   134→        const found: Template | undefined = data.templates?.find(
   135→   135→          (t: Template) => t.id === wizardTemplateId
   136→   136→        )
   137→   137→        if (!found) {
   138→   138→          toast.error('Plantilla no encontrada')
   139→   139→          setCurrentPage('catalog')
   140→   140→          return
   141→   141→        }
   142→   142→        setTemplate(found)
   143→   143→        try {
   144→   144→          const config: WizardConfig = typeof found.wizardConfig === 'string' 
   145→   145→            ? JSON.parse(found.wizardConfig) 
   146→   146→            : found.wizardConfig
   147→   147→          setWizardConfig(config)
   148→   148→        } catch {
   149→   149→          toast.error('Error al leer configuración de la plantilla')
   150→   150→          setCurrentPage('catalog')
   151→   151→          return
   152→   152→        }
   153→   153→
   154→   154→        // Load existing draft if editing
   155→   155→        if (wizardDocumentId && !hasLoadedDraft.current) {
   156→   156→          hasLoadedDraft.current = true
   157→   157→          const docRes = await fetch('/api/documents')
   158→   158→          if (docRes.ok) {
   159→   159→            const docData = await docRes.json()
   160→   160→            const existing = docData.documents?.find(
   161→   161→              (d: { id: string; templateId: string }) => d.id === wizardDocumentId
   162→   162→            )
   163→   163→            if (existing && existing.answers) {
   164→   164→              try {
   165→   165→                const parsed = typeof existing.answers === 'string' 
   166→   166→                  ? JSON.parse(existing.answers) 
   167→   167→                  : existing.answers
   168→   168→                setAnswers(parsed)
   169→   169→                setDocId(existing.id)
   170→   170→              } catch { /* ignore */ }
   171→   171→            }
   172→   172→          }
   173→   173→        }
   174→   174→      } catch {
   175→   175→        toast.error('Error al cargar la plantilla')
   176→   176→        setCurrentPage('catalog')
   177→   177→      } finally {
   178→   178→        setLoading(false)
   179→   179→      }
   180→   180→    }
   181→   181→    load()
   182→   182→  }, [wizardTemplateId])
   183→   183→
   184→   184→  /* ---- fetch contacts for a field ---- */
   185→   185→  const fetchContacts = useCallback(async (contactType: string) => {
   186→   186→    if (contactsCache[contactType]) return contactsCache[contactType]
   187→   187→    if (!user) return []

   189→   189→      const res = await fetch(`/api/contacts?type=${encodeURIComponent(contactType)}`, {
   190→   190→        headers: { 'x-user-id': user.id },
   191→   191→      })
   192→   192→      if (!res.ok) return []
   193→   193→      const data = await res.json()
   194→   194→      const contacts: Contact[] = data.contacts || []
   195→   195→      setContactsCache((prev) => ({ ...prev, [contactType]: contacts }))
   196→   196→      return contacts
   197→   197→    } catch {
   198→   198→      return []
   199→   199→    }
   200→   200→  }, [contactsCache, user])
   201→   201→
   202→   202→  /* ---- auto-open contacts dropdown on focus ---- */
   203→   203→  const handleFieldFocus = useCallback(async (fieldKey: string) => {
   204→   204→    const contactType = detectContactType(fieldKey)
   205→   205→    if (!contactType) return
   206→   206→    const contacts = await fetchContacts(contactType)
   207→   207→    if (contacts.length > 0) {
   208→   208→      setShowContacts(fieldKey)
   209→   209→    }
   210→   210→  }, [fetchContacts])
   211→   211→
   212→   212→  /* ---- apply contact data ---- */
   213→   213→  const applyContact = useCallback((contact: Contact, fieldKey: string) => {
   214→   214→    const updates: Record<string, string> = {}
   215→   215→    // Map contact fields to wizard answer keys
   216→   216→    if (contact.name) {
   217→   217→      const nameKey = fieldKey.includes('nombre') ? fieldKey : fieldKey.replace(/_cc$|_cedula$|_nit$|_direccion$|_ciudad$|_telefono$|_email$/, '_nombre')
   218→   218→      updates[nameKey] = contact.name
   219→   219→    }
   220→   220→    if (contact.documentNumber) {
   221→   221→      const ccKey = fieldKey.replace(/_nombre$|_direccion$|_ciudad$|_telefono$|_email$/, '_cc').replace(/_nombre$/, '_cc')
   222→   222→      // Try common patterns
   223→   223→      for (const suffix of ['_cc', '_cedula', '_nit', '_documento']) {
   224→   224→        const candidate = fieldKey.replace(/_nombre.*$/, suffix)
   225→   225→        updates[candidate] = contact.documentNumber
   226→   226→      }
   227→   227→    }
   228→   228→    if (contact.address) {
   229→   229→      updates[fieldKey.replace(/_nombre.*$/, '_direccion')] = contact.address
   230→   230→    }
   231→   231→    if (contact.city) {
   232→   232→      updates[fieldKey.replace(/_nombre.*$/, '_ciudad')] = contact.city
   233→   233→    }
   234→   234→    if (contact.phone) {
   235→   235→      updates[fieldKey.replace(/_nombre.*$/, '_telefono')] = contact.phone
   236→   236→    }
   237→   237→    if (contact.email) {
   238→   238→      updates[fieldKey.replace(/_nombre.*$/, '_email')] = contact.email
   239→   239→    }
   240→   240→
   241→   241→    // For the current field itself, just set the name
   242→   242→    if (contact.name && ['text'].includes(wizardConfig?.steps.flatMap(s => s.fields).find(f => f.key === fieldKey)?.type || '')) {
   243→   243→      updates[fieldKey] = contact.name
   244→   244→    }
   245→   245→
   246→   246→    setAnswers((prev) => {
   247→   247→      const next = { ...prev }
   248→   248→      for (const [k, v] of Object.entries(updates)) {
   249→   249→        // Only set if the key exists in wizard config
   250→   250→        const allFields = wizardConfig?.steps.flatMap((s) => s.fields) || []
   251→   251→        if (allFields.some((f) => f.key === k)) {
   252→   252→          next[k] = v
   253→   253→        }
   254→   254→      }
   255→   255→      return next
   256→   256→    })
   257→   257→    setShowContacts(null)
   258→   258→    toast.success(`Datos de ${contact.name} aplicados`)
   259→   259→  }, [wizardConfig])
   260→   260→
   261→   261→  /* ---- update answer ---- */
   262→   262→  const updateAnswer = useCallback((key: string, value: string | number | boolean) => {
   263→   263→    setAnswers((prev) => ({ ...prev, [key]: value }))
   264→   264→  }, [])
   265→   265→
   266→   266→  /* ---- check condition ---- */
   267→   267→  const isFieldVisible = useCallback(
   268→   268→    (field: WizardField) => {
   269→   269→      if (!field.condition) return true
   270→   270→      return answers[field.condition.field] === field.condition.value
   271→   271→    },
   272→   272→    [answers]
   273→   273→  )
   274→   274→
   275→   275→  /* ---- navigation ---- */
   276→   276→  const totalSteps = wizardConfig ? wizardConfig.steps.length : 0
   277→   277→  const isReviewStep = currentStep === totalSteps
   278→   278→
   279→   279→  const goToStep = useCallback(
   280→   280→    (step: number) => {
   281→   281→      if (step < 0 || step > totalSteps) return
   282→   282→      setCurrentStep(step)
   283→   283→    },
   284→   284→    [totalSteps]
   285→   285→  )
   286→   286→
   287→   287→  const goNext = useCallback(() => {
   288→   288→    if (currentStep < totalSteps) setCurrentStep((s) => s + 1)
   289→   289→  }, [currentStep, totalSteps])
   290→   290→
   291→   291→  const goPrev = useCallback(() => {
   292→   292→    if (currentStep > 0) setCurrentStep((s) => s - 1)
   293→   293→  }, [currentStep])
   294→   294→
   295→   295→  /* ---- save draft ---- */
   296→   296→  const saveDraft = useCallback(async () => {
   297→   297→    if (!user || !template) return
   298→   298→    setSaving(true)
   299→   299→    try {
   300→   300→      const headers: HeadersInit = {
   301→   301→        'Content-Type': 'application/json',
   302→   302→        'x-user-id': user.id,
   303→   303→      }
   304→   304→      if (docId) {
   305→   305→        await fetch('/api/documents', {
   306→   306→          method: 'PUT',
   307→   307→          headers,
   308→   308→          body: JSON.stringify({
   309→   309→            id: docId,
   310→   310→            title: template.name,
   311→   311→            answers,
   312→   312→            status: 'draft',
   313→   313→          }),
   314→   314→        })
   315→   315→      } else {
   316→   316→        const res = await fetch('/api/documents', {
   317→   317→          method: 'POST',
   318→   318→          headers,
   319→   319→          body: JSON.stringify({
   320→   320→            templateId: template.id,
   321→   321→            title: template.name,
   322→   322→            answers,
   323→   323→          }),
   324→   324→        })
   325→   325→        const data = await res.json()
   326→   326→        if (data.document?.id) {
   327→   327→          setDocId(data.document.id)
   328→   328→        }
   329→   329→      }
   330→   330→      toast.success('Borrador guardado')
   331→   331→    } catch {
   332→   332→      toast.error('Error al guardar borrador')
   333→   333→    } finally {
   334→   334→      setSaving(false)
   335→   335→    }
   336→   336→  }, [user, template, docId, answers])
   337→   337→
   338→   338→  /* ---- generate document ---- */
   339→   339→  const generateDocument = useCallback(async () => {
   340→   340→    if (!user || !template) return
   341→   341→    setGenerating(true)
   342→   342→    try {
   343→   343→      const generated = replaceVariables(template.baseContent, true)
   344→   344→      const headers: HeadersInit = {
   345→   345→        'Content-Type': 'application/json',
   346→   346→        'x-user-id': user.id,
   347→   347→      }
   348→   348→      const body: Record<string, unknown> = {
   349→   349→        id: docId,
   350→   350→        title: template.name,
   351→   351→        answers,
   352→   352→        status: 'completed',
   353→   353→        generatedContent: generated,
   354→   354→      }
   355→   355→      if (!docId) {
   356→   356→        body.templateId = template.id
   357→   357→      }
   358→   358→      const method = docId ? 'PUT' : 'POST'
   359→   359→      const res = await fetch('/api/documents', {
   360→   360→        method,
   361→   361→        headers,
   362→   362→        body: JSON.stringify(body),
   363→   363→      })
   364→   364→      const data = await res.json()
   365→   365→      if (!res.ok) throw new Error(data.error || 'Error')
   366→   366→      if (!docId && data.document?.id) {
   367→   367→        setDocId(data.document.id)
   368→   368→      }
   369→   369→      toast.success('¡Documento generado exitosamente!')
   370→   370→      setTimeout(() => setCurrentPage('documents'), 800)
   371→   371→    } catch (err) {
   372→   372→      toast.error(err instanceof Error ? err.message : 'Error al generar documento')
   373→   373→    } finally {
   374→   374→      setGenerating(false)
   375→   375→    }
   376→   376→  }, [user, template, docId, answers, setCurrentPage])
   377→   377→
   378→   378→  /* ---- live preview ---- */
   379→   379→  const replaceVariables = useCallback(
   380→   380→    (content: string, final = false) => {
   381→   381→      let result = content
   382→   382→      const allFields = wizardConfig?.steps.flatMap((s) => s.fields) || []
   383→   383→      for (const field of allFields) {
   384→   384→        const placeholder = `{{${field.key}}}`
   385→   385→        const value = answers[field.key]
   386→   386→        if (value !== undefined && value !== '') {
   387→   387→          result = result.replaceAll(placeholder, String(value))
   388→   388→        }
   389→   389→      }
   390→   390→      if (final) {
   391→   391→        // Replace remaining placeholders with empty string
   392→   392→        result = result.replace(/\{\{[^}]+\}\}/g, '')
   393→   393→      }
   394→   394→      return result
   395→   395→    },
   396→   396→    [answers, wizardConfig]
   397→   397→  )
   398→   398→
   399→   399→  const previewContent = useMemo(() => {
   400→   400→    if (!template) return ''
   401→   401→    return replaceVariables(template.baseContent)
   402→   402→  }, [template, replaceVariables])
   403→   403→
   404→   404→  /* ---- auto-scroll preview ---- */
   405→   405→  useEffect(() => {
   406→   406→    if (!previewRef.current) return
   407→   407→    const currentFields = wizardConfig?.steps[currentStep]?.fields || []
   408→   408→    const visibleFields = currentFields.filter(isFieldVisible)
   409→   409→    if (visibleFields.length === 0) return
   410→   410→    const firstKey = visibleFields[0].key
   411→   411→    const placeholder = `{{${firstKey}}}`
   412→   412→    const textContent = previewRef.current.textContent || ''
   413→   413→    const idx = textContent.indexOf(firstKey)
   414→   414→    if (idx >= 0) {
   415→   415→      // Rough scroll proportion
   416→   416→      const proportion = idx / textContent.length
   417→   417→      previewRef.current.scrollTop = proportion * previewRef.current.scrollHeight
   418→   418→    }
   419→   419→  }, [currentStep])
   420→   420→
   421→   421→  /* ---- compute progress ---- */
   422→   422→  const completedSteps = useMemo(() => {
   423→   423→    if (!wizardConfig) return 0
   424→   424→    let count = 0
   425→   425→    for (let i = 0; i < Math.min(currentStep, wizardConfig.steps.length); i++) {
   426→   426→      const step = wizardConfig.steps[i]
   427→   427→      const visibleFields = step.fields.filter(isFieldVisible)
   428→   428→      if (visibleFields.length === 0 || visibleFields.every((f) => answers[f.key] !== undefined && answers[f.key] !== '')) {
   429→   429→        count++
   430→   430→      }
   431→   431→    }
   432→   432→    return count
   433→   433→  }, [wizardConfig, currentStep, answers, isFieldVisible])
   434→   434→
   435→   435→  /* ---- loading ---- */
   436→   436→  if (loading) {
   437→   437→    return (
   438→   438→      <div className="flex h-full items-center justify-center bg-[#0A1628]">
   439→   439→        <div className="flex flex-col items-center gap-4">
   440→   440→          <div className="h-10 w-10 animate-spin rounded-full border-2 border-[#C9A94E] border-t-transparent" />
   441→   441→          <p className="text-sm text-[#CBD5E1]">Cargando plantilla...</p>
   442→   442→        </div>
   443→   443→      </div>
   444→   444→    )
   445→   445→  }
   446→   446→
   447→   447→  if (!template || !wizardConfig) return null
   448→   448→
   449→   449→  const allFields = wizardConfig.steps.flatMap((s) => s.fields)
   450→   450→  const currentFields = isReviewStep ? [] : wizardConfig.steps[currentStep]?.fields || []
   451→   451→
   452→   452→  /* ====================================================================== */
   453→   453→  /*  RENDER                                                                 */
   454→   454→  /* ====================================================================== */
   455→   455→
   456→   456→  return (
   457→   457→    <TooltipProvider delayDuration={300}>
   458→   458→      <div className="flex h-full flex-col bg-[#0A1628]">
   459→   459→        {/* ---- Top bar ---- */}
   460→   460→        <div className="flex items-center justify-between border-b border-[#1E3A5F] px-6 py-3">
   461→   461→          <div className="flex items-center gap-3">
   462→   462→            <Button
   463→   463→              variant="ghost"
   464→   464→              size="sm"
   465→   465→              className="text-[#CBD5E1] hover:bg-[#152A4A] hover:text-white"
   466→   466→              onClick={() => setCurrentPage('catalog')}
   467→   467→            >
   468→   468→              <ArrowLeft className="mr-1 h-4 w-4" />
   469→   469→              Volver
   470→   470→            </Button>
   471→   471→            <div className="h-4 w-px bg-[#1E3A5F]" />
   472→   472→            <div className="flex items-center gap-2">
   473→   473→              <FileText className="h-4 w-4 text-[#C9A94E]" />
   474→   474→              <span className="text-sm font-medium text-white">{template.name}</span>
   475→   475→            </div>
   476→   476→          </div>
   477→   477→          <Button
   478→   478→            variant="outline"
   479→   479→            size="sm"
   480→   480→            className="border-[#C9A94E]/30 text-[#C9A94E] hover:bg-[#C9A94E]/10"
   481→   481→            onClick={saveDraft}
   482→   482→            disabled={saving}
   483→   483→          >
   484→   484→            <Save className="mr-1.5 h-3.5 w-3.5" />
   485→   485→            {saving ? 'Guardando...' : 'Guardar borrador'}
   486→   486→          </Button>
   487→   487→        </div>

   914→   914→          </SelectContent>
   915→   915→        </Select>
   916→   916→      )}
   917→   917→
   918→   918→      {field.type === 'boolean' && (
   919→   919→        <RadioGroup
   920→   920→          value={value !== undefined ? String(value) : undefined}
   921→   921→          onValueChange={(val) => onChange(val === 'true')}
   922→   922→          className="flex gap-4"
   923→   923→        >
   924→   924→          <div className="flex items-center gap-2">
   925→   925→            <RadioGroupItem
   926→   926→              value="true"
   927→   927→              id={`${field.key}-yes`}
   928→   928→              className="border-[#1E3A5F] text-[#C9A94E] data-[state=checked]:border-[#C9A94E] data-[state=checked]:bg-[#C9A94E]"
   929→   929→            />
   930→   930→            <Label
   931→   931→              htmlFor={`${field.key}-yes`}
   932→   932→              className="text-sm text-[#CBD5E1] cursor-pointer"
   933→   933→            >
   934→   934→              Sí
   935→   935→            </Label>
   936→   936→          </div>
   937→   937→          <div className="flex items-center gap-2">
   938→   938→            <RadioGroupItem
   939→   939→              value="false"
   940→   940→              id={`${field.key}-no`}
   941→   941→              className="border-[#1E3A5F] text-[#C9A94E] data-[state=checked]:border-[#C9A94E] data-[state=checked]:bg-[#C9A94E]"
   942→   942→            />
   943→   943→            <Label
   944→   944→              htmlFor={`${field.key}-no`}
   945→   945→              className="text-sm text-[#CBD5E1] cursor-pointer"
   946→   946→            >
   947→   947→              No
   948→   948→            </Label>
   949→   949→          </div>
   950→   950→        </RadioGroup>
   951→   951→      )}
   952→   952→    </div>
   953→   953→  )
   954→   954→}

   960→   960→function formatPreviewContent(content: string): string {
   961→   961→  // Replace newlines with paragraph breaks
   962→   962→  let html = content
   963→   963→    .split('\n')
   964→   964→    .filter((line) => line.trim() !== '')
   965→   965→    .map((line) => {
   966→   966→      // Highlight unresolved {{variables}} in gold
   967→   967→      const highlighted = line.replace(
   968→   968→        /\{\{([^}]+)\}\}/g,
   969→   969→        '<span class="inline-block rounded bg-[#C9A94E]/10 px-1.5 py-0.5 text-[#C9A94E] font-semibold text-xs mx-0.5">{{$1}}</span>'
   970→   970→      )
   971→   971→      return `<p>${highlighted}</p>`
   972→   972→    })
   973→   973→    .join('')
   974→   974→  return html
   975→   975→}
