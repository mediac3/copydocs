   362→/*  Media Field Editor (Header / Footer with Image+Dimensions)                 */
   363→/* -------------------------------------------------------------------------- */
   364→
   365→function MediaFieldEditor({
   366→  label, rawValue, onChange, defaultW, defaultH, maxH, hint
   367→}: {
   368→  label: string
   369→  rawValue: string
   370→  onChange: (encoded: string) => void
   371→  defaultW: number
   372→  defaultH: number
   373→  maxH: number
   374→  hint: string
   375→}) {
   376→  const [media, setMedia] = useState<MediaContent>(() => parseMedia(rawValue))
   377→  const [prevRaw, setPrevRaw] = useState(rawValue)
   378→
   379→  // Sync from parent if rawValue changes externally
   380→  if (rawValue !== prevRaw) {
   381→    setPrevRaw(rawValue)
   382→    setMedia(parseMedia(rawValue))
   383→  }
   384→
   385→  const update = (m: MediaContent) => {
   386→    setMedia(m)
   387→    onChange(encodeMedia(m))
   388→  }
   389→
   390→  const setType = (t: 'text' | 'image') => {
   391→    if (t === 'text') {
   392→      update({ type: 'text', text: '' })
   393→    } else {
   394→      update({ type: 'image', dataUrl: '', width: defaultW, height: defaultH })
   395→    }
   396→  }
   397→
   398→  const handleFile = (e: React.ChangeEvent<HTMLInputElement>) => {
   399→    const file = e.target.files?.[0]
   400→    if (!file) return
   401→    if (file.size > 2 * 1024 * 1024) { toast.error('La imagen no puede superar 2 MB'); return }
   402→    const reader = new FileReader()
   403→    reader.onload = () => {
   404→      const img = new Image()
   405→      img.onload = () => {
   406→        const ratio = img.width / img.height
   407→        let h = maxH
   408→        let w = Math.round(h * ratio)
   409→        if (w > 468) { w = 468; h = Math.round(w / ratio) }
   410→        update({ type: 'image', dataUrl: reader.result as string, width: w, height: h })
   411→      }
   412→      img.src = reader.result as string
   413→    }
   414→    reader.readAsDataURL(file)
   415→    e.target.value = ''
   416→  }
   417→
   418→  return (
   419→    <div className="grid gap-2">
   420→      <Label className="text-white/70">{label}</Label>
   421→      <div className="flex gap-1 mb-1">
   422→        {(['text', 'image'] as const).map(t => (
   423→          <button
   424→            key={t}
   425→            type="button"
   426→            onClick={() => setType(t)}
   427→            className={`flex-1 rounded-md px-3 py-1.5 text-[11px] font-medium transition-colors ${
   428→              media.type === t
   429→                ? 'bg-[#28A745]/20 text-[#28A745] border border-[#28A745]/30'
   430→                : 'bg-white/5 text-white/40 border border-white/10 hover:bg-white/10'
   431→            }`}
   432→          >
   433→            {t === 'text' ? 'Texto' : 'Imagen'}
   434→          </button>
   435→        ))}
   436→      </div>
   437→      {media.type === 'text' ? (
   438→        <Textarea
   439→          value={media.text || ''}
   440→          onChange={(e) => update({ type: 'text', text: e.target.value })}
   441→          placeholder={`Texto del ${label.toLowerCase()}...`}
   442→          rows={3}
   443→          className="border-white/10 bg-white/5 text-white placeholder:text-white/30"
   444→        />
   445→      ) : (
   446→        <div className="space-y-2">
   447→          {media.dataUrl ? (
   448→            <div className="flex items-center justify-center rounded-lg border border-dashed border-white/15 bg-white/[0.02] p-3">
   449→              <img
   450→                src={media.dataUrl}
   451→                alt="Preview"
   452→                style={{ width: `${Math.min(media.width || defaultW, 300)}px`, height: 'auto' }}
   453→                className="rounded"
   454→              />
   455→            </div>
   456→          ) : null}
   457→          <label className="flex cursor-pointer items-center justify-center gap-2 rounded-md border border-white/10 bg-white/5 px-3 py-2 text-xs text-white/50 hover:bg-white/10 hover:text-white/70 transition-colors">
   458→            <Download className="h-3.5 w-3.5" />
   459→            {media.dataUrl ? 'Cambiar imagen' : 'Subir imagen (PNG, JPG, WebP)'}
   460→            <input type="file" accept="image/png,image/jpeg,image/webp" className="hidden" onChange={handleFile} />
   461→          </label>
   462→          <div className="grid grid-cols-2 gap-2">
   463→            <div className="grid gap-1">
   464→              <span className="text-[9px] uppercase tracking-wider text-white/25">Ancho (px)</span>
   465→              <Input
   466→                type="number"
   467→                value={media.width || defaultW}
   468→                onChange={(e) => update({ ...media, width: parseInt(e.target.value) || defaultW })}
   469→                className="h-7 border-white/10 bg-white/5 text-white text-xs"
   470→              />
   471→            </div>
   472→            <div className="grid gap-1">
   473→              <span className="text-[9px] uppercase tracking-wider text-white/25">Alto (px)</span>
   474→              <Input
   475→                type="number"
   476→                value={media.height || defaultH}
   477→                onChange={(e) => update({ ...media, height: parseInt(e.target.value) || defaultH })}
   478→                className="h-7 border-white/10 bg-white/5 text-white text-xs"
   479→              />
   480→            </div>
   481→          </div>
   482→          <p className="text-[10px] text-white/25 text-center">{hint}. Max 2 MB.</p>
   483→          {media.dataUrl && (
   484→            <button
   485→              type="button"
   486→              onClick={() => update({ type: 'image', dataUrl: '', width: defaultW, height: defaultH })}
   487→              className="w-full text-center text-[11px] text-red-400/60 hover:text-red-400 transition-colors"
   488→            >
   489→              Eliminar imagen
   490→            </button>
   491→          )}
   492→        </div>
   493→      )}
   494→    </div>
   495→  )
   496→}
