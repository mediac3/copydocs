    79→
    80→const CONTACT_TYPE_MAP: Record<string, string[]> = {
    81→  arrendador: ['arrendador', 'propietario'],
    82→  arrendatario: ['arrendatario', 'inquilino'],
    83→  fiador: ['fiador', 'codeudor'],
    84→  comprador: ['comprador', 'adquirente'],
    85→  vendedor: ['vendedor', 'enajenante'],
    86→  empleador: ['empleador', 'patrono', 'empresa'],
    87→  trabajador: ['trabajador', 'empleado'],
    88→  demandante: ['demandante', 'actor'],
    89→  demandado: ['demandado'],
    90→  deudor: ['deudor'],
    91→  acreedor: ['acreedor'],
    92→}
    93→
    94→function detectContactType(fieldKey: string): string | null {
    95→  const lower = fieldKey.toLowerCase()
    96→  for (const [contactType, prefixes] of Object.entries(CONTACT_TYPE_MAP)) {
    97→    for (const prefix of prefixes) {
    98→      if (lower.includes(prefix)) return contactType
    99→    }
   100→  }
   101→  return null
   102→}

   949→
   950→interface FieldRendererProps {
   951→  field: WizardField
   952→  value: string | number | boolean | undefined
   953→  onChange: (value: string | number | boolean) => void
   954→  onFocus: () => void
   955→  contacts: Contact[]
   956→  onApplyContact: (contact: Contact) => void
   957→  onDismissContacts: () => void
   958→}
   959→
   960→function FieldRenderer({
   961→  field,
   962→  value,
   963→  onChange,
   964→  onFocus,
   965→  contacts,
   966→  onApplyContact,
   967→  onDismissContacts,
   968→}: FieldRendererProps) {
   969→  const fieldRef = useRef<HTMLDivElement>(null)
   970→
   971→  // Close contacts dropdown on outside click
   972→  useEffect(() => {
   973→    if (contacts.length === 0) return
   974→    function handleClick(e: MouseEvent) {
   975→      if (fieldRef.current && !fieldRef.current.contains(e.target as Node)) {
   976→        onDismissContacts()
   977→      }
   978→    }
   979→    document.addEventListener('mousedown', handleClick)
   980→    return () => document.removeEventListener('mousedown', handleClick)
   981→  }, [contacts.length, onDismissContacts])
   982→
   983→  const labelElement = (
   984→    <div className="mb-1.5 flex items-center gap-1.5">
   985→      <Label className="text-xs font-medium text-[#CBD5E1]">{field.label}</Label>
   986→      {field.tooltip && (
   987→        <Tooltip>
   988→          <TooltipTrigger asChild>
   989→            <button type="button" className="text-[#475569] hover:text-[#C9A94E] transition-colors">
   990→              <Info className="h-3.5 w-3.5" />
   991→            </button>
   992→          </TooltipTrigger>
   993→          <TooltipContent
   994→            side="right"
   995→            className="max-w-[280px] border-[#1E3A5F] bg-[#152A4A] text-[#CBD5E1] text-xs shadow-xl"
   996→          >
   997→            <p className="leading-relaxed">{field.tooltip}</p>
   998→          </TooltipContent>
   999→        </Tooltip>
  1000→      )}
  1001→    </div>
  1002→  )
  1003→
  1004→  return (
  1005→    <div ref={fieldRef} className="relative animate-fade-in-up">
  1006→      {labelElement}
  1007→
  1008→      {/* Contact autocomplete dropdown */}
  1009→      {contacts.length > 0 && (
  1010→        <div className="absolute bottom-full left-0 right-0 mb-1 z-50 rounded-lg border border-[#C9A94E]/30 bg-[#152A4A] shadow-xl animate-fade-in-up">
  1011→          <div className="px-3 py-1.5 text-[10px] font-medium text-[#C9A94E] uppercase tracking-wider">
  1012→            Contactos guardados
  1013→          </div>
  1014→          {contacts.slice(0, 5).map((contact) => (
  1015→            <button
  1016→              key={contact.id}
  1017→              type="button"
  1018→              className="flex w-full items-center gap-3 px-3 py-2 text-left text-xs text-[#CBD5E1] hover:bg-[#1E3A5F] transition-colors first:rounded-t-lg last:rounded-b-lg"
  1019→              onClick={() => onApplyContact(contact)}
  1020→            >
  1021→              <div className="flex h-7 w-7 items-center justify-center rounded-full bg-[#C9A94E]/10 text-[#C9A94E]">
  1022→                {contact.name.charAt(0).toUpperCase()}
  1023→              </div>
  1024→              <div className="flex-1 min-w-0">
  1025→                <p className="font-medium truncate">{contact.name}</p>
  1026→                <p className="text-[10px] text-[#475569] truncate">
  1027→                  {[contact.documentType, contact.documentNumber].filter(Boolean).join(': ') ||
  1028→                    contact.companyName ||
  1029→                    ''}
  1030→                </p>
  1031→              </div>
  1032→            </button>
  1033→          ))}
  1034→        </div>
  1035→      )}
  1036→
  1037→      {/* Field input */}
  1038→      {field.type === 'text' && (
  1039→        <Input
  1040→          value={String(value ?? '')}
  1041→          onChange={(e) => onChange(e.target.value)}
  1042→          onFocus={onFocus}
  1043→          placeholder={field.label}
  1044→          className="border-[#1E3A5F] bg-[#0F1F38] text-white placeholder:text-[#475569] focus:border-[#C9A94E] focus:ring-[#C9A94E]/20 h-10 text-sm"
  1045→        />
  1046→      )}
  1047→
  1048→      {field.type === 'number' && (
  1049→        <Input
  1050→          type="number"
  1051→          value={value !== undefined ? String(value) : ''}
  1052→          onChange={(e) => onChange(e.target.value === '' ? '' : Number(e.target.value))}
  1053→          placeholder={field.label}
  1054→          className="border-[#1E3A5F] bg-[#0F1F38] text-white placeholder:text-[#475569] focus:border-[#C9A94E] focus:ring-[#C9A94E]/20 h-10 text-sm"
  1055→        />
  1056→      )}
  1057→
  1058→      {field.type === 'date' && (
  1059→        <Input
  1060→          type="date"
  1061→          value={String(value ?? '')}
  1062→          onChange={(e) => onChange(e.target.value)}
  1063→          className="border-[#1E3A5F] bg-[#0F1F38] text-white placeholder:text-[#475569] focus:border-[#C9A94E] focus:ring-[#C9A94E]/20 h-10 text-sm [color-scheme:dark]"
  1064→        />

  1066→

  1069→          value={String(value ?? '')}
  1070→          onChange={(e) => onChange(e.target.value)}
  1071→          placeholder={field.label}
  1072→          rows={3}
  1073→          className="border-[#1E3A5F] bg-[#0F1F38] text-white placeholder:text-[#475569] focus:border-[#C9A94E] focus:ring-[#C9A94E]/20 text-sm resize-none"
  1074→        />
  1075→      )}
  1076→
  1077→      {field.type === 'select' && field.options && (
  1078→        <Select
  1079→          value={value !== undefined ? String(value) : undefined}
  1080→          onValueChange={(val) => onChange(val)}
  1081→        >
  1082→          <SelectTrigger className="border-[#1E3A5F] bg-[#0F1F38] text-white focus:border-[#C9A94E] focus:ring-[#C9A94E]/20 h-10 text-sm">
  1083→            <SelectValue placeholder={`Seleccionar ${field.label.toLowerCase()}`} />
  1084→          </SelectTrigger>
  1085→          <SelectContent className="border-[#1E3A5F] bg-[#152A4A] text-[#CBD5E1]">
  1086→            {field.options.map((opt) => (
  1087→              <SelectItem key={opt} value={opt} className="text-sm focus:bg-[#1E3A5F] focus:text-white">
  1088→                {opt}
  1089→              </SelectItem>
  1090→            ))}
  1091→          </SelectContent>
  1092→        </Select>
  1093→      )}
  1094→
  1095→      {field.type === 'boolean' && (
  1096→        <RadioGroup
  1097→          value={value !== undefined ? String(value) : undefined}
  1098→          onValueChange={(val) => onChange(val === 'true')}
  1099→          className="flex gap-4"
  1100→        >
  1101→          <div className="flex items-center gap-2">
  1102→            <RadioGroupItem
  1103→              value="true"
  1104→              id={`${field.key}-yes`}
  1105→              className="border-[#1E3A5F] text-[#C9A94E] data-[state=checked]:border-[#C9A94E] data-[state=checked]:bg-[#C9A94E]"
  1106→            />
  1107→            <Label
  1108→              htmlFor={`${field.key}-yes`}
  1109→              className="text-sm text-[#CBD5E1] cursor-pointer"
  1110→            >
  1111→              Sí
  1112→            </Label>
  1113→          </div>
  1114→          <div className="flex items-center gap-2">
  1115→            <RadioGroupItem
  1116→              value="false"
  1117→              id={`${field.key}-no`}
  1118→              className="border-[#1E3A5F] text-[#C9A94E] data-[state=checked]:border-[#C9A94E] data-[state=checked]:bg-[#C9A94E]"
  1119→            />
  1120→            <Label
  1121→              htmlFor={`${field.key}-no`}
  1122→              className="text-sm text-[#CBD5E1] cursor-pointer"
  1123→            >
  1124→              No
  1125→            </Label>
  1126→          </div>
  1127→        </RadioGroup>
  1128→      )}
  1129→    </div>
  1130→  )
  1131→}

  1137→function formatPreviewContent(content: string): string {
  1138→  // Replace newlines with paragraph breaks
  1139→  let html = content
  1140→    .split('\n')
  1141→    .filter((line) => line.trim() !== '')
  1142→    .map((line) => {
  1143→      // Highlight unresolved {{variables}} in gold
  1144→      const highlighted = line.replace(
  1145→        /\{\{([^}]+)\}\}/g,
  1146→        '<span class="inline-block rounded bg-[#C9A94E]/10 px-1.5 py-0.5 text-[#C9A94E] font-semibold text-xs mx-0.5">{{$1}}</span>'
  1147→      )
  1148→      return `<p>${highlighted}</p>`
  1149→    })
  1150→    .join('')
  1151→  return html
  1152→}
