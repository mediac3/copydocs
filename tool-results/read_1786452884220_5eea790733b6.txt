    76→/* -------------------------------------------------------------------------- */

    79→
    80→const CONTACT_TYPE_MAP: Record<string, string[]> = {
    81→  arrendador: ['arrendador', 'propietario'],
    82→  arrendatario: ['arrendatario', 'inquilino'],
    83→  fiador: ['fiador', 'codeudor'],
    84→  comprador: ['comprador', 'adquirente'],
    85→  vendedor: ['vendedor', 'enajenante'],
    86→  empleador: ['empleador', 'patrono', 'empresa'],
    87→  trabajador: ['trabajador', 'empleado'],
    88→  demandante: ['demandante', 'actor'],
    89→  demandado: ['demandado'],
    90→  deudor: ['deudor'],
    91→  acreedor: ['acreedor'],
    92→}
    93→
    94→function detectContactType(fieldKey: string): string | null {
    95→  const lower = fieldKey.toLowerCase()
    96→  for (const [contactType, prefixes] of Object.entries(CONTACT_TYPE_MAP)) {
    97→    for (const prefix of prefixes) {
    98→      if (lower.includes(prefix)) return contactType
    99→    }
   100→  }
   101→  return null
   102→}

   954→
   955→interface FieldRendererProps {
   956→  field: WizardField
   957→  value: string | number | boolean | undefined
   958→  onChange: (value: string | number | boolean) => void
   959→  onFocus: () => void
   960→  contacts: Contact[]
   961→  onApplyContact: (contact: Contact) => void
   962→  onDismissContacts: () => void
   963→}
   964→
   965→function FieldRenderer({
   966→  field,
   967→  value,
   968→  onChange,
   969→  onFocus,
   970→  contacts,
   971→  onApplyContact,
   972→  onDismissContacts,
   973→}: FieldRendererProps) {
   974→  const fieldRef = useRef<HTMLDivElement>(null)
   975→
   976→  // Close contacts dropdown on outside click
   977→  useEffect(() => {
   978→    if (contacts.length === 0) return
   979→    function handleClick(e: MouseEvent) {
   980→      if (fieldRef.current && !fieldRef.current.contains(e.target as Node)) {
   981→        onDismissContacts()
   982→      }
   983→    }
   984→    document.addEventListener('mousedown', handleClick)
   985→    return () => document.removeEventListener('mousedown', handleClick)
   986→  }, [contacts.length, onDismissContacts])
   987→
   988→  const labelElement = (
   989→    <div className="mb-1.5 flex items-center gap-1.5">
   990→      <Label className="text-xs font-medium text-[#CBD5E1]">{field.label}</Label>
   991→      {field.tooltip && (
   992→        <Tooltip>
   993→          <TooltipTrigger asChild>
   994→            <button type="button" className="text-[#475569] hover:text-[#C9A94E] transition-colors">
   995→              <Info className="h-3.5 w-3.5" />
   996→            </button>
   997→          </TooltipTrigger>
   998→          <TooltipContent
   999→            side="right"
  1000→            className="max-w-[280px] border-[#1E3A5F] bg-[#152A4A] text-[#CBD5E1] text-xs shadow-xl"
  1001→          >
  1002→            <p className="leading-relaxed">{field.tooltip}</p>
  1003→          </TooltipContent>
  1004→        </Tooltip>
  1005→      )}
  1006→    </div>
  1007→  )
  1008→
  1009→  return (
  1010→    <div ref={fieldRef} className="relative animate-fade-in-up">
  1011→      {labelElement}
  1012→
  1013→      {/* Contact autocomplete dropdown */}
  1014→      {contacts.length > 0 && (
  1015→        <div className="absolute bottom-full left-0 right-0 mb-1 z-50 rounded-lg border border-[#C9A94E]/30 bg-[#152A4A] shadow-xl animate-fade-in-up">

  1017→            Contactos guardados
  1018→          </div>
  1019→          {contacts.slice(0, 5).map((contact) => (
  1020→            <button
  1021→              key={contact.id}
  1022→              type="button"
  1023→              className="flex w-full items-center gap-3 px-3 py-2 text-left text-xs text-[#CBD5E1] hover:bg-[#1E3A5F] transition-colors first:rounded-t-lg last:rounded-b-lg"
  1024→              onClick={() => onApplyContact(contact)}
  1025→            >
  1026→              <div className="flex h-7 w-7 items-center justify-center rounded-full bg-[#C9A94E]/10 text-[#C9A94E]">
  1027→                {contact.name.charAt(0).toUpperCase()}
  1028→              </div>
  1029→              <div className="flex-1 min-w-0">
  1030→                <p className="font-medium truncate">{contact.name}</p>
  1031→                <p className="text-[10px] text-[#475569] truncate">
  1032→                  {[contact.documentType, contact.documentNumber].filter(Boolean).join(': ') ||
  1033→                    contact.companyName ||
  1034→                    ''}
  1035→                </p>
  1036→              </div>
  1037→            </button>
  1038→          ))}

  1040→      )}
  1041→
  1042→      {/* Field input */}
  1043→      {field.type === 'text' && (
  1044→        <Input
  1045→          value={String(value ?? '')}
  1046→          onChange={(e) => onChange(e.target.value)}
  1047→          onFocus={onFocus}
  1048→          placeholder={field.label}
  1049→          className="border-[#1E3A5F] bg-[#0F1F38] text-white placeholder:text-[#475569] focus:border-[#C9A94E] focus:ring-[#C9A94E]/20 h-10 text-sm"
  1050→        />
  1051→      )}
  1052→
  1053→      {field.type === 'number' && (
  1054→        <Input
  1055→          type="number"
  1056→          value={value !== undefined ? String(value) : ''}
  1057→          onChange={(e) => onChange(e.target.value === '' ? '' : Number(e.target.value))}
  1058→          placeholder={field.label}
  1059→          className="border-[#1E3A5F] bg-[#0F1F38] text-white placeholder:text-[#475569] focus:border-[#C9A94E] focus:ring-[#C9A94E]/20 h-10 text-sm"
  1060→        />
  1061→      )}
  1062→
  1063→      {field.type === 'date' && (
  1064→        <Input
  1065→          type="date"
  1066→          value={String(value ?? '')}
  1067→          onChange={(e) => onChange(e.target.value)}
  1068→          className="border-[#1E3A5F] bg-[#0F1F38] text-white placeholder:text-[#475569] focus:border-[#C9A94E] focus:ring-[#C9A94E]/20 h-10 text-sm [color-scheme:dark]"

  1079→        />
  1080→      )}
  1081→
  1082→      {field.type === 'select' && field.options && (
  1083→        <Select
  1084→          value={value !== undefined ? String(value) : undefined}
  1085→          onValueChange={(val) => onChange(val)}
  1086→        >
  1087→          <SelectTrigger className="border-[#1E3A5F] bg-[#0F1F38] text-white focus:border-[#C9A94E] focus:ring-[#C9A94E]/20 h-10 text-sm">
  1088→            <SelectValue placeholder={`Seleccionar ${field.label.toLowerCase()}`} />
  1089→          </SelectTrigger>
  1090→          <SelectContent className="border-[#1E3A5F] bg-[#152A4A] text-[#CBD5E1]">
  1091→            {field.options.map((opt) => (
  1092→              <SelectItem key={opt} value={opt} className="text-sm focus:bg-[#1E3A5F] focus:text-white">
  1093→                {opt}
  1094→              </SelectItem>
  1095→            ))}
  1096→          </SelectContent>
  1097→        </Select>
  1098→      )}
  1099→
  1100→      {field.type === 'boolean' && (
  1101→        <RadioGroup
  1102→          value={value !== undefined ? String(value) : undefined}
  1103→          onValueChange={(val) => onChange(val === 'true')}
  1104→          className="flex gap-4"
  1105→        >
  1106→          <div className="flex items-center gap-2">
  1107→            <RadioGroupItem
  1108→              value="true"
  1109→              id={`${field.key}-yes`}
  1110→              className="border-[#1E3A5F] text-[#C9A94E] data-[state=checked]:border-[#C9A94E] data-[state=checked]:bg-[#C9A94E]"
  1111→            />
  1112→            <Label
  1113→              htmlFor={`${field.key}-yes`}
  1114→              className="text-sm text-[#CBD5E1] cursor-pointer"
  1115→            >
  1116→              Sí
  1117→            </Label>
  1118→          </div>
  1119→          <div className="flex items-center gap-2">
  1120→            <RadioGroupItem
  1121→              value="false"
  1122→              id={`${field.key}-no`}
  1123→              className="border-[#1E3A5F] text-[#C9A94E] data-[state=checked]:border-[#C9A94E] data-[state=checked]:bg-[#C9A94E]"
  1124→            />
  1125→            <Label
  1126→              htmlFor={`${field.key}-no`}
  1127→              className="text-sm text-[#CBD5E1] cursor-pointer"
  1128→            >
  1129→              No
  1130→            </Label>
  1131→          </div>
  1132→        </RadioGroup>
  1133→      )}
  1134→    </div>
  1135→  )
  1136→}

  1141→
  1142→function formatPreviewContent(content: string): string {
  1143→  // Replace newlines with paragraph breaks
  1144→  let html = content
  1145→    .split('\n')
  1146→    .filter((line) => line.trim() !== '')
  1147→    .map((line) => {
  1148→      // Highlight unresolved {{variables}} in gold
  1149→      const highlighted = line.replace(
  1150→        /\{\{([^}]+)\}\}/g,
  1151→        '<span class="inline-block rounded bg-[#C9A94E]/10 px-1.5 py-0.5 text-[#C9A94E] font-semibold text-xs mx-0.5">{{$1}}</span>'
  1152→      )
  1153→      return `<p>${highlighted}</p>`
  1154→    })
  1155→    .join('')
  1156→  return html
  1157→}
