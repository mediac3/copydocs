   779→/* -------------------------------------------------------------------------- */
   780→/*  Field Renderer                                                           */
   781→/* -------------------------------------------------------------------------- */
   782→
   783→interface FieldRendererProps {
   784→  field: WizardField
   785→  value: string | number | boolean | undefined
   786→  onChange: (value: string | number | boolean) => void
   787→  onFocus: () => void
   788→  contacts: Contact[]
   789→  onApplyContact: (contact: Contact) => void
   790→  onDismissContacts: () => void
   791→}
   792→
   793→function FieldRenderer({
   794→  field,
   795→  value,
   796→  onChange,
   797→  onFocus,
   798→  contacts,
   799→  onApplyContact,
   800→  onDismissContacts,
   801→}: FieldRendererProps) {
   802→  const fieldRef = useRef<HTMLDivElement>(null)
   803→

   805→  useEffect(() => {
   806→    if (contacts.length === 0) return
   807→    function handleClick(e: MouseEvent) {
   808→      if (fieldRef.current && !fieldRef.current.contains(e.target as Node)) {
   809→        onDismissContacts()
   810→      }
   811→    }
   812→    document.addEventListener('mousedown', handleClick)
   813→    return () => document.removeEventListener('mousedown', handleClick)
   814→  }, [contacts.length, onDismissContacts])
   815→
   816→  const labelElement = (
   817→    <div className="mb-1.5 flex items-center gap-1.5">
   818→      <Label className="text-xs font-medium text-[#CBD5E1]">{field.label}</Label>
   819→      {field.tooltip && (
   820→        <Tooltip>
   821→          <TooltipTrigger asChild>
   822→            <button type="button" className="text-[#475569] hover:text-[#C9A94E] transition-colors">
   823→              <Info className="h-3.5 w-3.5" />
   824→            </button>
   825→          </TooltipTrigger>
   826→          <TooltipContent
   827→            side="right"
   828→            className="max-w-[280px] border-[#1E3A5F] bg-[#152A4A] text-[#CBD5E1] text-xs shadow-xl"
   829→          >
   830→            <p className="leading-relaxed">{field.tooltip}</p>
   831→          </TooltipContent>
   832→        </Tooltip>
   833→      )}
   834→    </div>
   835→  )
   836→
   837→  return (
   838→    <div ref={fieldRef} className="relative animate-fade-in-up">
   839→      {labelElement}
   840→
   841→      {/* Contact autocomplete dropdown */}
   842→      {contacts.length > 0 && (
   843→        <div className="absolute bottom-full left-0 right-0 mb-1 z-50 rounded-lg border border-[#C9A94E]/30 bg-[#152A4A] shadow-xl animate-fade-in-up">
   844→          <div className="px-3 py-1.5 text-[10px] font-medium text-[#C9A94E] uppercase tracking-wider">
   845→            Contactos guardados
   846→          </div>
   847→          {contacts.slice(0, 5).map((contact) => (
   848→            <button
   849→              key={contact.id}
   850→              type="button"
   851→              className="flex w-full items-center gap-3 px-3 py-2 text-left text-xs text-[#CBD5E1] hover:bg-[#1E3A5F] transition-colors first:rounded-t-lg last:rounded-b-lg"
   852→              onClick={() => onApplyContact(contact)}
   853→            >
   854→              <div className="flex h-7 w-7 items-center justify-center rounded-full bg-[#C9A94E]/10 text-[#C9A94E]">
   855→                {contact.name.charAt(0).toUpperCase()}
   856→              </div>

   858→                <p className="font-medium truncate">{contact.name}</p>
   859→                <p className="text-[10px] text-[#475569] truncate">
   860→                  {[contact.documentType, contact.documentNumber].filter(Boolean).join(': ') ||
   861→                    contact.companyName ||
   862→                    ''}
   863→                </p>
   864→              </div>
   865→            </button>
   866→          ))}
   867→        </div>

   869→
   870→      {/* Field input */}
   871→      {field.type === 'text' && (
   872→        <Input
   873→          value={String(value ?? '')}
   874→          onChange={(e) => onChange(e.target.value)}
   875→          onFocus={onFocus}
   876→          placeholder={field.label}
   877→          className="border-[#1E3A5F] bg-[#0F1F38] text-white placeholder:text-[#475569] focus:border-[#C9A94E] focus:ring-[#C9A94E]/20 h-10 text-sm"
   878→        />
   879→      )}
   880→
   881→      {field.type === 'number' && (
   882→        <Input
   883→          type="number"
   884→          value={value !== undefined ? String(value) : ''}
   885→          onChange={(e) => onChange(e.target.value === '' ? '' : Number(e.target.value))}
   886→          placeholder={field.label}
   887→          className="border-[#1E3A5F] bg-[#0F1F38] text-white placeholder:text-[#475569] focus:border-[#C9A94E] focus:ring-[#C9A94E]/20 h-10 text-sm"
   888→        />
   889→      )}
   890→
   891→      {field.type === 'date' && (
   892→        <Input
   893→          type="date"
   894→          value={String(value ?? '')}
   895→          onChange={(e) => onChange(e.target.value)}
   896→          className="border-[#1E3A5F] bg-[#0F1F38] text-white placeholder:text-[#475569] focus:border-[#C9A94E] focus:ring-[#C9A94E]/20 h-10 text-sm [color-scheme:dark]"
   897→        />
   898→      )}
   899→
   900→      {field.type === 'textarea' && (
   901→        <Textarea
   902→          value={String(value ?? '')}
   903→          onChange={(e) => onChange(e.target.value)}
   904→          placeholder={field.label}
   905→          rows={3}
   906→          className="border-[#1E3A5F] bg-[#0F1F38] text-white placeholder:text-[#475569] focus:border-[#C9A94E] focus:ring-[#C9A94E]/20 text-sm resize-none"
   907→        />
   908→      )}
   909→
   910→      {field.type === 'select' && field.options && (
   911→        <Select
   912→          value={value !== undefined ? String(value) : undefined}
   913→          onValueChange={(val) => onChange(val)}
   914→        >
   915→          <SelectTrigger className="border-[#1E3A5F] bg-[#0F1F38] text-white focus:border-[#C9A94E] focus:ring-[#C9A94E]/20 h-10 text-sm">
   916→            <SelectValue placeholder={`Seleccionar ${field.label.toLowerCase()}`} />
   917→          </SelectTrigger>
   918→          <SelectContent className="border-[#1E3A5F] bg-[#152A4A] text-[#CBD5E1]">
   919→            {field.options.map((opt) => (
   920→              <SelectItem key={opt} value={opt} className="text-sm focus:bg-[#1E3A5F] focus:text-white">
   921→                {opt}
   922→              </SelectItem>
   923→            ))}
   924→          </SelectContent>
   925→        </Select>
   926→      )}
   927→
   928→      {field.type === 'boolean' && (
   929→        <RadioGroup
   930→          value={value !== undefined ? String(value) : undefined}
   931→          onValueChange={(val) => onChange(val === 'true')}
   932→          className="flex gap-4"
   933→        >
   934→          <div className="flex items-center gap-2">
   935→            <RadioGroupItem
   936→              value="true"
   937→              id={`${field.key}-yes`}
   938→              className="border-[#1E3A5F] text-[#C9A94E] data-[state=checked]:border-[#C9A94E] data-[state=checked]:bg-[#C9A94E]"
   939→            />
   940→            <Label
   941→              htmlFor={`${field.key}-yes`}
   942→              className="text-sm text-[#CBD5E1] cursor-pointer"
   943→            >
   944→              Sí
   945→            </Label>
   946→          </div>
   947→          <div className="flex items-center gap-2">
   948→            <RadioGroupItem
   949→              value="false"
   950→              id={`${field.key}-no`}
   951→              className="border-[#1E3A5F] text-[#C9A94E] data-[state=checked]:border-[#C9A94E] data-[state=checked]:bg-[#C9A94E]"
   952→            />
   953→            <Label
   954→              htmlFor={`${field.key}-no`}
   955→              className="text-sm text-[#CBD5E1] cursor-pointer"
   956→            >
   957→              No
   958→            </Label>
   959→          </div>
   960→        </RadioGroup>
   961→      )}
   962→    </div>
   963→  )
   964→}

   966→/* -------------------------------------------------------------------------- */
   967→/*  Preview formatting                                                       */
   968→/* -------------------------------------------------------------------------- */
   969→
   970→function formatPreviewContent(content: string): string {
   971→  // Replace newlines with paragraph breaks
   972→  let html = content
   973→    .split('\n')
   974→    .filter((line) => line.trim() !== '')
   975→    .map((line) => {
   976→      // Highlight unresolved {{variables}} in gold
   977→      const highlighted = line.replace(
   978→        /\{\{([^}]+)\}\}/g,
   979→        '<span class="inline-block rounded bg-[#C9A94E]/10 px-1.5 py-0.5 text-[#C9A94E] font-semibold text-xs mx-0.5">{{$1}}</span>'
   980→      )
   981→      return `<p>${highlighted}</p>`
   982→    })
   983→    .join('')
   984→  return html
   985→}
