   898→   997→  1056→function formatPreviewContent(content: string): string {
   899→   998→  1057→  // Replace newlines with paragraph breaks
   900→   999→  1058→  let html = content
   901→  1000→  1059→    .split('\n')
   902→  1001→  1060→    .filter((line) => line.trim() !== '')
   903→  1002→  1061→    .map((line) => {
   904→  1003→  1062→      // Highlight unresolved {{variables}} in gold
   905→  1004→  1063→      const highlighted = line.replace(
   906→  1005→  1064→        /\{\{([^}]+)\}\}/g,
   907→  1006→  1065→        '<span class="inline-block rounded bg-[#C9A94E]/10 px-1.5 py-0.5 text-[#C9A94E] font-semibold text-xs mx-0.5">{{$1}}</span>'
   908→  1007→  1066→      )
   909→  1008→  1067→      return `<p>${highlighted}</p>`
   910→  1009→  1068→    })
   911→  1010→  1069→    .join('')
   912→  1011→  1070→  return html
   913→  1012→  1071→}
