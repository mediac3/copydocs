   307→function InlineTerms() {
   308→  const { authHeaders } = useAppStore() as { authHeaders: () => Record<string, string> }
   309→  const [terms, setTerms] = useState('')
   310→  const [loading, setLoading] = useState(true)
   311→  const [saving, setSaving] = useState(false)
   312→
   313→  useEffect(() => {
   314→    setLoading(true)
   315→    fetch('/api/settings').then(r => r.json()).then(data => {
   316→      setTerms(data.terms_and_conditions || '')
   317→    }).catch(() => {}).finally(() => setLoading(false))
   318→  }, [activeTab])
   319→
   320→  const handleSave = async () => {
   321→    setSaving(true)
   322→    try {
   323→      const res = await fetch('/api/settings', {
   324→        method: 'PUT',
   325→        headers: authHeaders(),
   326→        body: JSON.stringify({ terms_and_conditions: terms }),
   327→      })
   328→      if (res.ok) toast.success('Términos y condiciones actualizados')
   329→      else toast.error('Error al guardar')
   330→    } catch { toast.error('Error de conexión') }
   331→    finally { setSaving(false) }
   332→  }
   333→
   334→  return (
   335→    <div className="space-y-6">
   336→      <div>
   337→        <h2 className="text-lg font-semibold text-white">Términos y Condiciones</h2>
   338→        <p className="text-xs text-white/40">Configura los términos que los visitantes deben aceptar antes de generar un documento.</p>
   339→      </div>
   340→      <Card className="border-white/5 bg-[#0F1D32]/80 backdrop-blur-sm">
   341→        <CardContent className="p-6">
   342→          {loading ? (
   343→            <div className="h-64 animate-pulse rounded-lg bg-white/[0.04]" />
   344→          ) : (
   345→            <>
   346→              <div className="grid gap-2 mb-4">
   347→                <Label className="text-white/70">Contenido de Términos y Condiciones</Label>
   348→                <p className="text-xs text-white/30">Este texto se mostrará a los visitantes antes de iniciar un documento. Usa texto plano.</p>
   349→              </div>
   350→              <Textarea
   351→                value={terms}
   352→                onChange={(e) => setTerms(e.target.value)}
   353→                placeholder="Escribe aquí los términos y condiciones del servicio..."
   354→                rows={16}
   355→                className="border-white/10 bg-white/5 text-white placeholder:text-white/30 text-sm leading-relaxed"
   356→              />
   357→            </>
   358→          )}
   359→        </CardContent>
   360→      </Card>
   361→      <div className="flex justify-end">
   362→        <Button onClick={handleSave} disabled={saving} className="bg-[#C9A94E] text-[#0A1628] hover:bg-[#D4B965] font-semibold min-w-[120px]">
   363→          {saving ? 'Guardando...' : 'Guardar Términos'}
   364→        </Button>
   365→      </div>
   366→    </div>
   367→  )
