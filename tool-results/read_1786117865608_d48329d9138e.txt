   135→   135→function useDebounce<T>(value: T, delayMs: number): T {
   136→   136→  const [debounced, setDebounced] = useState(value)
   137→   137→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   138→   138→
   139→   139→  useEffect(() => {
   140→   140→    if (timerRef.current) clearTimeout(timerRef.current)
   141→   141→    timerRef.current = setTimeout(() => setDebounced(value), delayMs)
   142→   142→    return () => {
   143→   143→      if (timerRef.current) clearTimeout(timerRef.current)
   144→   144→    }
   145→   145→  }, [value, delayMs])
   146→   146→
   147→   147→  return debounced
   148→   148→}

   346→   346→
   347→   347→export default function DocumentsPage() {
   348→   348→  const { user, setCurrentPage, startWizard } = useAppStore()
   349→   349→
   350→   350→  /* ---- State ---- */
   351→   351→  const [activeTab, setActiveTab] = useState<TabFilter>('all')
   352→   352→  const [searchQuery, setSearchQuery] = useState('')
   353→   353→  const debouncedSearch = useDebounce(searchQuery, 300)
   354→   354→  const [documents, setDocuments] = useState<Document[]>([])
   355→   355→  const [loading, setLoading] = useState(true)
   356→   356→
   357→   357→  /* Delete dialog */
   358→   358→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false)
   359→   359→  const [docToDelete, setDocToDelete] = useState<Document | null>(null)
   360→   360→  const [deleting, setDeleting] = useState(false)
   361→   361→
   362→   362→  /* Action loading helpers */
   363→   363→  const [actionLoading, setActionLoading] = useState<string | null>(null)
   364→   364→
   365→   365→  /* ---- Fetch documents ---- */
   366→   366→  const fetchDocuments = useCallback(async () => {
   367→   367→    if (!user) return
   368→   368→    setLoading(true)
   369→   369→    try {
   370→   370→      const params = new URLSearchParams()
   371→   371→      params.set('status', activeTab)
   372→   372→      if (debouncedSearch.trim()) {
   373→   373→        params.set('search', debouncedSearch.trim())
   374→   374→      }
   375→   375→      const res = await fetch(`/api/documents?${params.toString()}`, {
   376→   376→        headers: { 'x-user-id': user.id },
   377→   377→      })
   378→   378→      if (!res.ok) throw new Error('Error fetching documents')
   379→   379→      const data = await res.json()
   380→   380→      setDocuments(Array.isArray(data) ? data : data.documents || [])
   381→   381→    } catch {
   382→   382→      toast.error('Error al cargar documentos')
   383→   383→    } finally {
   384→   384→      setLoading(false)
   385→   385→    }
   386→   386→  }, [user, activeTab, debouncedSearch])
   387→   387→
   388→   388→  useEffect(() => {
   389→   389→    fetchDocuments()
   390→   390→  }, [fetchDocuments])
   391→   391→
   392→   392→  /* ---- Filtered documents for client-side (API may not support search) ---- */
   393→   393→  const filteredDocuments = documents.filter((doc) => {
   394→   394→    if (!debouncedSearch.trim()) return true
   395→   395→    const q = debouncedSearch.toLowerCase()
   396→   396→    return (
   397→   397→      doc.title.toLowerCase().includes(q) ||
   398→   398→      doc.template?.name?.toLowerCase().includes(q) ||
   399→   399→      doc.template?.category?.toLowerCase().includes(q)
   400→   400→    )
   401→   401→  })
   402→   402→
   403→   403→  /* ---- Action handlers ---- */
   404→   404→
   405→   405→  const handleEdit = (doc: Document) => {
   406→   406→    // Edit opens wizard with existing document
   407→   407→    startWizard(doc.template?.name || '', doc.id)
   408→   408→  }
   409→   409→
   410→   410→  const handleDuplicate = async (doc: Document) => {
   411→   411→    if (!user) return
   412→   412→    setActionLoading(doc.id)
   413→   413→    try {
   414→   414→      const res = await fetch('/api/documents', {
   415→   415→        method: 'POST',
   416→   416→        headers: {
   417→   417→          'Content-Type': 'application/json',
   418→   418→          'x-user-id': user.id,
   419→   419→        },
   420→   420→        body: JSON.stringify({
   421→   421→          duplicateFromId: doc.id,
   422→   422→        }),
   423→   423→      })
   424→   424→      if (!res.ok) throw new Error('Error duplicando documento')
   425→   425→      const newDoc = await res.json()
   426→   426→      toast.success('Documento duplicado', {
   427→   427→        description: `"${doc.title}" se copió como borrador.`,
   428→   428→      })
   429→   429→      // If we're on 'all' or 'draft', refetch to show new item
   430→   430→      if (activeTab === 'all' || activeTab === 'draft') {
   431→   431→        fetchDocuments()
   432→   432→      }
   433→   433→    } catch {
   434→   434→      toast.error('Error al duplicar', {
   435→   435→        description: 'No se pudo crear la copia del documento.',
   436→   436→      })
   437→   437→    } finally {
   438→   438→      setActionLoading(null)
   439→   439→    }
   440→   440→  }
   441→   441→
   442→   442→  const handleDownloadPdf = async (doc: Document) => {
   443→   443→    setActionLoading(doc.id)
   444→   444→    try {
   445→   445→      const res = await fetch(
   446→   446→        `/api/documents/export?id=${doc.id}&format=pdf`,
   447→   447→        {
   448→   448→          headers: { 'x-user-id': user!.id },
   449→   449→        }
   450→   450→      )
   451→   451→      if (!res.ok) throw new Error('Error generando PDF')
   452→   452→      const blob = await res.blob()
   453→   453→      const url = URL.createObjectURL(blob)
   454→   454→      const a = document.createElement('a')
   455→   455→      a.href = url
   456→   456→      a.download = `${doc.title}.pdf`
   457→   457→      document.body.appendChild(a)
   458→   458→      a.click()
   459→   459→      a.remove()
   460→   460→      URL.revokeObjectURL(url)
   461→   461→      toast.success('PDF descargado', {
   462→   462→        description: `"${doc.title}.pdf" se descargó correctamente.`,
   463→   463→      })
   464→   464→    } catch {
   465→   465→      toast.error('Error al descargar PDF')
   466→   466→    } finally {
   467→   467→      setActionLoading(null)
   468→   468→    }
   469→   469→  }
   470→   470→
   471→   471→  const handleDownloadDocx = async (doc: Document) => {
   472→   472→    setActionLoading(doc.id)
   473→   473→    try {
   474→   474→      const res = await fetch(
   475→   475→        `/api/documents/export?id=${doc.id}&format=docx`,
   476→   476→        {
   477→   477→          headers: { 'x-user-id': user!.id },
   478→   478→        }
   479→   479→      )
   480→   480→      if (!res.ok) throw new Error('Error generando DOCX')
   481→   481→      const blob = await res.blob()
   482→   482→      const url = URL.createObjectURL(blob)
   483→   483→      const a = document.createElement('a')
   484→   484→      a.href = url
   485→   485→      a.download = `${doc.title}.docx`
   486→   486→      document.body.appendChild(a)
   487→   487→      a.click()
   488→   488→      a.remove()
   489→   489→      URL.revokeObjectURL(url)
   490→   490→      toast.success('DOCX descargado', {
   491→   491→        description: `"${doc.title}.docx" se descargó correctamente.`,
   492→   492→      })
   493→   493→    } catch {
   494→   494→      toast.error('Error al descargar DOCX')
   495→   495→    } finally {
   496→   496→      setActionLoading(null)
   497→   497→    }
   498→   498→  }
   499→   499→
   500→   500→  const handleArchive = async (doc: Document) => {
   501→   501→    if (!user) return
   502→   502→    const newStatus: DocumentStatus =
   503→   503→      doc.status === 'archived' ? 'draft' : 'archived'
   504→   504→    setActionLoading(doc.id)
   505→   505→    try {
   506→   506→      const res = await fetch('/api/documents', {
   507→   507→        method: 'PUT',
   508→   508→        headers: {
   509→   509→          'Content-Type': 'application/json',
   510→   510→          'x-user-id': user.id,
   511→   511→        },
   512→   512→        body: JSON.stringify({
   513→   513→          id: doc.id,
   514→   514→          status: newStatus,
   515→   515→        }),
   516→   516→      })
