    79→    79→const CONTACT_TYPE_MAP: Record<string, string[]> = {
    80→    80→  arrendador: ['arrendador', 'propietario'],
    81→    81→  arrendatario: ['arrendatario', 'inquilino'],
    82→    82→  fiador: ['fiador', 'codeudor'],
    83→    83→  comprador: ['comprador', 'adquirente'],
    84→    84→  vendedor: ['vendedor', 'enajenante'],
    85→    85→  empleador: ['empleador', 'patrono', 'empresa'],
    86→    86→  trabajador: ['trabajador', 'empleado'],
    87→    87→  demandante: ['demandante', 'actor'],
    88→    88→  demandado: ['demandado'],
    89→    89→  deudor: ['deudor'],
    90→    90→  acreedor: ['acreedor'],
    91→    91→}
    92→    92→
    93→    93→function detectContactType(fieldKey: string): string | null {
    94→    94→  const lower = fieldKey.toLowerCase()
    95→    95→  for (const [contactType, prefixes] of Object.entries(CONTACT_TYPE_MAP)) {
    96→    96→    for (const prefix of prefixes) {
    97→    97→      if (lower.includes(prefix)) return contactType
    98→    98→    }
    99→    99→  }
   100→   100→  return null
   101→   101→}

   869→   869→interface FieldRendererProps {
   870→   870→  field: WizardField

   873→   873→  onFocus: () => void

   878→   878→
   879→   879→function FieldRenderer({
   880→   880→  field,
   881→   881→  value,
   882→   882→  onChange,
   883→   883→  onFocus,
   884→   884→  contacts,
   885→   885→  onApplyContact,
   886→   886→  onDismissContacts,
   887→   887→}: FieldRendererProps) {
   888→   888→  const fieldRef = useRef<HTMLDivElement>(null)
   889→   889→
   890→   890→  // Close contacts dropdown on outside click
   891→   891→  useEffect(() => {
   892→   892→    if (contacts.length === 0) return
   893→   893→    function handleClick(e: MouseEvent) {
   894→   894→      if (fieldRef.current && !fieldRef.current.contains(e.target as Node)) {
   895→   895→        onDismissContacts()
   896→   896→      }
   897→   897→    }
   898→   898→    document.addEventListener('mousedown', handleClick)
   899→   899→    return () => document.removeEventListener('mousedown', handleClick)
   900→   900→  }, [contacts.length, onDismissContacts])
   901→   901→
   902→   902→  const labelElement = (
   903→   903→    <div className="mb-1.5 flex items-center gap-1.5">
   904→   904→      <Label className="text-xs font-medium text-[#CBD5E1]">{field.label}</Label>
   905→   905→      {field.tooltip && (
   906→   906→        <Tooltip>
   907→   907→          <TooltipTrigger asChild>
   908→   908→            <button type="button" className="text-[#475569] hover:text-[#C9A94E] transition-colors">
   909→   909→              <Info className="h-3.5 w-3.5" />
   910→   910→            </button>

  1056→  1056→function formatPreviewContent(content: string): string {
