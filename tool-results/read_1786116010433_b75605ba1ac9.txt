    96→function formatPrice(price: number | null): string {
    97→  if (price === null || price === 0) return 'GRATUITO'
    98→  return `COP $${price.toLocaleString('es-CO')}`

   351→function TemplateDetailDialog({
   352→  template,
   353→  open,
   354→  onOpenChange,
   355→}: {
   356→  template: Template | null
   357→  open: boolean
   358→  onOpenChange: (open: boolean) => void
   359→}) {
   360→  const { user, startWizard, startVisitorWizard } = useAppStore()
   361→
   362→  if (!template) return null
   363→
   364→  const isFree = template.price === null || template.price === 0
   365→  const handleStart = () => {
   366→    onOpenChange(false)
   367→    if (user) {
   368→      startWizard(template.id)
   369→    } else {
   370→      startVisitorWizard(template.id)
   371→    }
   372→  }
   373→
   374→  return (
   375→    <Dialog open={open} onOpenChange={onOpenChange}>
   376→      <DialogContent className="max-h-[85vh] overflow-y-auto border-white/10 bg-[#0F1D32] sm:max-w-lg">
   377→        <DialogHeader>
   378→          <div className="flex items-center gap-3">
   379→            <div className="flex h-12 w-12 shrink-0 items-center justify-center rounded-xl bg-[#C9A94E]/10 text-2xl">
   380→              {getLegalAreaEmoji(template.legalArea)}
   381→            </div>
   382→            <div>
   383→              <DialogTitle className="text-base font-semibold text-white">
   384→                {template.name}
   385→              </DialogTitle>
   386→              <DialogDescription className="text-xs text-white/40">
   387→                {template.category} · {template.legalArea}
   388→              </DialogDescription>
   389→            </div>
   390→          </div>
   391→        </DialogHeader>
   392→
   393→        <div className="space-y-5 pt-2">
   394→          {/* Description */}
   395→          <div>
   396→            <h4 className="mb-1.5 text-xs font-semibold uppercase tracking-wider text-white/30">
   397→              Descripción
   398→            </h4>
   399→            <p className="text-sm leading-relaxed text-white/60">
   400→              {template.description}
   401→            </p>
   402→          </div>
   403→
   404→          {/* Badges */}
   405→          <div className="flex flex-wrap gap-2">
   406→            <Badge
   407→              variant="outline"
   408→              className={
   409→                isFree
   410→                  ? 'border-[#C9A94E]/30 bg-[#C9A94E]/10 text-[#C9A94E] text-[10px] font-semibold'
   411→                  : 'border-green-500/30 bg-green-500/10 text-green-400 text-[10px] font-semibold'
   412→              }
   413→            >
   414→              {isFree ? '✦ GRATUITO' : formatPrice(template.price)}
   415→            </Badge>
   416→            <Badge
   417→              variant="outline"
   418→              className="border-white/10 bg-white/5 text-white/50 text-[10px]"
   419→            >
   420→              {template.audience === 'Profesionales' ? '👨‍💼' : '👤'} {template.audience}
   421→            </Badge>
   422→            <Badge
   423→              variant="outline"
   424→              className="border-white/10 bg-white/5 text-white/50 text-[10px]"
   425→            >
   426→              {template._count?.documents ?? 0} documentos generados
   427→            </Badge>
   428→          </div>
   429→
   430→          {/* Stats */}
   431→          <div className="grid grid-cols-3 gap-3">
   432→            <div className="rounded-lg border border-white/5 bg-white/[0.02] p-3 text-center">
   433→              <Clock className="mx-auto mb-1 h-4 w-4 text-[#C9A94E]" />

   481→
   482→          {/* CTA */}
   483→          <Button
   484→            className="w-full bg-[#C9A94E] text-[#0A1628] font-semibold text-sm transition-all hover:bg-[#D4B965] hover:shadow-lg hover:shadow-[#C9A94E]/20"

   531→/* -------------------------------------------------------------------------- */
   532→
   533→export default function CatalogPage() {
   534→  const { user, isVisitor } = useAppStore()
   535→
   536→  /* ---- State ---- */
   537→  const [templates, setTemplates] = useState<Template[]>([])
   538→  const [loading, setLoading] = useState(true)
   539→  const [viewMode, setViewMode] = useState<'grid' | 'list'>('grid')
   540→
   541→  // Filters
   542→  const [searchInput, setSearchInput] = useState('')
   543→  const [debouncedSearch, setDebouncedSearch] = useState('')
   544→  const [category, setCategory] = useState('all')
   545→  const [legalArea, setLegalArea] = useState('all')
   546→  const [audience, setAudience] = useState('all')
   547→  const [priceFilter, setPriceFilter] = useState('all')
   548→
   549→  // Dialog
   550→  const [selectedTemplate, setSelectedTemplate] = useState<Template | null>(null)
   551→  const [dialogOpen, setDialogOpen] = useState(false)
   552→
   553→  // Debounce timer ref
   554→  const debounceTimerRef = useRef<NodeJS.Timeout | null>(null)
   555→
   556→  /* ---- Debounced search ---- */
   557→  const updateDebouncedSearch = useCallback((value: string) => {
   558→    if (debounceTimerRef.current) {
   559→      clearTimeout(debounceTimerRef.current)
   560→    }
   561→    debounceTimerRef.current = setTimeout(() => {
   562→      setDebouncedSearch(value)
   563→    }, 300)
   564→  }, [])
   565→
   566→  // Sync search input to debounced value
   567→  useEffect(() => {
   568→    updateDebouncedSearch(searchInput)
   569→    return () => {
   570→      if (debounceTimerRef.current) {
   571→        clearTimeout(debounceTimerRef.current)
   572→      }
   573→    }
   574→  }, [searchInput, updateDebouncedSearch])
   575→
   576→  /* ---- Fetch templates ---- */
   577→  const fetchTemplates = useCallback(async () => {
   578→    setLoading(true)
   579→    try {
   580→      const params = new URLSearchParams()
   581→
   582→      if (category !== 'all') params.set('category', category)
   583→      if (legalArea !== 'all') params.set('legalArea', legalArea)
   584→      if (audience !== 'all') params.set('audience', audience)
   585→      if (priceFilter === 'free') params.set('price', 'free')
   586→      if (priceFilter === 'paid') params.set('price', 'paid')
   587→      if (debouncedSearch.trim()) params.set('search', debouncedSearch.trim())
   588→
   589→      const query = params.toString() ? `?${params.toString()}` : ''
   590→      const headers: HeadersInit = {}
   591→      if (user) headers['x-user-id'] = user.id
   592→      const res = await fetch(`/api/templates${query}`, { headers })
   593→
   594→      if (res.ok) {
   595→        const data = await res.json()
   596→        setTemplates(Array.isArray(data) ? data : data.templates || [])
   597→      }
   598→    } catch {
   599→      // keep empty on error
   600→    } finally {
   601→      setLoading(false)
   602→    }
   603→  }, [user, category, legalArea, audience, priceFilter, debouncedSearch])
   604→
   605→  useEffect(() => {
   606→    fetchTemplates()
   607→  }, [fetchTemplates])
   608→
   609→  /* ---- Handlers ---- */
   610→  const handleSelectTemplate = useCallback((template: Template) => {
   611→    setSelectedTemplate(template)
   612→    setDialogOpen(true)
   613→  }, [])
   614→
   615→  const handleDialogClose = useCallback((open: boolean) => {
   616→    setDialogOpen(open)
   617→    if (!open) {
   618→      // Keep template reference for animation but can clear if desired
   619→    }
   620→  }, [])
   621→
   622→  const hasActiveFilters =
   623→    searchInput.trim() !== '' ||
   624→    category !== 'all' ||
   625→    legalArea !== 'all' ||
   626→    audience !== 'all' ||
   627→    priceFilter !== 'all'
   628→
   629→  /* ---- Render ---- */
   630→  return (
   631→    <main className="min-h-screen bg-[#0A1628]">
   632→      <div className="mx-auto max-w-7xl space-y-6 px-4 py-8 sm:px-6 lg:px-8">
   633→        {/* ---------- Header ---------- */}
   634→        <div>
   635→          <h1 className="text-2xl font-bold tracking-tight text-white sm:text-3xl">
   636→            Catálogo de Plantillas
   637→          </h1>
   638→          <p className="mt-1 text-sm text-white/50">
   639→            Explora y genera documentos legales colombianos con plantillas profesionales.
   640→          </p>
   641→        </div>
   642→
   643→        {/* ---------- Search bar ---------- */}
   644→        <div className="relative">
   645→          <Search className="absolute top-1/2 left-3 h-4 w-4 -translate-y-1/2 text-white/30" />
   646→          <Input
   647→            placeholder="Buscar plantillas por nombre, tipo o palabra clave..."
   648→            value={searchInput}
   649→            onChange={(e) => setSearchInput(e.target.value)}
   650→            className="h-11 w-full border-white/10 bg-[#0F1D32] pl-10 pr-4 text-sm text-white placeholder:text-white/30 focus-visible:border-[#C9A94E]/40 focus-visible:ring-[#C9A94E]/20"
   651→          />
   652→        </div>
   653→
   654→        {/* ---------- Filters row ---------- */}
   655→        <div className="flex flex-col gap-3 sm:flex-row sm:flex-wrap sm:items-center sm:gap-3">
   656→          {/* Category */}
   657→          <Select value={category} onValueChange={setCategory}>
   658→            <SelectTrigger
   659→              className="w-full border-white/10 bg-[#0F1D32] text-white/70 text-xs sm:w-[155px]"
   660→              size="sm"
   661→            >
   662→              <SelectValue placeholder="Categoría" />
   663→            </SelectTrigger>
   664→            <SelectContent className="border-white/10 bg-[#0F1D32]">
   665→              <SelectItem value="all" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   666→                Todas
   667→              </SelectItem>
   668→              <SelectItem value="Contratos" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   669→                Contratos
   670→              </SelectItem>
   671→              <SelectItem value="Actas" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   672→                Actas
   673→              </SelectItem>
   674→              <SelectItem value="Derechos de Petición" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   675→                Derechos de Petición
   676→              </SelectItem>
   677→            </SelectContent>
   678→          </Select>
   679→
   680→          {/* Legal Area */}
   681→          <Select value={legalArea} onValueChange={setLegalArea}>
   682→            <SelectTrigger
   683→              className="w-full border-white/10 bg-[#0F1D32] text-white/70 text-xs sm:w-[165px]"
   684→              size="sm"
   685→            >
   686→              <SelectValue placeholder="Área del Derecho" />
   687→            </SelectTrigger>
   688→            <SelectContent className="border-white/10 bg-[#0F1D32]">
   689→              <SelectItem value="all" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   690→                Todos
   691→              </SelectItem>
   692→              <SelectItem value="Civil" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   693→                Civil
   694→              </SelectItem>
   695→              <SelectItem value="Mercantil" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   696→                Mercantil
   697→              </SelectItem>
   698→              <SelectItem value="Laboral" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   699→                Laboral
   700→              </SelectItem>
   701→              <SelectItem value="Constitucional" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   702→                Constitucional
   703→              </SelectItem>
   704→            </SelectContent>
   705→          </Select>
   706→
   707→          {/* Audience */}
   708→          <Select value={audience} onValueChange={setAudience}>
   709→            <SelectTrigger
   710→              className="w-full border-white/10 bg-[#0F1D32] text-white/70 text-xs sm:w-[155px]"
   711→              size="sm"
   712→            >
   713→              <SelectValue placeholder="Audiencia" />
   714→            </SelectTrigger>
   715→            <SelectContent className="border-white/10 bg-[#0F1D32]">
   716→              <SelectItem value="all" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   717→                Todos
   718→              </SelectItem>
   719→              <SelectItem value="Profesionales" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   720→                Profesionales
   721→              </SelectItem>
   722→              <SelectItem value="Particulares" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   723→                Particulares
   724→              </SelectItem>
   725→            </SelectContent>
   726→          </Select>
   727→
   728→          {/* Price */}
   729→          <Select value={priceFilter} onValueChange={setPriceFilter}>
   730→            <SelectTrigger
   731→              className="w-full border-white/10 bg-[#0F1D32] text-white/70 text-xs sm:w-[135px]"
   732→              size="sm"
   733→            >
   734→              <SelectValue placeholder="Precio" />
   735→            </SelectTrigger>
   736→            <SelectContent className="border-white/10 bg-[#0F1D32]">
   737→              <SelectItem value="all" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   738→                Todos
   739→              </SelectItem>
   740→              <SelectItem value="free" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   741→                Gratuito
   742→              </SelectItem>
   743→              <SelectItem value="paid" className="text-white/70 focus:bg-[#C9A94E]/10 focus:text-[#C9A94E]">
   744→                De Pago
   745→              </SelectItem>
   746→            </SelectContent>
   747→          </Select>
   748→
   749→          {/* Spacer */}
   750→          <div className="flex-1" />
   751→
   752→          {/* View toggle */}
   753→          <ToggleGroup
   754→            type="single"
   755→            value={viewMode}
   756→            onValueChange={(v) => {
   757→              if (v) setViewMode(v as 'grid' | 'list')
   758→            }}
   759→            className="border-white/10 bg-[#0F1D32]"
   760→          >
   761→            <ToggleGroupItem
   762→              value="grid"
   763→              aria-label="Vista de cuadrícula"
   764→              className="data-[state=on]:bg-[#C9A94E]/15 data-[state=on]:text-[#C9A94E] text-white/40 hover:bg-white/5 hover:text-white/60 h-8 w-8 rounded-l-md"
   765→            >
   766→              <LayoutGrid className="h-4 w-4" />
   767→            </ToggleGroupItem>
   768→            <ToggleGroupItem
   769→              value="list"
   770→              aria-label="Vista de lista"
   771→              className="data-[state=on]:bg-[#C9A94E]/15 data-[state=on]:text-[#C9A94E] text-white/40 hover:bg-white/5 hover:text-white/60 h-8 w-8 rounded-r-md"
   772→            >
   773→              <List className="h-4 w-4" />
   774→            </ToggleGroupItem>
   775→          </ToggleGroup>
   776→        </div>
   777→
   778→        {/* ---------- Results count ---------- */}
   779→        <div className="flex items-center justify-between">
   780→          <p className="text-xs text-white/40">
   781→            {loading
   782→              ? 'Cargando...'
   783→              : `${templates.length} plantilla${templates.length !== 1 ? 's' : ''} encontrada${templates.length !== 1 ? 's' : ''}`}
   784→          </p>
   785→          {hasActiveFilters && (
   786→            <button
   787→              onClick={() => {
   788→                setSearchInput('')
   789→                setCategory('all')
   790→                setLegalArea('all')
   791→                setAudience('all')
   792→                setPriceFilter('all')
   793→              }}
   794→              className="text-[11px] font-medium text-[#C9A94E]/70 transition-colors hover:text-[#C9A94E]"
   795→            >
   796→              Limpiar filtros
   797→            </button>
   798→          )}
   799→        </div>
   800→
   801→        {/* ---------- Content ---------- */}
   802→        {loading ? (
   803→          viewMode === 'grid' ? (
   804→            <div className="grid grid-cols-1 gap-4 md:grid-cols-2 xl:grid-cols-3">
   805→              {Array.from({ length: 6 }).map((_, i) => (
   806→                <SkeletonCard key={i} />
   807→              ))}
   808→            </div>
   809→          ) : (
   810→            <div className="flex flex-col gap-3">
   811→              {Array.from({ length: 6 }).map((_, i) => (
   812→                <SkeletonRow key={i} />
   813→              ))}
   814→            </div>
   815→          )
   816→        ) : templates.length === 0 ? (
   817→          <EmptyState hasFilters={hasActiveFilters} />
   818→        ) : viewMode === 'grid' ? (
   819→          <div className="grid grid-cols-1 gap-4 md:grid-cols-2 xl:grid-cols-3">
   820→            {templates.map((t) => (
   821→              <TemplateCard key={t.id} template={t} onSelect={handleSelectTemplate} />
   822→            ))}
   823→          </div>
   824→        ) : (
   825→          <div className="flex flex-col gap-3">
   826→            {templates.map((t) => (
   827→              <TemplateRow key={t.id} template={t} onSelect={handleSelectTemplate} />
   828→            ))}
   829→          </div>
   830→        )}
   831→
   832→        {/* ---------- Bottom CTA ---------- */}
   833→        <div className="flex items-center justify-center pt-4">
   834→          <button
   835→            onClick={() => {
   836→              // Navigate to a request page or open a dialog for custom document
   837→              // For now, we'll set page to documents where a request flow might exist
   838→            }}
   839→            className="group flex items-center gap-3 rounded-xl border border-[#C9A94E]/15 bg-[#0F1D32]/60 px-6 py-4 text-left backdrop-blur-sm transition-all hover:border-[#C9A94E]/30 hover:bg-[#C9A94E]/5 hover:shadow-lg hover:shadow-[#C9A94E]/5"
   840→          >
   841→            <div className="flex h-11 w-11 shrink-0 items-center justify-center rounded-lg bg-[#C9A94E]/15 text-[#C9A94E] transition-colors group-hover:bg-[#C9A94E]/25">
   842→              <Plus className="h-5 w-5" />
   843→            </div>
   844→            <div className="min-w-0 flex-1">
   845→              <p className="text-sm font-semibold text-white/80 group-hover:text-white">
   846→                ¿No encuentras lo que buscas?
   847→              </p>
   848→              <p className="text-xs text-white/40">
   849→                Solicita un documento personalizado y nuestro equipo lo preparará para ti.
   850→              </p>
   851→            </div>
   852→            <ArrowRight className="h-4 w-4 shrink-0 text-white/20 transition-transform group-hover:translate-x-0.5 group-hover:text-[#C9A94E]" />
   853→          </button>
   854→        </div>
   855→      </div>
   856→
   857→      {/* ---------- Detail Dialog ---------- */}
   858→      <TemplateDetailDialog
   859→        template={selectedTemplate}
   860→        open={dialogOpen}
   861→        onOpenChange={handleDialogClose}
   862→      />
   863→    </main>
   864→  )
   865→}
