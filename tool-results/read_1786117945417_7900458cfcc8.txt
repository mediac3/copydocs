    75→/* -------------------------------------------------------------------------- */
    76→/*  Contact type mapping - field name prefix → contact type                   */
    77→/* -------------------------------------------------------------------------- */
    78→
    79→const CONTACT_TYPE_MAP: Record<string, string[]> = {
    80→  arrendador: ['arrendador', 'propietario'],
    81→  arrendatario: ['arrendatario', 'inquilino'],
    82→  fiador: ['fiador', 'codeudor'],
    83→  comprador: ['comprador', 'adquirente'],
    84→  vendedor: ['vendedor', 'enajenante'],
    85→  empleador: ['empleador', 'patrono', 'empresa'],
    86→  trabajador: ['trabajador', 'empleado'],
    87→  demandante: ['demandante', 'actor'],
    88→  demandado: ['demandado'],
    89→  deudor: ['deudor'],
    90→  acreedor: ['acreedor'],
    91→}
    92→
    93→function detectContactType(fieldKey: string): string | null {
    94→  const lower = fieldKey.toLowerCase()
    95→  for (const [contactType, prefixes] of Object.entries(CONTACT_TYPE_MAP)) {
    96→    for (const prefix of prefixes) {
    97→      if (lower.includes(prefix)) return contactType
    98→    }
    99→  }
   100→  return null
   101→}

  1044→            </Label>
  1045→          </div>

  1047→      )}
  1048→    </div>
  1049→  )

  1055→
  1056→function formatPreviewContent(content: string): string {
  1057→  // Replace newlines with paragraph breaks
  1058→  let html = content
  1059→    .split('\n')
  1060→    .filter((line) => line.trim() !== '')
  1061→    .map((line) => {
  1062→      // Highlight unresolved {{variables}} in gold
  1063→      const highlighted = line.replace(
  1064→        /\{\{([^}]+)\}\}/g,
  1065→        '<span class="inline-block rounded bg-[#C9A94E]/10 px-1.5 py-0.5 text-[#C9A94E] font-semibold text-xs mx-0.5">{{$1}}</span>'
  1066→      )
  1067→      return `<p>${highlighted}</p>`
  1068→    })
  1069→    .join('')
  1070→  return html
  1071→}
