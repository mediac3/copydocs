   161→function useDebounce<T>(value: T, delayMs: number): T {
   162→  const [debounced, setDebounced] = useState(value)
   163→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   164→
   165→  useEffect(() => {
   166→    if (timerRef.current) clearTimeout(timerRef.current)
   167→    timerRef.current = setTimeout(() => setDebounced(value), delayMs)
   168→    return () => {
   169→      if (timerRef.current) clearTimeout(timerRef.current)
   170→    }
   171→  }, [value, delayMs])
   172→
   173→  return debounced
   174→}

   586→          </AlertDialogDescription>
   587→        </AlertDialogHeader>
   588→        <AlertDialogFooter className="gap-2 sm:gap-0">
   589→          <AlertDialogCancel
   590→            disabled={loading}
   591→            className="border-white/10 bg-white/5 text-white/70 hover:bg-white/10 hover:text-white"
   592→          >
   593→            Cancelar
   594→          </AlertDialogCancel>
   595→          <AlertDialogAction
   596→            onClick={(e) => {
   597→              e.preventDefault()
   598→              onConfirm()
   599→            }}
   600→            disabled={loading}
   601→            className="bg-red-600 text-white hover:bg-red-700 focus:ring-red-500"
   602→          >
   603→            {loading ? 'Eliminando…' : 'Eliminar'}
   604→          </AlertDialogAction>
   605→        </AlertDialogFooter>
   606→      </AlertDialogContent>
   607→    </AlertDialog>
   608→  )
   609→}

   613→/* -------------------------------------------------------------------------- */
   614→
   615→export default function ContactsPage() {
   616→  const { user } = useAppStore()
   617→
   618→  /* ---- State ---- */
   619→  const [activeTab, setActiveTab] = useState<TabKey>('all')
   620→  const [searchQuery, setSearchQuery] = useState('')
   621→  const debouncedSearch = useDebounce(searchQuery, 300)
   622→  const [contacts, setContacts] = useState<Contact[]>([])
   623→  const [loading, setLoading] = useState(true)
   624→
   625→  /* Form dialog */
   626→  const [formDialogOpen, setFormDialogOpen] = useState(false)
   627→  const [editingContact, setEditingContact] = useState<Contact | null>(null)
   628→  const [saving, setSaving] = useState(false)
   629→
   630→  /* Delete dialog */
   631→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false)
   632→  const [contactToDelete, setContactToDelete] = useState<Contact | null>(null)
   633→  const [deleting, setDeleting] = useState(false)
   634→
   635→  /* ---- Fetch contacts ---- */
   636→  const fetchContacts = useCallback(async () => {
   637→    if (!user) return
   638→    setLoading(true)
   639→    try {
   640→      const params = new URLSearchParams()
   641→      const tabConfig = TAB_CONFIG[activeTab]
   642→      if (tabConfig.apiValue) {
   643→        params.set('type', tabConfig.apiValue)
   644→      }
   645→      if (debouncedSearch.trim()) {
   646→        params.set('search', debouncedSearch.trim())
   647→      }
   648→      const res = await fetch(`/api/contacts?${params.toString()}`, {
   649→        headers: { 'x-user-id': user.id },
   650→      })
   651→      if (!res.ok) throw new Error('Error fetching contacts')
   652→      const data = await res.json()
   653→      setContacts(Array.isArray(data) ? data : [])
   654→    } catch {
   655→      toast.error('Error al cargar contactos')
   656→    } finally {
   657→      setLoading(false)
   658→    }
   659→  }, [user, activeTab, debouncedSearch])
   660→
   661→  useEffect(() => {
   662→    fetchContacts()
   663→  }, [fetchContacts])
   664→
   665→  /* ---- Client-side search fallback ---- */
   666→  const filteredContacts = contacts.filter((c) => {
   667→    if (!debouncedSearch.trim()) return true
   668→    const q = debouncedSearch.toLowerCase()
   669→    return (
   670→      c.name.toLowerCase().includes(q) ||
   671→      c.contactType.toLowerCase().includes(q) ||
   672→      c.documentNumber.toLowerCase().includes(q) ||
   673→      c.companyName.toLowerCase().includes(q) ||
   674→      c.city.toLowerCase().includes(q) ||
   675→      c.email.toLowerCase().includes(q)
   676→    )
   677→  })
   678→
   679→  /* ---- Create / Update ---- */
   680→  const handleSave = async (formData: ContactFormData) => {
   681→    if (!user) return
   682→    setSaving(true)
   683→    try {
   684→      const isEdit = !!formData.id
   685→      const method = isEdit ? 'PUT' : 'POST'
   686→      const res = await fetch('/api/contacts', {
   687→        method,
   688→        headers: {
   689→          'Content-Type': 'application/json',
   690→          'x-user-id': user.id,
   691→        },
   692→        body: JSON.stringify(formData),
   693→      })
   694→      if (!res.ok) throw new Error('Error saving contact')
   695→      toast.success(
   696→        isEdit
   697→          ? 'Contacto actualizado correctamente'
   698→          : 'Contacto creado correctamente'
   699→      )
   700→      setFormDialogOpen(false)
   701→      setEditingContact(null)
   702→      fetchContacts()
   703→    } catch {
   704→      toast.error('Error al guardar el contacto')
   705→    } finally {
   706→      setSaving(false)
   707→    }
   708→  }
   709→
   710→  /* ---- Delete ---- */
   711→  const handleDelete = async () => {
   712→    if (!user || !contactToDelete) return
   713→    setDeleting(true)
   714→    try {
   715→      const res = await fetch(
   716→        `/api/contacts?id=${contactToDelete.id}`,
   717→        {
   718→          method: 'DELETE',
   719→          headers: { 'x-user-id': user.id },
   720→        }
   721→      )
   722→      if (!res.ok) throw new Error('Error deleting contact')
   723→      toast.success('Contacto eliminado correctamente')
   724→      setDeleteDialogOpen(false)
   725→      setContactToDelete(null)
   726→      fetchContacts()
   727→    } catch {
   728→      toast.error('Error al eliminar el contacto')
   729→    } finally {
   730→      setDeleting(false)
   731→    }
   732→  }
   733→
   734→  /* ---- Handlers ---- */
   735→  const openNewContact = () => {
   736→    setEditingContact(null)
   737→    setFormDialogOpen(true)
   738→  }
   739→
   740→  const openEditContact = (contact: Contact) => {
   741→    setEditingContact(contact)
   742→    setFormDialogOpen(true)
   743→  }
   744→
   745→  const openDeleteContact = (contact: Contact) => {
   746→    setContactToDelete(contact)
   747→    setDeleteDialogOpen(true)
   748→  }
   749→
   750→  /* ---- Render ---- */
   751→  return (
   752→    <div className="flex h-full flex-col overflow-hidden bg-[#0A1628]">
   753→      {/* Header */}
   754→      <div className="flex shrink-0 items-center justify-between border-b border-white/[0.06] px-6 py-4">
   755→        <div>
   756→          <h1 className="text-lg font-semibold text-white">
   757→            Mis Datos
   758→          </h1>
   759→          <p className="mt-0.5 text-xs text-white/40">
   760→            Gestiona tus contactos y partes para documentos legales
   761→          </p>
   762→        </div>
   763→        <Button
   764→          onClick={openNewContact}
   765→          className="bg-[#C9A94E] text-[#0A1628] font-semibold hover:bg-[#B8983E] focus:ring-[#C9A94E]/50"
   766→        >
   767→          <Plus className="mr-1.5 h-4 w-4" />
   768→          Nuevo contacto
   769→        </Button>
   770→      </div>
   771→
   772→      {/* Toolbar: search + tabs */}
   773→      <div className="shrink-0 space-y-3 border-b border-white/[0.06] px-6 py-4">
   774→        {/* Search bar */}
   775→        <div className="relative">
   776→          <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-white/30" />
   777→          <Input
   778→            value={searchQuery}
   779→            onChange={(e) => setSearchQuery(e.target.value)}
   780→            placeholder="Buscar por nombre, documento, ciudad, empresa..."
   781→            className="border-white/10 bg-white/[0.03] pl-9 text-sm text-white placeholder:text-white/25 focus-visible:border-[#C9A94E]/40 focus-visible:ring-[#C9A94E]/20"
   782→          />
   783→        </div>
   784→
   785→        {/* Type filter tabs */}
   786→        <Tabs
   787→          value={activeTab}
   788→          onValueChange={(v) => setActiveTab(v as TabKey)}
   789→        >
   790→          <TabsList className="h-9 bg-white/[0.04] p-0.5">
   791→            {(Object.keys(TAB_CONFIG) as TabKey[]).map((key) => (
   792→              <TabsTrigger
   793→                key={key}
   794→                value={key}
   795→                className="h-7 rounded-md px-3 text-xs data-[state=active]:bg-[#C9A94E]/15 data-[state=active]:text-[#C9A94E] data-[state=active]:shadow-none text-white/40 hover:text-white/60"
   796→              >
   797→                {TAB_CONFIG[key].label}
   798→              </TabsTrigger>
   799→            ))}
   800→          </TabsList>
   801→        </Tabs>
   802→      </div>
   803→
   804→      {/* Contact grid */}
   805→      <div className="flex-1 overflow-y-auto px-6 py-5">
   806→        {loading ? (
   807→          <div className="flex items-center justify-center py-20">
   808→            <div className="h-8 w-8 animate-spin rounded-full border-2 border-[#C9A94E]/30 border-t-[#C9A94E]" />
   809→          </div>
   810→        ) : filteredContacts.length === 0 ? (
   811→          <EmptyState />
   812→        ) : (
   813→          <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-3">
   814→            {filteredContacts.map((contact) => (
   815→              <ContactCard
   816→                key={contact.id}
   817→                contact={contact}
   818→                onEdit={openEditContact}
   819→                onDelete={openDeleteContact}
   820→              />
   821→            ))}
   822→          </div>
   823→        )}
   824→      </div>
   825→
   826→      {/* Form Dialog */}
   827→      <ContactFormDialog
   828→        open={formDialogOpen}
   829→        onOpenChange={setFormDialogOpen}
   830→        contact={editingContact}
   831→        onSave={handleSave}
   832→        saving={saving}
   833→      />
   834→
   835→      {/* Delete Dialog */}
   836→      <DeleteDialog
   837→        open={deleteDialogOpen}
   838→        onOpenChange={setDeleteDialogOpen}
   839→        contactName={contactToDelete?.name ?? ''}
   840→        onConfirm={handleDelete}
   841→        loading={deleting}
   842→      />
   843→    </div>
   844→  )
   845→}
