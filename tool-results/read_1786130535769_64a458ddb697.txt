   242→   242→      {payload.map((p, i) => (
   243→   243→        <div key={i} className="flex items-center gap-2 text-xs">
   244→   244→          <span className="inline-block h-2 w-2 rounded-full" style={{ backgroundColor: p.color }} />
   245→   245→          <span className="text-white/60">{p.name}:</span>
   246→   246→          <span className="font-medium text-white">{p.value}</span>
   247→   247→        </div>
