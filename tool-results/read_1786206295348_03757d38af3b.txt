  1122→
  1123→function formatPreviewContent(content: string): string {
  1124→  // Replace newlines with paragraph breaks
  1125→  let html = content
  1126→    .split('\n')
  1127→    .filter((line) => line.trim() !== '')
  1128→    .map((line) => {
  1129→      // Highlight unresolved {{variables}} in gold
  1130→      const highlighted = line.replace(
  1131→        /\{\{([^}]+)\}\}/g,
  1132→        '<span class="inline-block rounded bg-[#C9A94E]/10 px-1.5 py-0.5 text-[#C9A94E] font-semibold text-xs mx-0.5">{{$1}}</span>'
  1133→      )
  1134→      return `<p>${highlighted}</p>`
  1135→    })
  1136→    .join('')
  1137→  return html
  1138→}
