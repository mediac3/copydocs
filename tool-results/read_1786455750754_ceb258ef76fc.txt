   553→  const [audience, setAudience] = useState('all')
   554→  const [priceFilter, setPriceFilter] = useState('all')
   555→
   556→  // Dialog
   557→  const [selectedTemplate, setSelectedTemplate] = useState<Template | null>(null)
   558→  const [dialogOpen, setDialogOpen] = useState(false)
   559→
   560→  // Debounce timer ref
   561→  const debounceTimerRef = useRef<NodeJS.Timeout | null>(null)
   562→
   563→  /* ---- Debounced search ---- */
   564→  const updateDebouncedSearch = useCallback((value: string) => {
   565→    if (debounceTimerRef.current) {
   566→      clearTimeout(debounceTimerRef.current)
   567→    }
   568→    debounceTimerRef.current = setTimeout(() => {
   569→      setDebouncedSearch(value)
   570→    }, 300)
   571→  }, [])
   572→
   573→  // Sync search input to debounced value
   574→  useEffect(() => {
   575→    updateDebouncedSearch(searchInput)
   576→    return () => {
   577→      if (debounceTimerRef.current) {
   578→        clearTimeout(debounceTimerRef.current)
   579→      }
   580→    }
   581→  }, [searchInput, updateDebouncedSearch])
   582→
   583→  /* ---- Fetch templates ---- */
   584→  const fetchTemplates = useCallback(async () => {
   585→    setLoading(true)
   586→    try {
   587→      const params = new URLSearchParams()
   588→
   589→      if (category !== 'all') params.set('category', category)
   590→      if (legalArea !== 'all') params.set('legalArea', legalArea)
   591→      if (audience !== 'all') params.set('audience', audience)
   592→      if (priceFilter === 'free') params.set('price', 'free')
   593→      if (priceFilter === 'paid') params.set('price', 'paid')
   594→      if (debouncedSearch.trim()) params.set('search', debouncedSearch.trim())
   595→
   596→      const query = params.toString() ? `?${params.toString()}` : ''
   597→      const headers: HeadersInit = {}
   598→      if (user) headers['x-user-id'] = user.id
   599→      const res = await fetch(`/api/templates${query}`, { headers })
   600→
   601→      if (res.ok) {
   602→        const data = await res.json()
   603→        setTemplates(Array.isArray(data) ? data : data.templates || [])
   604→      }
   605→    } catch {
   606→      // keep empty on error
   607→    } finally {
   608→      setLoading(false)
   609→    }
   610→  }, [user, category, legalArea, audience, priceFilter, debouncedSearch])
   611→
   612→  useEffect(() => {
   613→    fetchTemplates()
   614→  }, [fetchTemplates])
   615→
   616→  /* ---- Handlers ---- */
   617→  const handleSelectTemplate = useCallback((template: Template) => {
   618→    setSelectedTemplate(template)
   619→    setDialogOpen(true)
   620→  }, [])
   621→
   622→  const handleDialogClose = useCallback((open: boolean) => {
   623→    setDialogOpen(open)
   624→    if (!open) {
   625→      // Keep template reference for animation but can clear if desired
   626→    }
   627→  }, [])
   628→
   629→  const hasActiveFilters =
   630→    searchInput.trim() !== '' ||
   631→    category !== 'all' ||
   632→    legalArea !== 'all' ||
   633→    audience !== 'all' ||
   634→    priceFilter !== 'all'
   635→
   636→  /* ---- Render ---- */
   637→  return (
   638→    <main className="min-h-screen bg-background">
   639→      {/* Visitor header bar */}
   640→      {isVisitor && (
   641→        <div className="border-b border-border bg-card/80 backdrop-blur-sm">
   642→          <div className="mx-auto flex max-w-7xl items-center justify-between px-4 py-3 sm:px-6 lg:px-8">
   643→            <div className="flex items-center gap-4">
   644→              <div className="flex items-center gap-2">
   645→                {/* eslint-disable-next-line @next/next/no-img-element */}
   646→                <img src="/logo.png" alt="CopyExpress" className="h-7 w-auto rounded" />
   647→                <span className="text-sm font-bold text-foreground">CopyExpress</span>
   648→              </div>
   649→            </div>
   650→            <div className="flex items-center gap-2">
   651→              <Button
   652→                variant="ghost"
   653→                size="sm"
   654→                onClick={() => {
   655→                  const t = theme || 'light'
   656→                  if (t === 'light') setTheme('dark')
   657→                  else if (t === 'dark') setTheme('warm')
   658→                  else setTheme('light')
   659→                }}
   660→                className="text-muted-foreground hover:text-foreground h-8 w-8 p-0"
   661→              >
   662→                {(theme || 'light') === 'dark' ? <Sun className="h-4 w-4" /> : (theme || 'light') === 'warm' ? <Palette className="h-4 w-4" /> : <Moon className="h-4 w-4" />}
   663→              </Button>
   664→              <Button
   665→                variant="outline"
   666→                size="sm"
   667→                onClick={() => { showLoginPage() }}
   668→                className="border-primary/30 text-primary text-xs font-medium hover:bg-primary/10"
   669→              >
   670→                <LogIn className="mr-1.5 h-3.5 w-3.5" />
   671→                Iniciar sesion
   672→              </Button>
   673→            </div>
   674→          </div>
   675→        </div>
   676→      )}
   677→      <div className="mx-auto max-w-7xl space-y-6 px-4 py-8 sm:px-6 lg:px-8">
   678→        {/* ---------- Header ---------- */}
   679→        <div>
   680→          <h1 className="text-2xl font-bold tracking-tight text-foreground sm:text-3xl">
   681→            {isVisitor ? 'Genera tu Documento Legal' : 'Catálogo de Plantillas'}
   682→          </h1>
   683→          <p className="mt-1 text-sm text-foreground/50">
   684→            {isVisitor
   685→              ? 'Selecciona una plantilla, completa los campos y envia tu solicitud por WhatsApp.'
   686→              : 'Explora y genera documentos legales colombianos con plantillas profesionales.'}
   687→          </p>
   688→        </div>
   689→
   690→        {/* ---------- Search bar ---------- */}
   691→        <div className="relative">
   692→          <Search className="absolute top-1/2 left-3 h-4 w-4 -translate-y-1/2 text-foreground/30" />
   693→          <Input
   694→            placeholder="Buscar plantillas por nombre, tipo o palabra clave..."
   695→            value={searchInput}
   696→            onChange={(e) => setSearchInput(e.target.value)}
   697→            className="h-11 w-full border-border bg-card pl-10 pr-4 text-sm text-foreground placeholder:text-foreground/30 focus-visible:border-accent/40 focus-visible:ring-primary/20"
   698→          />
   699→        </div>
   700→
   701→        {/* ---------- Filters row ---------- */}
   702→        <div className="flex flex-col gap-3 sm:flex-row sm:flex-wrap sm:items-center sm:gap-3">
   703→          {/* Category */}
   704→          <Select value={category} onValueChange={setCategory}>
   705→            <SelectTrigger
   706→              className="w-full border-border bg-card text-foreground/70 text-xs sm:w-[155px]"
   707→              size="sm"
   708→            >
   709→              <SelectValue placeholder="Categoría" />
   710→            </SelectTrigger>
   711→            <SelectContent className="border-border bg-card">
   712→              <SelectItem value="all" className="text-foreground/70 focus:bg-accent/10 focus:text-accent">
   713→                Todas
   714→              </SelectItem>
   715→              <SelectItem value="Contratos" className="text-foreground/70 focus:bg-accent/10 focus:text-accent">
   716→                Contratos
   717→              </SelectItem>
   718→              <SelectItem value="Actas" className="text-foreground/70 focus:bg-accent/10 focus:text-accent">
   719→                Actas
   720→              </SelectItem>
   721→              <SelectItem value="Derechos de Petición" className="text-foreground/70 focus:bg-accent/10 focus:text-accent">
   722→                Derechos de Petición
   723→              </SelectItem>
   724→            </SelectContent>
   725→          </Select>
   726→
   727→          {/* Legal Area */}
   728→          <Select value={legalArea} onValueChange={setLegalArea}>
   729→            <SelectTrigger
   730→              className="w-full border-border bg-card text-foreground/70 text-xs sm:w-[165px]"
   731→              size="sm"
   732→            >
   733→              <SelectValue placeholder="Área del Derecho" />
   734→            </SelectTrigger>
   735→            <SelectContent className="border-border bg-card">
   736→              <SelectItem value="all" className="text-foreground/70 focus:bg-accent/10 focus:text-accent">
   737→                Todos
   738→              </SelectItem>
   739→              <SelectItem value="Civil" className="text-foreground/70 focus:bg-accent/10 focus:text-accent">
   740→                Civil
   741→              </SelectItem>
   742→              <SelectItem value="Mercantil" className="text-foreground/70 focus:bg-accent/10 focus:text-accent">
   743→                Mercantil
   744→              </SelectItem>
   745→              <SelectItem value="Laboral" className="text-foreground/70 focus:bg-accent/10 focus:text-accent">
   746→                Laboral
   747→              </SelectItem>
   748→              <SelectItem value="Constitucional" className="text-foreground/70 focus:bg-accent/10 focus:text-accent">
   749→                Constitucional
   750→              </SelectItem>
   751→            </SelectContent>
   752→          </Select>
   753→
   754→          {/* Audience */}
   755→          <Select value={audience} onValueChange={setAudience}>
   756→            <SelectTrigger
   757→              className="w-full border-border bg-card text-foreground/70 text-xs sm:w-[155px]"
   758→              size="sm"
   759→            >
   760→              <SelectValue placeholder="Audiencia" />
   761→            </SelectTrigger>
   762→            <SelectContent className="border-border bg-card">
   763→              <SelectItem value="all" className="text-foreground/70 focus:bg-accent/10 focus:text-accent">
   764→                Todos
   765→              </SelectItem>
   766→              <SelectItem value="Profesionales" className="text-foreground/70 focus:bg-accent/10 focus:text-accent">
   767→                Profesionales
   768→              </SelectItem>
   769→              <SelectItem value="Particulares" className="text-foreground/70 focus:bg-accent/10 focus:text-accent">
   770→                Particulares
   771→              </SelectItem>
   772→            </SelectContent>
   773→          </Select>
   774→
   775→          {/* Price */}
   776→          <Select value={priceFilter} onValueChange={setPriceFilter}>
   777→            <SelectTrigger
   778→              className="w-full border-border bg-card text-foreground/70 text-xs sm:w-[135px]"
   779→              size="sm"
   780→            >
   781→              <SelectValue placeholder="Precio" />
   782→            </SelectTrigger>
   783→            <SelectContent className="border-border bg-card">
   784→              <SelectItem value="all" className="text-foreground/70 focus:bg-accent/10 focus:text-accent">
   785→                Todos
   786→              </SelectItem>
   787→              <SelectItem value="free" className="text-foreground/70 focus:bg-accent/10 focus:text-accent">
   788→                Gratuito
   789→              </SelectItem>
   790→              <SelectItem value="paid" className="text-foreground/70 focus:bg-accent/10 focus:text-accent">
   791→                De Pago
   792→              </SelectItem>
   793→            </SelectContent>
   794→          </Select>
   795→
   796→          {/* Spacer */}
   797→          <div className="flex-1" />
   798→
   799→          {/* View toggle */}
   800→          <ToggleGroup
   801→            type="single"
   802→            value={viewMode}
   803→            onValueChange={(v) => {
   804→              if (v) setViewMode(v as 'grid' | 'list')
   805→            }}
   806→            className="border-border bg-card"
   807→          >
   808→            <ToggleGroupItem
   809→              value="grid"
   810→              aria-label="Vista de cuadrícula"
   811→              className="data-[state=on]:bg-accent/15 data-[state=on]:text-accent text-foreground/40 hover:bg-secondary hover:text-foreground/60 h-8 w-8 rounded-l-md"
   812→            >
   813→              <LayoutGrid className="h-4 w-4" />
   814→            </ToggleGroupItem>
   815→            <ToggleGroupItem
   816→              value="list"
   817→              aria-label="Vista de lista"
   818→              className="data-[state=on]:bg-accent/15 data-[state=on]:text-accent text-foreground/40 hover:bg-secondary hover:text-foreground/60 h-8 w-8 rounded-r-md"
   819→            >
   820→              <List className="h-4 w-4" />
   821→            </ToggleGroupItem>
   822→          </ToggleGroup>
   823→        </div>
   824→
   825→        {/* ---------- Results count ---------- */}
   826→        <div className="flex items-center justify-between">
   827→          <p className="text-xs text-foreground/40">
   828→            {loading
   829→              ? 'Cargando...'
   830→              : `${templates.length} plantilla${templates.length !== 1 ? 's' : ''} encontrada${templates.length !== 1 ? 's' : ''}`}
   831→          </p>
   832→          {hasActiveFilters && (
   833→            <button
   834→              onClick={() => {
   835→                setSearchInput('')
   836→                setCategory('all')
   837→                setLegalArea('all')
   838→                setAudience('all')
   839→                setPriceFilter('all')
   840→              }}
   841→              className="text-[11px] font-medium text-accent/70 transition-colors hover:text-accent"
   842→            >
   843→              Limpiar filtros
   844→            </button>
   845→          )}
   846→        </div>
   847→
   848→        {/* ---------- Content ---------- */}
   849→        {loading ? (
   850→          viewMode === 'grid' ? (
   851→            <div className="grid grid-cols-1 gap-4 md:grid-cols-2 xl:grid-cols-3">
   852→              {Array.from({ length: 6 }).map((_, i) => (
   853→                <SkeletonCard key={i} />
   854→              ))}
   855→            </div>
   856→          ) : (
   857→            <div className="flex flex-col gap-3">
   858→              {Array.from({ length: 6 }).map((_, i) => (
   859→                <SkeletonRow key={i} />
   860→              ))}
   861→            </div>
   862→          )
   863→        ) : templates.length === 0 ? (
   864→          <EmptyState hasFilters={hasActiveFilters} />
   865→        ) : viewMode === 'grid' ? (
   866→          <div className="grid grid-cols-1 gap-4 md:grid-cols-2 xl:grid-cols-3">
   867→            {templates.map((t) => (
   868→              <TemplateCard key={t.id} template={t} onSelect={handleSelectTemplate} />
   869→            ))}
   870→          </div>
   871→        ) : (
   872→          <div className="flex flex-col gap-3">
   873→            {templates.map((t) => (
   874→              <TemplateRow key={t.id} template={t} onSelect={handleSelectTemplate} />
   875→            ))}
   876→          </div>
   877→        )}
   878→
   879→        {/* ---------- Bottom CTA (logged-in only) ---------- */}
   880→        {!isVisitor && (
   881→        <div className="flex items-center justify-center pt-4">
   882→          <button
   883→            className="group flex items-center gap-3 rounded-xl border border-accent/15 bg-card/60 px-6 py-4 text-left backdrop-blur-sm transition-all hover:border-accent/30 hover:bg-accent/5 hover:shadow-lg hover:shadow-accent/5"
   884→          >
   885→            <div className="flex h-11 w-11 shrink-0 items-center justify-center rounded-lg bg-accent/15 text-accent transition-colors group-hover:bg-accent/25">
   886→              <Plus className="h-5 w-5" />
   887→            </div>
   888→            <div className="min-w-0 flex-1">
   889→              <p className="text-sm font-semibold text-foreground/80 group-hover:text-foreground">
   890→                No encuentras lo que buscas?
   891→              </p>
   892→              <p className="text-xs text-foreground/40">
   893→                Solicita un documento personalizado y nuestro equipo lo preparara para ti.
   894→              </p>
   895→            </div>
   896→            <ArrowRight className="h-4 w-4 shrink-0 text-foreground/20 transition-transform group-hover:translate-x-0.5 group-hover:text-accent" />
   897→          </button>
   898→        </div>
   899→        )}
   900→      </div>
   901→
   902→      {/* ---------- Detail Dialog ---------- */}
   903→      <TemplateDetailDialog
   904→        template={selectedTemplate}
   905→        open={dialogOpen}
   906→        onOpenChange={handleDialogClose}
   907→      />
   908→
   909→      {/* Visitor: Publications Panel (anchored left) */}
   910→      {isVisitor && <PublicationsPanel />}
   911→
   912→      {/* Visitor: Footer */}
   913→      {isVisitor && (
   914→        <footer className="border-t border-border bg-card/60 backdrop-blur-sm py-4">
   915→          <div className="mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
   916→            <p className="text-center text-xs text-muted-foreground">
   917→              CopyDocs · Diseñado con ❤️ por CopyExpress · Línea de soporte +573226575422
   918→            </p>
   919→          </div>

   922→    </main>
   923→  )
