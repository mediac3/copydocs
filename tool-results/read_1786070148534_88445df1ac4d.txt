   122→function formatDate(iso: string): string {
   123→  try {
   124→    const d = new Date(iso)
   125→    return d.toLocaleDateString('es-CO', {
   126→      day: '2-digit',
   127→      month: 'short',
   128→      year: 'numeric',
   129→    })
   130→  } catch {
   131→    return '—'
   132→  }
   133→}

   135→function useDebounce<T>(value: T, delayMs: number): T {
   136→  const [debounced, setDebounced] = useState(value)
   137→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   138→
   139→  useEffect(() => {
   140→    if (timerRef.current) clearTimeout(timerRef.current)
   141→    timerRef.current = setTimeout(() => setDebounced(value), delayMs)
   142→    return () => {
   143→      if (timerRef.current) clearTimeout(timerRef.current)
   144→    }
   145→  }, [value, delayMs])
   146→
   147→  return debounced
   148→}

   347→export default function DocumentsPage() {
   348→  const { user, setCurrentPage, startWizard } = useAppStore()
   349→
   350→  /* ---- State ---- */
   351→  const [activeTab, setActiveTab] = useState<TabFilter>('all')
   352→  const [searchQuery, setSearchQuery] = useState('')
   353→  const debouncedSearch = useDebounce(searchQuery, 300)
   354→  const [documents, setDocuments] = useState<Document[]>([])
   355→  const [loading, setLoading] = useState(true)
   356→
   357→  /* Delete dialog */
   358→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false)
   359→  const [docToDelete, setDocToDelete] = useState<Document | null>(null)
   360→  const [deleting, setDeleting] = useState(false)
   361→
   362→  /* Action loading helpers */
   363→  const [actionLoading, setActionLoading] = useState<string | null>(null)
   364→
   365→  /* ---- Fetch documents ---- */
   366→  const fetchDocuments = useCallback(async () => {
   367→    if (!user) return
   368→    setLoading(true)
   369→    try {
   370→      const params = new URLSearchParams()
   371→      params.set('status', activeTab)
   372→      if (debouncedSearch.trim()) {
   373→        params.set('search', debouncedSearch.trim())
   374→      }
   375→      const res = await fetch(`/api/documents?${params.toString()}`, {
   376→        headers: { 'x-user-id': user.id },
   377→      })
   378→      if (!res.ok) throw new Error('Error fetching documents')
   379→      const data = await res.json()
   380→      setDocuments(Array.isArray(data) ? data : data.documents || [])
   381→    } catch {
   382→      toast.error('Error al cargar documentos')
   383→    } finally {
   384→      setLoading(false)
   385→    }
   386→  }, [user, activeTab, debouncedSearch])
   387→
   388→  useEffect(() => {
   389→    fetchDocuments()
   390→  }, [fetchDocuments])
   391→
   392→  /* ---- Filtered documents for client-side (API may not support search) ---- */
   393→  const filteredDocuments = documents.filter((doc) => {
   394→    if (!debouncedSearch.trim()) return true
   395→    const q = debouncedSearch.toLowerCase()
   396→    return (
   397→      doc.title.toLowerCase().includes(q) ||
   398→      doc.template?.name?.toLowerCase().includes(q) ||
   399→      doc.template?.category?.toLowerCase().includes(q)
   400→    )
   401→  })
   402→
   403→  /* ---- Action handlers ---- */
   404→
   405→  const handleEdit = (doc: Document) => {
   406→    // Edit opens wizard with existing document
   407→    startWizard(doc.template?.name || '', doc.id)
   408→  }
   409→
   410→  const handleDuplicate = async (doc: Document) => {
   411→    if (!user) return
   412→    setActionLoading(doc.id)
   413→    try {
   414→      const res = await fetch('/api/documents', {
   415→        method: 'POST',
   416→        headers: {
   417→          'Content-Type': 'application/json',
   418→          'x-user-id': user.id,
   419→        },
   420→        body: JSON.stringify({
   421→          duplicateFromId: doc.id,
   422→        }),
   423→      })
   424→      if (!res.ok) throw new Error('Error duplicando documento')
   425→      const newDoc = await res.json()
   426→      toast.success('Documento duplicado', {
   427→        description: `"${doc.title}" se copió como borrador.`,
   428→      })
   429→      // If we're on 'all' or 'draft', refetch to show new item
   430→      if (activeTab === 'all' || activeTab === 'draft') {
   431→        fetchDocuments()
   432→      }
   433→    } catch {
   434→      toast.error('Error al duplicar', {
   435→        description: 'No se pudo crear la copia del documento.',
   436→      })
   437→    } finally {
   438→      setActionLoading(null)
   439→    }
   440→  }
   441→
   442→  const handleDownloadPdf = async (doc: Document) => {
   443→    setActionLoading(doc.id)
   444→    try {
   445→      const res = await fetch(
   446→        `/api/documents/export?id=${doc.id}&format=pdf`,
   447→        {
   448→          headers: { 'x-user-id': user!.id },
   449→        }
   450→      )
   451→      if (!res.ok) throw new Error('Error generando PDF')
   452→      const blob = await res.blob()
   453→      const url = URL.createObjectURL(blob)
   454→      const a = document.createElement('a')
   455→      a.href = url
   456→      a.download = `${doc.title}.pdf`
   457→      document.body.appendChild(a)
   458→      a.click()
   459→      a.remove()
   460→      URL.revokeObjectURL(url)
   461→      toast.success('PDF descargado', {
   462→        description: `"${doc.title}.pdf" se descargó correctamente.`,
   463→      })
   464→    } catch {
   465→      toast.error('Error al descargar PDF')
   466→    } finally {
   467→      setActionLoading(null)
   468→    }
   469→  }
   470→
   471→  const handleDownloadDocx = async (doc: Document) => {
   472→    setActionLoading(doc.id)
   473→    try {
   474→      const res = await fetch(
   475→        `/api/documents/export?id=${doc.id}&format=docx`,
   476→        {
   477→          headers: { 'x-user-id': user!.id },
   478→        }
   479→      )
   480→      if (!res.ok) throw new Error('Error generando DOCX')
   481→      const blob = await res.blob()
   482→      const url = URL.createObjectURL(blob)
   483→      const a = document.createElement('a')
   484→      a.href = url
   485→      a.download = `${doc.title}.docx`
   486→      document.body.appendChild(a)
   487→      a.click()
   488→      a.remove()
   489→      URL.revokeObjectURL(url)
   490→      toast.success('DOCX descargado', {
   491→        description: `"${doc.title}.docx" se descargó correctamente.`,
   492→      })
   493→    } catch {
   494→      toast.error('Error al descargar DOCX')
   495→    } finally {
   496→      setActionLoading(null)
   497→    }
   498→  }
   499→
   500→  const handleArchive = async (doc: Document) => {
   501→    if (!user) return
   502→    const newStatus: DocumentStatus =
   503→      doc.status === 'archived' ? 'draft' : 'archived'
   504→    setActionLoading(doc.id)
   505→    try {
   506→      const res = await fetch('/api/documents', {
   507→        method: 'PUT',
   508→        headers: {
   509→          'Content-Type': 'application/json',
   510→          'x-user-id': user.id,
   511→        },
   512→        body: JSON.stringify({
   513→          id: doc.id,
   514→          status: newStatus,
   515→        }),
   516→      })
   517→      if (!res.ok) throw new Error('Error actualizando estado')
   518→      toast.success(
   519→        newStatus === 'archived'
   520→          ? 'Documento archivado'
   521→          : 'Documento desarchivado',
   522→        {
   523→          description: `"${doc.title}" ${
   524→            newStatus === 'archived' ? 'se movió a archivados' : 'volvió a borradores'
   525→          }.`,
   526→        }
   527→      )
   528→      fetchDocuments()
   529→    } catch {
   530→      toast.error('Error al actualizar estado')
   531→    } finally {
   532→      setActionLoading(null)
   533→    }
   534→  }
   535→
   536→  const handleDeleteRequest = (doc: Document) => {
   537→    setDocToDelete(doc)
   538→    setDeleteDialogOpen(true)
   539→  }
   540→
   541→  const handleDeleteConfirm = async () => {
   542→    if (!user || !docToDelete) return
   543→    setDeleting(true)
   544→    try {
   545→      const res = await fetch(
   546→        `/api/documents?id=${docToDelete.id}`,
   547→        {
   548→          method: 'DELETE',
   549→          headers: { 'x-user-id': user.id },
   550→        }
   551→      )
   552→      if (!res.ok) throw new Error('Error eliminando documento')
   553→      toast.success('Documento eliminado', {
   554→        description: `"${docToDelete.title}" fue eliminado permanentemente.`,
   555→      })
   556→      setDeleteDialogOpen(false)
   557→      setDocToDelete(null)
   558→      fetchDocuments()
   559→    } catch {
   560→      toast.error('Error al eliminar documento')
   561→    } finally {
   562→      setDeleting(false)
   563→    }
   564→  }
   565→
   566→  /* ---- Tab counts ---- */
   567→  const tabCounts: Record<TabFilter, number> = {
   568→    all: documents.length,
   569→    draft: documents.filter((d) => d.status === 'draft').length,
   570→    completed: documents.filter((d) => d.status === 'completed').length,
   571→    archived: documents.filter((d) => d.status === 'archived').length,
   572→  }
   573→
   574→  /* ---- Render ---- */
   575→  return (
   576→    <main className="min-h-screen bg-[#0A1628]">
   577→      <div className="mx-auto max-w-7xl space-y-6 px-4 py-8 sm:px-6 lg:px-8">
   578→        {/* ---------- Header ---------- */}
   579→        <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   580→          <div>
   581→            <h1 className="text-2xl font-bold tracking-tight text-white sm:text-3xl">
   582→              Mis Documentos
   583→            </h1>
   584→            <p className="mt-1 text-sm text-white/50">
   585→              Gestiona, edita y descarga tus documentos legales.
   586→            </p>
   587→          </div>
   588→          <Button
   589→            onClick={() => setCurrentPage('catalog')}
   590→            className="self-start bg-[#C9A94E] text-[#0A1628] font-semibold hover:bg-[#D4B965] focus-visible:ring-[#C9A94E]/50"
   591→          >
   592→            <Plus className="mr-2 h-4 w-4" />
   593→            Nuevo Documento
   594→          </Button>
   595→        </div>
   596→
   597→        {/* ---------- Search + Tabs ---------- */}
   598→        <div className="space-y-4">
   599→          {/* Search bar */}
   600→          <div className="relative max-w-md">
   601→            <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-white/30" />
   602→            <Input
   603→              placeholder="Buscar por título, plantilla o categoría…"
   604→              value={searchQuery}
   605→              onChange={(e) => setSearchQuery(e.target.value)}
   606→              className="border-white/10 bg-[#0F1D32]/80 pl-10 text-white placeholder:text-white/30 focus-visible:ring-[#C9A94E]/30 focus-visible:border-[#C9A94E]/40"
   607→            />
   608→          </div>
   609→
   610→          {/* Tabs */}
   611→          <Tabs
   612→            value={activeTab}
   613→            onValueChange={(v) => setActiveTab(v as TabFilter)}
   614→          >
   615→            <TabsList className="border-white/10 bg-[#0F1D32]/80">
   616→              {(['all', 'draft', 'completed', 'archived'] as TabFilter[]).map(
   617→                (tab) => (
   618→                  <TabsTrigger
