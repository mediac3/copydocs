   170→function formatCurrency(value: number): string {
   171→  return new Intl.NumberFormat('es-CO', {
   172→    style: 'currency',
   173→    currency: 'COP',
   174→    minimumFractionDigits: 0,
   175→  }).format(value)
   176→}
   177→
   178→function formatDate(dateStr: string): string {
   179→  return new Date(dateStr).toLocaleDateString('es-CO', {
   180→    day: '2-digit',
   181→    month: 'short',
   182→    year: 'numeric',
   183→  })
   184→}
   185→
   186→function formatDateTime(dateStr: string): string {
   187→  return new Date(dateStr).toLocaleDateString('es-CO', {
   188→    day: '2-digit',
   189→    month: 'short',
   190→    year: 'numeric',
   191→    hour: '2-digit',
   192→    minute: '2-digit',
   193→  })
   194→}

   239→function ChartTooltip({ active, payload, label }: { active?: boolean; payload?: Array<{ value: number; name: string; color: string }>; label?: string }) {
   240→  if (!active || !payload?.length) return null
   241→  return (
   242→    <div className="rounded-lg border border-[#C9A94E]/20 bg-[#0A1628]/95 px-3 py-2 shadow-xl backdrop-blur-sm">
   243→      <p className="mb-1 text-xs font-semibold text-white/80 capitalize">{label}</p>
   244→      {payload.map((p, i) => (
   245→        <div key={i} className="flex items-center gap-2 text-xs">
   246→          <span className="inline-block h-2 w-2 rounded-full" style={{ backgroundColor: p.color }} />
   247→          <span className="text-white/60">{p.name}:</span>
   248→          <span className="font-medium text-white">{p.value}</span>
   249→        </div>
   250→      ))}
   251→    </div>
   252→  )
   253→}
